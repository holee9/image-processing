base = 'D:/workspace-github/xpe-pre/modules/preprocess/'


def edit(rel, pairs):
    s = open(base + rel, encoding='utf-8', newline='').read()
    crlf = '\r\n' in s
    s = s.replace('\r\n', '\n')
    for old, new in pairs:
        assert s.count(old) == 1, (rel, s.count(old), old[:110])
        s = s.replace(old, new)
    open(base + rel, 'w', encoding='utf-8', newline='').write(s.replace('\n', '\r\n') if crlf else s)

NL = 'bugprone-easily-swappable-parameters'


def reason(sig, text):
    return (GENERIC + '\n' + sig, '// NOLINTBEGIN(%s): %s\n%s' % (NL, text, sig))
