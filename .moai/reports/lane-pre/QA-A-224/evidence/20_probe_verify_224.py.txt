"""QA-A-224 probe: xpe_verify_gain and xpe_verify_pipeline on exact-zero-residual (perfect) corrections.
Calls the real DLL through ctypes. Argument 1 = label for the output file header."""
import ctypes, os, sys, math
import numpy as np

BIN = 'D:/workspace-github/xpe-pre/build/ci-preprocess/bin'
os.add_dll_directory(BIN)
dll = ctypes.CDLL(BIN + '/xpe_preprocess.dll')


class Buf(ctypes.Structure):
    _fields_ = [('width', ctypes.c_uint32), ('height', ctypes.c_uint32), ('bitsAllocated', ctypes.c_uint32),
                ('bitsStored', ctypes.c_uint32), ('format', ctypes.c_int32), ('data', ctypes.c_void_p),
                ('dataSize', ctypes.c_size_t)]


class Metrics(ctypes.Structure):
    _fields_ = [('dark_bias', ctypes.c_double), ('dsnu', ctypes.c_double), ('residual_noise', ctypes.c_double),
                ('prnu_before', ctypes.c_double), ('prnu_after', ctypes.c_double), ('flatness_pct', ctypes.c_double),
                ('gain_coverage', ctypes.c_double), ('invalid_gain_count', ctypes.c_uint32),
                ('defect_count', ctypes.c_uint32), ('defect_density', ctypes.c_double),
                ('correction_error', ctypes.c_double), ('snr_improvement_db', ctypes.c_double),
                ('overall_pass', ctypes.c_bool), ('dark_reduction_db', ctypes.c_double),
                ('dsnu_adu', ctypes.c_double), ('measured_mask', ctypes.c_uint32)]


assert Metrics.gain_coverage.offset == 48 and Metrics.snr_improvement_db.offset == 80 and Metrics.overall_pass.offset == 88

W = H = 64
keep = []


def buf(arr, fmt):
    keep.append(arr)
    bits = 16 if fmt == 0 else 32
    return Buf(W, H, bits, bits, fmt, arr.ctypes.data, arr.nbytes)


dll.xpe_verify_gain.argtypes = [ctypes.POINTER(Buf)] * 3 + [ctypes.c_int, ctypes.POINTER(Metrics)]
dll.xpe_verify_pipeline.argtypes = [ctypes.POINTER(Buf), ctypes.POINTER(Buf), ctypes.c_void_p, ctypes.POINTER(Metrics)]
dll.xpe_verify_gain.restype = dll.xpe_verify_pipeline.restype = ctypes.c_int


def fmt(x):
    return 'inf' if math.isinf(x) and x > 0 else '-inf' if math.isinf(x) else 'nan' if math.isnan(x) else '%.4f' % x


def gain(label, before_f, after_f):
    before = np.ascontiguousarray(before_f.astype(np.uint16)); after = np.ascontiguousarray(after_f.astype(np.float32))
    one = np.ones((H, W), np.float32)
    m = Metrics()
    rc = dll.xpe_verify_gain(ctypes.byref(buf(before, 0)), ctypes.byref(buf(after, 1)), ctypes.byref(buf(one, 1)), 0, ctypes.byref(m))
    print('gain     %-52s rc=%d prnu %.4f%% -> %.6f%%  snr_improvement_db=%s  overall_pass=%s mask=%d'
          % (label, rc, m.prnu_before, m.prnu_after, fmt(m.snr_improvement_db), m.overall_pass, m.measured_mask))


def pipe(label, raw_f, final_f):
    raw = np.ascontiguousarray(raw_f.astype(np.uint16)); final = np.ascontiguousarray(final_f.astype(np.float32))
    m = Metrics()
    rc = dll.xpe_verify_pipeline(ctypes.byref(buf(raw, 0)), ctypes.byref(buf(final, 1)), None, ctypes.byref(m))
    print('pipeline %-52s rc=%d snr_improvement_db=%s  overall_pass=%s mask=%d' % (label, rc, fmt(m.snr_improvement_db), m.overall_pass, m.measured_mask))


print('probe', sys.argv[1] if len(sys.argv) > 1 else '', '(64x64, gain map 1.0, semantics UNKNOWN)')
rng = np.random.default_rng(7)
noisy = 2000.0 * (1.0 + rng.normal(0, 0.05, (H, W)))
flat = np.full((H, W), 2000.0)
tiny = 2000.0 * (1.0 + rng.normal(0, 1e-6, (H, W)))
mild = 2000.0 * (1.0 + rng.normal(0, 0.005, (H, W)))
same = 2000.0 * (1.0 + rng.normal(0, 0.05, (H, W)))
gain('G1 noisy 5% -> perfectly flat (residual exactly 0)', noisy, flat)
gain('G2 noisy 5% -> 0.0001% noise (control)', noisy, tiny)
gain('G3 noisy 5% -> 0.5% noise (control)', noisy, mild)
gain('G4 noisy 5% -> noisy 5% (no improvement)', noisy, same)
gain('G5 uniform -> uniform (both flat)', flat, flat)
gain('G6 uniform -> noisy 5% (made worse)', flat, noisy)
pipe('P1 noisy 5% -> perfectly flat', noisy, flat)
pipe('P2 noisy 5% -> 0.0001% noise (control)', noisy, tiny)
pipe('P3 noisy 5% -> 0.5% noise (control)', noisy, mild)
pipe('P4 noisy 5% -> noisy 5% (no improvement)', noisy, same)
pipe('P5 uniform -> uniform (both flat)', flat, flat)
pipe('P6 uniform -> noisy 5% (made worse)', flat, noisy)
