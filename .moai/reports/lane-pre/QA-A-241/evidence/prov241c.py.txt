import sys
import numpy as np

R = 'D:/workspace-github/image-processing/'
N = 3072


def load(p):
    return np.fromfile(p, dtype='<u2').reshape(N, N)


def box_hp(img, w=9):
    f = img.astype(np.float64)
    c = np.cumsum(np.cumsum(np.pad(f, ((w // 2 + 1, w // 2), (w // 2 + 1, w // 2)), mode='reflect'), 0), 1)
    s = c[w:, w:] - c[:-w, w:] - c[w:, :-w] + c[:-w, :-w]
    return f - s / (w * w)


def corr(a, b):
    return float(np.corrcoef(a.ravel(), b.ravel())[0, 1])


def rcorr(a, b):
    # robust: Spearman-like on clipped (5 robust sigma) values
    sa, sb = 1.4826 * np.median(np.abs(a - np.median(a))), 1.4826 * np.median(np.abs(b - np.median(b)))
    return corr(np.clip(a, -5 * sa, 5 * sa), np.clip(b, -5 * sb, 5 * sb))


wrist = box_hp(load(R + 'gui/ImageProcTest/fixtures/gui-s0/raw/wrist_lat_3072x3072.raw'))
C6 = R + 'tests/test_data/CalData_6/'
refs = {'C6 dark': box_hp(load(C6 + 'dark.raw'))}
for i in range(1, 7):
    refs['C6 bright0%d' % i] = box_hp(load(C6 + 'bright0%d.raw' % i))
A = R + 'tests/test_data/calibration_cases/'
refs['aed(HAD1717MC) dark'] = box_hp(load(A + 'aed_shock_had1717mc/calibration/cDark.raw'))
refs['aed(HAD1717MC) image 230811_img'] = box_hp(load(A + 'aed_shock_had1717mc/images/230811_img.raw'))
refs['aed(HAD1717MC) shock image'] = box_hp(load(A + 'aed_shock_had1717mc/images/230811_img13_shock.raw'))
for f in ('corner_blemish_16j20b3/calibration/CalSet02_02874.raw', 'corner_blemish_17a06b1/calibration/CalSet01_02559.raw', 'auradr_release_line_trg/calibration/CalSet00_00001.raw', 'auradr_release_line_trg/calibration/cBr_00001A.raw'):
    refs[f.split('/')[0] + ' ' + f.split('/')[-1]] = box_hp(load(A + f))

out = []
P = out.append
P('== wrist_lat vs references: correlation of 9x9 box high-passed images over all 9,437,184 pixels (clipped at 5 robust sigma)')
P('(a reference from the same detector shares its fixed pattern; a reference from another detector does not)')
for k, v in refs.items():
    P('%-52s %.4f' % (k, rcorr(wrist, v)))
P('')
P('== the same measure between references of the SAME detector and of DIFFERENT detectors (what a match looks like)')
P('C6 bright02 ~ C6 bright03 (same detector)        %.4f' % rcorr(refs['C6 bright02'], refs['C6 bright03']))
P('C6 bright01 ~ C6 bright05 (same detector)        %.4f' % rcorr(refs['C6 bright01'], refs['C6 bright05']))
P('C6 bright02 ~ aed dark (different detector)      %.4f' % rcorr(refs['C6 bright02'], refs['aed(HAD1717MC) dark']))
P('C6 bright02 ~ aed image (different detector)     %.4f' % rcorr(refs['C6 bright02'], refs['aed(HAD1717MC) image 230811_img']))
P('aed image ~ aed dark (same detector)             %.4f' % rcorr(refs['aed(HAD1717MC) image 230811_img'], refs['aed(HAD1717MC) dark']))
P('')
P('== position null: the wrist image shifted against the C6 bright02 by a few pixels (a real shared pixel pattern must vanish)')
b2 = refs['C6 bright02']
for dy, dx in ((0, 0), (0, 1), (1, 0), (0, 3), (3, 3), (100, 0)):
    P('shift (dy %3d, dx %3d): %.4f' % (dy, dx, rcorr(np.roll(np.roll(wrist, dy, 0), dx, 1), b2)))
open(sys.argv[1], 'w', encoding='utf-8').write('\n'.join(out) + '\n')
print('\n'.join(out))
