import numpy as np

W = R + 'gui/ImageProcTest/fixtures/gui-s0/raw/wrist_lat_3072x3072.raw'
N = 3072


def load(p, dt='<u2'):
    return np.fromfile(p, dtype=dt).reshape(N, N)


def sha(p):
    return hashlib.sha256(open(p, 'rb').read()).hexdigest().upper()


wrist = load(W)
sets = {
    'CalData_6 (wrist suspect)': R + 'tests/test_data/CalData_6/dark.raw',
    'aed_shock_had1717mc': R + 'tests/test_data/calibration_cases/aed_shock_had1717mc/calibration/cDark.raw',
    'auradr_release_line_trg': R + 'tests/test_data/calibration_cases/auradr_release_line_trg/calibration/cDark.raw',
}
darks = {k: load(v) for k, v in sets.items()}
rng = np.random.default_rng(241)
out = []
P = out.append

P('== (a) size, bit depth, byte order')

P('file bytes %d = 3072*3072*2 -> %s' % (len(b), len(b) == N * N * 2))
P('sha256 %s' % sha(W))
le = wrist.astype(np.int64)
be = wrist.byteswap().astype(np.int64)
dle = np.abs(np.diff(le[:, :], axis=1)).mean()
dbe = np.abs(np.diff(be[:, :], axis=1)).mean()
P('mean |horizontal neighbour difference|: as little-endian %.2f, as big-endian %.2f -> %s' % (dle, dbe, 'little-endian' if dle < dbe else 'big-endian'))
P('value range %d..%d, max needs %d bits; values >= 65535: %d; zeros: %d' % (wrist.min(), wrist.max(), int(wrist.max()).bit_length(), int((wrist >= 65535).sum()), int((wrist == 0).sum())))

    dd = np.abs(np.diff(d.astype(np.int64), axis=1)).mean()

    P('control %-28s: LE diff %.2f, BE diff %.2f, range %d..%d' % (k, dd, ddb, d.min(), d.max()))


def hp(v, w=33):
    k = np.ones(w) / w
    pad = w // 2
    vp = np.pad(v, pad, mode='reflect')
    return v - np.convolve(vp, k, mode='valid')


def prof_corr(a, b, axis):
    pa = hp(a.astype(np.float64).mean(axis=axis))
    pb = hp(b.astype(np.float64).mean(axis=axis))
    return float(np.corrcoef(pa, pb)[0, 1])


P('')
P('== (b) fixed-pattern correlation of high-passed row / column mean profiles (33-sample moving average removed)')
P('image: wrist_lat. rows = mean over columns per row (one value per row), columns likewise')

    P('wrist ~ dark of %-28s row corr %.3f   column corr %.3f' % (k, prof_corr(wrist, d, 1), prof_corr(wrist, d, 0)))

P('== (c) outliers at the CalData_6 defect-map positions')


def outlier_mask(img):
    f = img.astype(np.float32)
    # median of the 4 neighbours (cheap, robust enough for isolated pixels)
    nb = np.stack([np.roll(f, 1, 0), np.roll(f, -1, 0), np.roll(f, 1, 1), np.roll(f, -1, 1)])
    med = np.median(nb, axis=0)
    d = f - med
    sig = 1.4826 * np.median(np.abs(d - np.median(d)))
    return np.abs(d) > 5 * sig, float(sig)


bp = np.fromfile(R + 'tests/test_data/CalData_6/BPMap.map', dtype=np.uint8).reshape(N, N) != 0

    om, sig = outlier_mask(img)
    hit = om.ravel()[idx].mean()

    P('%-38s outliers (>5 robust sigma, sigma %.1f) at BP positions %.1f%%, at random positions: mean %.1f%%, max of 200 draws %.1f%%' % (name, sig, 100 * hit, 100 * np.mean(rnd), 100 * np.max(rnd)))
