"""QA-A-254 step 0a (leader decision on QA-A-252h): the pixels covered by two or more planted items are scored against a COMPOSITE truth.
- The response of the pixel is the PRODUCT of the responses of the covering response items (gain_loss, efficiency_loss: the item level); a product below 0.45 or above 1.5 is a response defect (the same rule as for one item).
- The noise axis is the measured verdict of the covering items, unchanged (an item whose noise verdict is a defect makes the pixel a defect, one whose verdict is undetermined makes it undetermined unless a defect applies).
- The pixel is a defect when either axis says so (a covering item that is a defect by itself, or a response product outside 0.45..1.5, or a noise defect).
- score --overlap composite switches it on; without the option nothing changes (the earlier scores stay reproducible).
- One extra line reports the overlapped pixels and their score: how many are defect / undetermined / normal by the composite, and how many of each the map flags."""

        s = s.replace(old, new)

/**
 * Composite truth of the overlapped pixels (QA-A-254, the decision on QA-A-252h): the response of a pixel covered by several response items is the product of their responses; the
 * noise axis is the measured verdict of the covering items; the pixel is a defect when either axis says so. `cover` receives, per pixel, 0 (not overlapped), 1 normal, 2 defect, 3
 * undetermined. The truth masks are updated: a composite defect is a true defect (truthReal), an undetermined pixel joins the undetermined zone.
 */
void applyOverlapComposite(Truth* t, std::vector<uint8_t>* undetMask, std::vector<uint8_t>* overlapClass, OverlapReport* rep) {
    const size_t n = static_cast<size_t>(t->w) * t->h;
    std::vector<uint8_t> count(n, 0), anyDefect(n, 0), anyUndet(n, 0);
    std::vector<float>   product(n, 1.0f);
    for (const Item& it : t->items) {
        const bool respItem = it.kind == "gain_loss" || it.kind == "efficiency_loss";
        forItemPixels(it, t->w, [&](size_t i) {
            if (count[i] < 255) ++count[i];
            if (respItem) product[i] *= static_cast<float>(it.level);
            if (it.cls == "undetermined") anyUndet[i] = 1;
            else if (it.cls != "control") anyDefect[i] = 1;
        });
    }
    overlapClass->assign(n, 0);
    *rep = OverlapReport{};
    for (size_t i = 0; i < n; ++i) {
        if (count[i] < 2) continue;
        ++rep->pixels;
        const bool byProduct = product[i] < 0.45f - 1e-6f || product[i] > 1.5f + 1e-6f;
        if (anyDefect[i] || byProduct) {
            (*overlapClass)[i] = 2;
            ++rep->defect;
            if (!anyDefect[i]) {
                ++rep->promotedByProduct;
                t->truthReal[i] = 1;
            }
        } else if (anyUndet[i]) {
            (*overlapClass)[i] = 3;
            ++rep->undetermined;
            (*undetMask)[i] = 1;
        } else {
            (*overlapClass)[i] = 1;
            ++rep->normal;
        }
    }
}
