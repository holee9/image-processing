  asan-tests:
    # QA-A-234 (#253): the common and preprocess tests under MSVC AddressSanitizer. Any AddressSanitizer report
    # aborts the test process, so a report fails this job. It is a gate from the first run: the local run of
    # 2026-10-03 had 0 reports.
    needs: detect-lane
    if: needs.detect-lane.outputs.is_main == 'true' || needs.detect-lane.outputs.lane != 'unknown'
    runs-on: windows-2025
    timeout-minutes: 45

    steps:
      - name: Checkout
        uses: actions/checkout@v6

      - name: Ensure Ninja is available
        shell: pwsh
        run: |
          if (-not (Get-Command ninja -ErrorAction SilentlyContinue)) {
            choco install ninja -y --no-progress
          }
          ninja --version

      - name: Configure MSVC developer environment
        shell: pwsh
        run: pwsh -NoProfile -ExecutionPolicy Bypass -File .\tools\ci\Use-MsvcDevShell.ps1 -PersistForGitHubActions

      - name: Check the AddressSanitizer runtime is reachable
        shell: pwsh
        run: |
          # The sanitized test executables load clang_rt.asan_dynamic-x86_64.dll from the MSVC bin directory that
          # the developer shell puts on PATH. If the image has no such file, say so here instead of failing later
          # with a loader error.
          $dll = Get-Command clang_rt.asan_dynamic-x86_64.dll -ErrorAction SilentlyContinue
          if (-not $dll) { Write-Error 'clang_rt.asan_dynamic-x86_64.dll is not on PATH: this runner image has no MSVC AddressSanitizer runtime'; exit 1 }
          Write-Host ("ASan runtime: {0}" -f $dll.Source)

      - name: Configure CMake (ASan)
        shell: pwsh
        run: cmake --preset ci-asan

      - name: Build (ASan)
        shell: pwsh
        run: cmake --build --preset ci-asan --parallel

      - name: Run tests under AddressSanitizer
        shell: pwsh
        run: |
          # Three tests are EXCLUDED, each for a stated reason (QA-A-234 M1, measured locally 2026-10-03: they fail
          # under AddressSanitizer with no AddressSanitizer report -- the instrumentation changes what they measure):
          #   EnduranceTest.LoadCycles_ControlLeakIsCaught
          #   XpePreprocessEndurance.ControlLeakIsCaught
          #       controls that prove a leak detector works by measuring heap growth; the sanitizer's allocator
          #       changes the heap numbers, so the control does not fire
          #   RuntimeDetectionPerformanceGateTest.Frame3072SquaredWithinMachineRatio
          #       a timing gate (54,969 ms under ASan); a time budget means nothing under instrumentation
          # Every other test runs. The names are one regular expression for ctest -E.
          $exclude = 'ControlLeakIsCaught|Frame3072SquaredWithinMachineRatio'
          ctest --test-dir build/ci-asan --build-config RelWithDebInfo --output-on-failure -E $exclude
          if ($LASTEXITCODE -ne 0) { exit $LASTEXITCODE }

      - name: Upload ASan test results
        if: always()
        uses: actions/upload-artifact@v6
        with:
          name: asan-test-results
          if-no-files-found: ignore
          retention-days: 14
          path: build/ci-asan/Testing/
