import hashlib, os, subprocess, time

ROOT = 'D:/workspace-github/xpe-pre/'
SRC = ROOT + 'modules/preprocess/src/xpe_bpm_v2.cpp'
LOG = ROOT + '.moai/reports/lane-pre/QA-A-252/evidence/41_mutation_log.txt'
BAT = 'C:/Users/drake/AppData/Local/Temp/claude/D--workspace-github-xpe-pre/c51d57cf-b9fc-4933-bc4a-9af9ebf82810/scratchpad/build237.bat'
EXE = ROOT + 'build/ci-preprocess/bin/xpe_preprocess_tests.exe'
FILT = 'BpmV2Test.*'
BS = chr(92)


def say(t):
    print(t)
    open(LOG, 'a', encoding='utf-8').write(t + chr(10))


def build():
    r = subprocess.run(['cmd.exe', '/c', BAT.replace('/', BS), 'xpe_preprocess_tests'], capture_output=True, text=True, encoding='cp949', errors='replace')
    return (r.stdout.strip().splitlines() or ['?'])[-1]


def run_tests():
    r = subprocess.run([EXE, '--gtest_filter=' + FILT], capture_output=True, text=True, encoding='utf-8', errors='replace', cwd=ROOT)
    ok = sorted({l.split(']')[1].split('(')[0].strip().split('.')[1] for l in r.stdout.splitlines() if l.startswith('[       OK ]')})
    bad = sorted({l.split(']')[1].split(',')[0].split('(')[0].strip().split('.')[1] for l in r.stdout.splitlines() if l.startswith('[  FAILED  ]') and '.' in l})
    return r.returncode, ok, bad


orig = open(SRC, 'rb').read()
h0 = hashlib.sha256(orig).hexdigest()[:16]
open(LOG, 'w', encoding='utf-8').write('')
say('QA-A-252 mutation check of the second-generation detector (src/xpe_bpm_v2.cpp). Source hash at the start: ' + h0)
say('Tests (BpmV2Test): Flags=FlagsEachPlantedDefectWithTheFeatureThatSeesIt Stripes=StripesThatOffsetAndGainRemoveAreNotDefects Thresholds=ThresholdsFollowFromTheTargetNumberOfFalseAlarms')
say('  Skip=TheFirstDarkFrameIsSkippedOnRequest Groups=DoseGroupsComeFromTheSignalsWhateverTheFrameOrder Lin3=TheLinearityFeatureNeedsThreeDoseGroups Refuses=RefusesInputsItCannotJudge Same=GivesTheSameMapForTheSameInput...')
say('PREDICTIONS (written before any run):')
say('  V1 first-dark skip always off                         -> red: Skip only')
say('  V2 map code 2 (dark side) written as 1                -> red: Flags only')
say('  V3 dose groups never split (ratio limit huge)         -> red: Groups, Lin3; Flags may also go red (the dead pixel is judged by the response features)')
say('  V4 union count counts every pixel                     -> red: at least one of Flags, Stripes, Thresholds (they bound or compare flagged_union); Same stays green')
say('  V5 two-sided threshold without the sqrt 2             -> red: Thresholds only')
say('  V6 the reserved field of the config is not refused    -> red: Refuses only, if that test covers the field; otherwise nothing is red and the test has a gap')
say('  V7 MAD-to-sigma factor 1.0 instead of 1.4826          -> red: at least one test that bounds false alarms (Stripes or Flags); not predicted which')
say('  every other test stays green in each mutant')

mutants = [
    ('V1', b's->skip = in->skip_first_dark != 0 ? 1u : 0u;', b's->skip = 0u;'),
    ('V2', b'((f & darkBits) ? 2 : 0)', b'((f & darkBits) ? 1 : 0)'),
    ('V3', b'signal[order[t]] / signal[order[t - 1]] > st.ratio', b'signal[order[t]] / signal[order[t - 1]] > 1.0e9'),
    ('V4', b'unionCount += f != 0;', b'unionCount += 1;'),
    ('V5', b'if (std::erfc(mid / std::sqrt(2.0)) > p) lo = mid;', b'if (std::erfc(mid) > p) lo = mid;'),
    ('V6', b'if (in->reserved != 0) return XPE_ERR_INVALID_INPUT;', b''),
    ('V7', b'constexpr double kMadToSigma = 1.4826;', b'constexpr double kMadToSigma = 1.0;'),
]
try:
    say('--- step 0: unmutated')
    say('build: ' + build())
    rc, ok, bad = run_tests()
    say('tests exit %d: %d passed, failed: %s' % (rc, len(ok), bad))
    for name, old, new in mutants:
        assert orig.count(old) == 1, (name, orig.count(old))
        say('--- ' + name)
        open(SRC, 'wb').write(orig.replace(old, new))
        b = build()
        say('build: ' + b)
        if b != 'BUILD_EXIT=0':
            say('mutant did not compile; tests NOT run (an old binary would give a false green)')
        else:
            rc, ok, bad = run_tests()
            say('tests exit %d: %d passed, FAILED: %s' % (rc, len(ok), bad))
        open(SRC, 'wb').write(orig)
        os.utime(SRC, None)
        time.sleep(1)
finally:
    open(SRC, 'wb').write(orig)
    os.utime(SRC, None)
say('--- restored: touch + rebuild')
say('source hash now: %s (start: %s)' % (hashlib.sha256(open(SRC, 'rb').read()).hexdigest()[:16], h0))
say('build: ' + build())
rc, ok, bad = run_tests()
say('tests exit %d: %d passed, failed: %s' % (rc, len(ok), bad))
