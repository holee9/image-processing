import os
base = 'D:/workspace-github/xpe-pre/modules/preprocess/'


def edit(rel, pairs, keep_crlf=True):
    p = base + rel
    raw = open(p, 'rb').read().decode('utf-8')
    crlf = '\r\n' in raw
    s = raw.replace('\r\n', '\n')
    for a, b in pairs:
        assert s.count(a) == 1, (rel, s.count(a), a[:70])
        s = s.replace(a, b)
    open(p, 'wb').write((s.replace('\n', '\r\n') if (crlf and keep_crlf) else s).encode('utf-8'))
