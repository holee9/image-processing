    private static TextBlock Label(string text) => new() { Text = text, TextWrapping = TextWrapping.Wrap, Margin = new Thickness(0, 4, 0, 4) };
    private static void Auto(DependencyObject d, string id) => AutomationProperties.SetAutomationId(d, id);

    private static UIElement Viewer(string title, ScrollViewer scroll, Grid host, Image image, Canvas overlay, ScaleTransform scale, string id)
    {

    }

    private void SetBusy(bool busy)
    {
        _busy = busy;
        _openButton.IsEnabled = !busy;
        _computeButton.IsEnabled = !busy && _validation?.Comparable == true;
    }

    /// <summary>Drops everything derived from the previous manifest: a result must never outlive the inputs it describes.</summary>
    private void Invalidate(string why)
    {
        _generation++;
        _manifest = null; _validation = null; _pixelsA = _pixelsB = null;
        _imageA.Source = _imageB.Source = null; _overlayA.Children.Clear(); _overlayB.Children.Clear();
        _computeButton.IsEnabled = false;
        foreach (var b in new[] { _applyWl, _fit, _actual, _zoomIn, _zoomOut }) b.IsEnabled = false;
        _verdict.Text = ""; _reasons.Text = ""; _engines.Text = ""; _roiText.Text = ""; _viewHint.Text = "";
        _result.Text = "";
        _state.Text = "상태: " + why + " 가져온 파일 비교이며 2G 실행이 아닙니다.";
    }

    private async Task OpenAsync()
    {
        if (_busy) return;
        var path = _manifestPath.Text.Trim();
        Invalidate("검증 중입니다.");
        var gen = _generation;
        SetBusy(true);
        try
        {
            if (!File.Exists(path)) throw new IOException("매니페스트 파일이 없습니다: " + path);
            var baseDir = Path.GetDirectoryName(Path.GetFullPath(path))!;
            (JsonNode Manifest, ComparisonContract.Validation Validation) parsed = await Task.Run(() =>
            {
                var node = JsonNode.Parse(File.ReadAllText(path)) ?? throw new IOException("매니페스트가 비어 있습니다.");
                return (node, ComparisonContract.Validate(node, baseDir));
            });
            if (gen != _generation) return;   // the path changed while the check ran: this answer is stale
            _manifest = parsed.Manifest; _baseDir = baseDir; _validation = parsed.Validation;
            ShowValidation();
            if (parsed.Validation.Comparable) await LoadDisplayAsync(gen);
        }
        catch (Exception e)
        {
            if (gen != _generation) return;
            _verdict.Text = "비교 불가 — 매니페스트를 읽지 못했습니다.";
            _reasons.Text = e.Message;
            _state.Text = "상태: 열기 실패. 가져온 파일 비교이며 2G 실행이 아닙니다. 실패는 통과가 아닙니다.";
        }
        finally { SetBusy(false); }
    }

    private void ShowValidation()
    {
        var v = _validation!;
        _verdict.Text = v.Comparable
            ? $"비교 가능 — 종류: {v.ComparisonKind}" + (v.ComparisonKind == ComparisonContract.E2eKind ? " (맵이 달라 알고리즘 효과를 분리하지 못하는 종단 비교)" : "")
            : "비교 불가 — 아래 사유 코드가 해소되기 전에는 숫자를 만들지 않습니다.";
        var lines = v.Reasons.Select(r => $"{r}: {ComparisonContract.ReasonText(r)}").ToList();
        lines.AddRange(v.Blockers.Select(b => $"{b}: {ComparisonContract.ReasonText(b)}"));
        _reasons.Text = string.Join(Environment.NewLine, lines);
        var a = ComparisonContract.Get(ComparisonContract.Get(_manifest, "engines"), "A");
        var b = ComparisonContract.Get(ComparisonContract.Get(_manifest, "engines"), "B");
        string Eng(string label, JsonNode? e)
        {
            var o = ComparisonContract.Get(e, "output");
            return $"{label}: 단계 {ComparisonContract.Str(ComparisonContract.Get(o, "stage")) ?? "?"} · 자료형 {ComparisonContract.Str(ComparisonContract.Get(o, "dtype")) ?? "?"} · 단위 {ComparisonContract.Str(ComparisonContract.Get(o, "units")) ?? "?"} · 실행 방식 {ComparisonContract.Str(ComparisonContract.Get(e, "execution")) ?? "?"} · 출력 SHA-256 {Short(v.Identities.GetValueOrDefault(label[..1] + ".output"))}";
        }
        _engines.Text = Eng("A", a) + Environment.NewLine + Eng("B", b) + Environment.NewLine
            + $"입력 SHA-256 {Short(v.Identities.GetValueOrDefault("input"))} · 보정 상태 {ComparisonContract.Str(ComparisonContract.Get(ComparisonContract.Get(ComparisonContract.Get(_manifest, "input"), "correction_state"), "value")) ?? "?"} · 맵 동일성: {v.MapEquivalence}";
        _state.Text = v.Comparable
            ? "상태: 가져온 파일의 계약 검증을 통과했습니다. 아직 수치를 계산하지 않았습니다. 2G 실행이 아닙니다."
            : "상태: 계약 검증 실패(비교 불가). 가져온 파일 비교이며 2G 실행이 아닙니다.";
    }

    private static string Short(string? sha) => string.IsNullOrEmpty(sha) ? "(없음)" : sha[..12] + "…";

    private async Task LoadDisplayAsync(int gen)
    {
        var engines = ComparisonContract.Get(_manifest, "engines");
        var outA = ComparisonContract.Get(ComparisonContract.Get(engines, "A"), "output")!;
        var outB = ComparisonContract.Get(ComparisonContract.Get(engines, "B"), "output")!;
        ComparisonContract.TryInt(ComparisonContract.Get(outA, "width"), out var w);
        ComparisonContract.TryInt(ComparisonContract.Get(outA, "height"), out var h);
        var step = (int)Math.Max(1, Math.Ceiling(Math.Sqrt((double)w * h / MaxDisplayPixels)));
        var baseDir = _baseDir;
        var (pa, pb, dw, dh) = await Task.Run(() =>
        {
            var (a, aw, ah) = ComparisonRunner.ReadForDisplay(outA, baseDir, step);
            var (b, _, _) = ComparisonRunner.ReadForDisplay(outB, baseDir, step);
            return (a, b, aw, ah);
        });
        if (gen != _generation) return;
        _pixelsA = pa; _pixelsB = pb; _displayWidth = dw; _displayHeight = dh; _step = step;
        double lo = double.PositiveInfinity, hi = double.NegativeInfinity;
        foreach (var arr in new[] { pa, pb })
            foreach (var v in arr)
                if (double.IsFinite(v)) { if (v < lo) lo = v; if (v > hi) hi = v; }
        if (!double.IsFinite(lo)) { lo = 0; hi = 1; }
        if (hi <= lo) hi = lo + 1;
        _dataMin = lo; _dataMax = hi;
        _window.Text = (hi - lo).ToString("R", CultureInfo.InvariantCulture);
        _level.Text = ((hi + lo) / 2).ToString("R", CultureInfo.InvariantCulture);
        var roi = ComparisonContract.Get(_manifest, "roi")!;
        _roiText.Text = $"ROI(매니페스트의 정수 좌표, A·B 양쪽에 같은 좌표로 표시): x={roi["x"]}, y={roi["y"]}, 가로={roi["width"]}, 세로={roi["height"]}";
        _viewHint.Text = $"표시 전용 복사본: 원본 {w}×{h}" + (step > 1 ? $", 표시는 {step}화소마다 한 점({dw}×{dh})로 줄였습니다 — 표시 좌표 = 원본 좌표 ÷ {step}" : ", 줄이지 않음")
            + ". 통계는 표시 배열이 아니라 원본 파일의 ROI 행을 직접 읽어 계산합니다. W/L·확대·줄임은 숫자에 영향을 주지 않습니다. 초기 W/L = 두 영상의 유한한 값 전체의 최솟값~최댓값(임계값을 정하지 않음).";
        foreach (var b in new[] { _applyWl, _fit, _actual, _zoomIn, _zoomOut }) b.IsEnabled = true;
        RenderBoth(useTextWl: false);
        DrawRoi(roi);
        ZoomFit();
        _computeButton.IsEnabled = true;
    }

    private void RenderBoth(bool useTextWl)
    {
        if (_pixelsA is null || _pixelsB is null) return;
        double lo = _dataMin, hi = _dataMax;
        if (useTextWl)
        {
            if (!double.TryParse(_window.Text, NumberStyles.Float, CultureInfo.InvariantCulture, out var ww) || !double.TryParse(_level.Text, NumberStyles.Float, CultureInfo.InvariantCulture, out var ll) || !(ww > 0) || !double.IsFinite(ll))
            {
                _viewHint.Text = "W/L 값이 올바르지 않습니다(Window > 0 인 숫자). 표시를 바꾸지 않았습니다.";
                return;
            }
            lo = ll - ww / 2; hi = ll + ww / 2;
        }
        _imageA.Source = ToBitmap(_pixelsA, lo, hi);
        _imageB.Source = ToBitmap(_pixelsB, lo, hi);   // the SAME lo/hi for both: one window/level
    }

    private BitmapSource ToBitmap(double[] pixels, double lo, double hi)
    {
        var bytes = new byte[pixels.Length];
        var scale = 255.0 / (hi - lo);
        for (var i = 0; i < pixels.Length; i++)
        {
            var v = pixels[i];
            bytes[i] = !double.IsFinite(v) ? (byte)0 : (byte)Math.Clamp(Math.Round((v - lo) * scale), 0, 255);
        }
        var bmp = BitmapSource.Create(_displayWidth, _displayHeight, 96, 96, PixelFormats.Gray8, null, bytes, _displayWidth);
        bmp.Freeze();
        return bmp;
    }

    private void DrawRoi(JsonNode roi)
    {
        foreach (var canvas in new[] { _overlayA, _overlayB })
        {
            canvas.Children.Clear();
            canvas.Width = _displayWidth; canvas.Height = _displayHeight;
            ComparisonContract.TryInt(roi["x"], out var x); ComparisonContract.TryInt(roi["y"], out var y);
            ComparisonContract.TryInt(roi["width"], out var rw); ComparisonContract.TryInt(roi["height"], out var rh);
            var rect = new System.Windows.Shapes.Rectangle { Width = Math.Max(1.0, (double)rw / _step), Height = Math.Max(1.0, (double)rh / _step), Stroke = Brushes.Yellow, StrokeThickness = 1.0 / Math.Max(_zoom, 1e-6), IsHitTestVisible = false };
            Canvas.SetLeft(rect, (double)x / _step); Canvas.SetTop(rect, (double)y / _step);
            canvas.Children.Add(rect);
        }
    }

    private void SetZoom(double zoom)
    {
        _zoom = Math.Clamp(zoom, 0.02, 64.0);
        _scaleA.ScaleX = _scaleA.ScaleY = _scaleB.ScaleX = _scaleB.ScaleY = _zoom;   // one scale for both
        foreach (var canvas in new[] { _overlayA, _overlayB })
            foreach (var r in canvas.Children.OfType<System.Windows.Shapes.Rectangle>()) r.StrokeThickness = 1.0 / _zoom;
    }

    private void ZoomFit()
    {
        if (_displayWidth <= 0 || _displayHeight <= 0) return;
        var vw = Math.Max(50.0, Math.Min(_scrollA.ViewportWidth > 0 ? _scrollA.ViewportWidth : 500, _scrollB.ViewportWidth > 0 ? _scrollB.ViewportWidth : 500) - 4);
        var vh = Math.Max(50.0, Math.Min(_scrollA.ViewportHeight > 0 ? _scrollA.ViewportHeight : 300, _scrollB.ViewportHeight > 0 ? _scrollB.ViewportHeight : 300) - 4);
        SetZoom(Math.Min(vw / _displayWidth, vh / _displayHeight));
    }

    private void Sync(ScrollViewer from, ScrollViewer to)
    {
        if (_syncing) return;
        _syncing = true;
        try { to.ScrollToHorizontalOffset(from.HorizontalOffset); to.ScrollToVerticalOffset(from.VerticalOffset); }
        finally { _syncing = false; }
    }

    private async Task ComputeAsync()
    {
        if (_busy || _manifest is null || _validation?.Comparable != true) return;
        var gen = _generation;
        var manifest = _manifest.DeepClone();
        var baseDir = _baseDir;
        var outRoot = _outputRoot.Text.Trim();
        SetBusy(true);
        _result.Text = "계산 중입니다. 원본 파일을 읽기만 하며 바꾸지 않습니다…";
        try
        {
            var runDir = await Task.Run(() => ComparisonRunner.RunAndRecord(manifest, baseDir, outRoot));
            if (gen != _generation) { _result.Text = ""; return; }   // a different manifest was opened meanwhile: this result describes the old one and is not shown as current
            var res = JsonNode.Parse(File.ReadAllText(Path.Combine(runDir, "result.json")))!;
            var status = res["status"]!.GetValue<string>();
            var reasons = res["reasons"] is JsonArray ra ? ra.Select(x => x!.GetValue<string>()).ToList() : [];
            var text = new System.Text.StringBuilder();
            text.AppendLine($"결과: {status}" + (reasons.Count > 0 ? "  사유: " + string.Join(", ", reasons) : ""));
            if (status == "computed")
            {
                text.AppendLine($"ROI 유효 화소 {res["valid_pixels"]}  ·  단위 {res["units"]}  ·  차이 = A − B");
                text.AppendLine($"평균 {F(res["mean_a_minus_b"])}  표준편차(모집단) {F(res["std_a_minus_b"])}  RMSE {F(res["rmse"])}  평균 절댓값 {F(res["mean_abs"])}  최솟값 {F(res["min_a_minus_b"])}  최댓값 {F(res["max_a_minus_b"])}");
                _state.Text = "상태: 가져온 파일의 수치 차이를 계산했습니다. 이것은 2G 실행도 화질 판정도 아닙니다.";
            }
            else
            {
                text.AppendLine("숫자를 만들지 않았습니다. " + string.Join(" ", reasons.Select(r => ComparisonContract.ReasonText(r))));
                _state.Text = $"상태: 비교가 {status} 로 끝났습니다. 통과가 아닙니다. 2G 실행이 아닙니다.";
            }
            text.AppendLine($"원본 파일 변경 없음: {(res["sources_unchanged"]!.GetValue<bool>() ? "예" : "아니오")}  ·  2G 실행됨: {(res["actual_2g_executed"]!.GetValue<bool>() ? "예" : "아니오")}  ·  화질 판정: 하지 않음");
            text.AppendLine("저장 위치: " + runDir);
            _result.Text = text.ToString();
        }
        catch (Exception e) { if (gen == _generation) { _result.Text = "실패: " + e.Message; _state.Text = "상태: 저장·계산 실패. 통과가 아닙니다."; } }
        finally { SetBusy(false); }
    }

    private static string F(JsonNode? n) => n is null ? "—" : n.GetValue<double>().ToString("G10", CultureInfo.InvariantCulture);
}
