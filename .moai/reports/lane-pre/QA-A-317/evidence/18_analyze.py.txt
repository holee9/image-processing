"""QA-A-317: A (2G output) / B (PRE + 2G maps) / C (PRE + PRE gain map) on the 180 frames. Read only on existing data.

Value domain (QA-A-294 / QA-A-296 cmp2g.py transform, extended to C):
  v_X = where(box, min(rint(p_X * C / M_X), 12800), 0), box = rows 2..2467, cols 15..3054, C = CMEAS[set] (cmp2g.py)
  M_B = mean(BrightAvg read as uint16)  (= cmp2g.py Cp, the mean the B gain map was normalised by)
  M_C = mG from gen_c.json (the mean the PRE generator normalised the C gain map by, xpe_calib_generate_gain.cpp:296-322)
  A = out/<set>/NNN.raw (uint16) as stored.
Pixel classes (cmp2g.py): normal = box & ~BPMap & raw < 12800, saturated = box & ~BPMap & raw >= 12800, defect = box & BPMap.
Pairs A-B, A-C, B-C: exact integer |d| histograms per class -> match %, bins 0/1/2-5/6-20/>20, p50/p99/max, mean |d|, mean signed d.
Metrics (qa317_metrics.exe = existing xpe_pre_e2e header code), on A, v_B, v_C:
  common ROI = box & B gain finite > 0 & C gain finite > 0 (no defect exclusion, as xpe_pre_e2e.cpp:432 "no spatial exclusion")
  flat residual % and stripe ratio: only on "Empty" (open-field) frames with 0 raw >= 12800 pixels in the ROI
  defect residual: every frame, truth = BPMap inside the box interior (rows 3..2466, cols 16..3053)
B regeneration check: sha256 of every new B output vs the same file of pre-w3072-20261023-1748.
Usage: python -I -X utf8 analyze.py OUT_B OUT_C   -> analyze_frames.jsonl, analyze_summary.json in T
"""
import hashlib, json, os, struct, subprocess, sys
import numpy as np

T = "D:/workspace-github/xpe-data/qa-317-tmp"
B = "D:/workspace-github/xpe-2g-offline/batch-20261022-1230/"
OLD = "D:/workspace-github/xpe-2g-offline/pre-w3072-20261023-1748"
W, H = 3072, 2560
R0, R1, C0, C1 = 2, 2468, 15, 3055
CMEAS = {"s0_gf": 6511.0, "s1_um300": 479.0, "s4_csi_pre": 5406.0, "s5_csi_post": 5647.0, "s6_gad_pre": 2486.0, "s7_gad_post": 2456.0}
CLASSES = ("normal", "saturated", "defect", "all")
PAIRS = ("AB", "AC", "BC")
NB = 65536


def sha(p):
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for c in iter(lambda: f.read(1 << 22), b""):
            h.update(c)
    return h.hexdigest()


def u16(p):
    return np.fromfile(p, "<u2").reshape(H, W)


def gain_payload(p):
    b = open(p, "rb").read()
    cfg = struct.unpack_from("<Q", b, 0x68)[0]
    return np.frombuffer(b, "<f4", offset=152 + cfg, count=W * H).reshape(H, W)


def metrics(img):
    img.astype(np.float32).tofile(f"{T}/m_img.f32")
    r = subprocess.run([f"{T}/qa317_metrics.exe", f"{T}/m_img.f32", f"{T}/m_roi.u8", f"{T}/m_truth.u8", str(W), str(H)],
                       capture_output=True, text=True)
    if r.returncode:
        raise RuntimeError(f"metrics rc={r.returncode}")
    f = r.stdout.split()
    return {"stripe_ratio": float(f[0]), "row_ratio": float(f[1]), "col_ratio": float(f[2]), "roi_pixels": int(f[3]),
            "flat_residual_pct": float(f[4]), "flat_mean": float(f[5]), "defect_residual": float(f[6]), "defect_pixels": int(f[7])}


def main(out_b, out_c):
    man = json.load(open(B + "manifest.json", encoding="utf-8"))
    mg = json.load(open(f"{T}/gen_c.json", encoding="utf-8"))
    hist = {(s, p, k): np.zeros(NB, np.int64) for s in [x["id"] for x in man["sets"]] + ["total"] for p in PAIRS for k in CLASSES}
    signed = {(s, p): 0.0 for s in [x["id"] for x in man["sets"]] + ["total"] for p in PAIRS}
    fj = open(f"{T}/analyze_frames.jsonl", "w", encoding="utf-8")
    shacmp = {"equal": 0, "differ": 0, "missing": 0}
    for s in man["sets"]:
        sid = s["id"]
        bright = u16(B + s["files"]["bright"]["path"]).astype(np.float64)
        D = np.fromfile(B + s["files"]["bpm"]["path"], np.uint8).reshape(H, W) != 0
        C, MB, MC = CMEAS[sid], float(bright.mean()), mg[sid]["mG"]
        box = np.zeros((H, W), bool); box[R0:R1, C0:C1] = True
        inner = np.zeros((H, W), bool); inner[R0 + 1:R1 - 1, C0 + 1:C1 - 1] = True
        gB = gain_payload(f"{T}/xcal_b/{sid}/gain.xcal"); gC = gain_payload(f"{T}/xcal_c/{sid}/gain.xcal")
        roi = box & np.isfinite(gB) & (gB > 0) & np.isfinite(gC) & (gC > 0)
        roi.astype(np.uint8).tofile(f"{T}/m_roi.u8"); (D & inner).astype(np.uint8).tofile(f"{T}/m_truth.u8")
        for img in s["images"]:
            stem = f"{img['idx']:03d}"
            raw = u16(f"{B}in/{sid}/img/{stem}.raw")
            y2 = u16(f"{B}out/{sid}/{stem}.raw").astype(np.float64)
            fb, fc = f"{out_b}/{sid}/out/{stem}.f32", f"{out_c}/{sid}/out/{stem}.f32"
            hb, hc = sha(fb), sha(fc)
            fo = f"{OLD}/{sid}/out/{stem}.f32"
            ho = sha(fo) if os.path.isfile(fo) else None
            shacmp["missing" if ho is None else ("equal" if ho == hb else "differ")] += 1
            pB = np.fromfile(fb, np.float32).reshape(H, W).astype(np.float64)
            pC = np.fromfile(fc, np.float32).reshape(H, W).astype(np.float64)
            vB = np.where(box, np.minimum(np.rint(pB * C / MB), 12800), 0.0)
            vC = np.where(box, np.minimum(np.rint(pC * C / MC), 12800), 0.0)
            sat = ~D & (raw >= 12800)
            cls = {"normal": box & ~D & ~sat, "saturated": box & sat, "defect": box & D, "all": box}
            rec = {"set": sid, "idx": img["idx"], "name": img["name"], "sha_B": hb, "sha_B_1748": ho, "sha_C": hc,
                   "empty": "empty" in img["name"].lower(), "roi_saturated": int((roi & (raw >= 12800)).sum()), "match": {}}
            for p, (x, y) in {"AB": (y2, vB), "AC": (y2, vC), "BC": (vB, vC)}.items():
                d = y - x
                for sk in (sid, "total"):
                    signed[(sk, p)] += float(d[box].sum())
                ad = np.abs(d).astype(np.int64)
                for k, m in cls.items():
                    h = np.bincount(ad[m], minlength=NB)
                    hist[(sid, p, k)] += h; hist[("total", p, k)] += h
                    rec["match"][f"{p}.{k}"] = [int(h[0]), int(m.sum())]
            rec["metrics"] = {"A": metrics(y2), "B": metrics(vB), "C": metrics(vC)}
            fj.write(json.dumps(rec, ensure_ascii=False) + "\n"); fj.flush()
            print(sid, stem, "B==1748" if hb == ho else "B!=1748", flush=True)
    summ = {"sha_B_vs_1748": shacmp, "sets": {}}
    for sk in [x["id"] for x in man["sets"]] + ["total"]:
        summ["sets"][sk] = {}
        for p in PAIRS:
            for k in CLASSES:
                h = hist[(sk, p, k)]; n = int(h.sum())
                c = np.cumsum(h)
                q = lambda f: int(np.searchsorted(c, f * n, side="left")) if n else None
                vals = np.arange(NB)
                summ["sets"][sk][f"{p}.{k}"] = {
                    "n": n, "match_pct": 100.0 * h[0] / n if n else None,
                    "bins_0_1_2to5_6to20_gt20": [int(h[0]), int(h[1]), int(h[2:6].sum()), int(h[6:21].sum()), int(h[21:].sum())],
                    "p50": q(0.5), "p99": q(0.99), "max": int(vals[h > 0].max()) if n else None,
                    "mean_abs": float((h * vals).sum() / n) if n else None}
            summ["sets"][sk][f"{p}.mean_signed_all"] = signed[(sk, p)] / summ["sets"][sk][f"{p}.all"]["n"]
    json.dump(summ, open(f"{T}/analyze_summary.json", "w", encoding="utf-8"), indent=1)
    for f in ("m_img.f32", "m_roi.u8", "m_truth.u8"):
        os.remove(f"{T}/{f}")


if __name__ == "__main__":
    main(sys.argv[1], sys.argv[2])
