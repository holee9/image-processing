141_report_clean.json / 142_report_planted.json (clean and planted run of measurement B)."""

def upper_median(a, axis):
    n = a.shape[axis]
    return np.partition(a, n // 2, axis=axis).take(n // 2, axis=axis)


def sliding_median(a, axis, win=WIN):
    r = win // 2
    pad = [(0, 0), (0, 0)]
    pad[axis] = (r, r)
    p = np.pad(a, pad, mode='reflect')
    out = np.empty_like(a)
    chunk = 128
    if axis == 1:
        for y0 in range(0, a.shape[0], chunk):
            v = np.lib.stride_tricks.sliding_window_view(p[y0:y0 + chunk], win, axis=1)
            out[y0:y0 + chunk] = np.partition(v, r, axis=2)[:, :, r]
    else:
        for x0 in range(0, a.shape[1], chunk):
            v = np.lib.stride_tricks.sliding_window_view(p[:, x0:x0 + chunk], win, axis=0)
            out[:, x0:x0 + chunk] = np.partition(v, r, axis=2)[:, :, r]
    return out


def components(f, multiplicative):
    """Row then column component; the multiplicative form guards a zero reference or factor like the detector (ratio 1 / factor not applied)."""
    f = f.astype(np.float32)

    def line_component(g, ref, axis):
        if not multiplicative:
            return g - ref, upper_median(g - ref, axis)
        ok = np.abs(ref) > 1e-30
        ratio = np.where(ok, g / np.where(ok, ref, 1), 1).astype(np.float32)
        return ratio, upper_median(ratio, axis)

    def apply(g, c, shape):
        if not multiplicative:
            return g - c.reshape(shape)
        okc = np.abs(c) > 1e-30
        return np.where(okc.reshape(shape), g / np.where(okc, c, 1).reshape(shape), g).astype(np.float32)

    _, rho = line_component(f, sliding_median(f, 0), 1)
    f = apply(f, rho, (-1, 1))
    _, kap = line_component(f, sliding_median(f, 1), 0)
    return apply(f, kap, (1, -1))


def block_scale(r):
    nbx = (W + BLOCK - 1) // BLOCK
    nby = (H + BLOCK - 1) // BLOCK
    sig = np.zeros((nby, nbx), np.float32)
    for by in range(nby):
        for bx in range(nbx):
            v = np.abs(r[by * BLOCK:(by + 1) * BLOCK, bx * BLOCK:(bx + 1) * BLOCK]).ravel()
            sig[by, bx] = np.float32(1.4826) * np.partition(v, v.size // 2)[v.size // 2]
    return sig


def scale_image(sig):
    nby, nbx = sig.shape
    fy = np.clip((np.arange(H, dtype=np.float32) + 0.5) / BLOCK - 0.5, 0, nby - 1)
    fx = np.clip((np.arange(W, dtype=np.float32) + 0.5) / BLOCK - 0.5, 0, nbx - 1)
    y0 = np.floor(fy).astype(int); y1 = np.minimum(y0 + 1, nby - 1); ty = (fy - y0).astype(np.float32)
    x0 = np.floor(fx).astype(int); x1 = np.minimum(x0 + 1, nbx - 1); tx = (fx - x0).astype(np.float32)
    ry = (1 - ty)[:, None] * sig[y0] + ty[:, None] * sig[y1]
    return (1 - tx)[None, :] * ry[:, x0] + tx[None, :] * ry[:, x1]


def zimage(feat, multiplicative):
    f = components(feat, multiplicative)
    bg = sliding_median(sliding_median(f, 1), 0)
    r = f - bg
    s = np.maximum(scale_image(block_scale(r)), 1e-6)
    return r / s


def zthr(p):
    lo, hi = 0.0, 40.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        if math.erfc(mid / math.sqrt(2.0)) > p:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)


def features(folder_dark, folder_bright):
    def raw(folder, name):
        return np.fromfile(folder + name, dtype=np.uint16).reshape(H0, W0)[Y0:Y1, X0:X1]
    darks = [raw(folder_dark, 'Dark_%s.raw' % k).astype(np.float64) for k in DK]
    D = (sum(darks) / len(darks)).astype(np.float32)
    bright = {k: raw(folder_bright, 'Bright_%s.raw' % k) for k in BR}
    dmean = float(D.astype(np.float64).mean())
    sig = {k: float(bright[k].astype(np.float64).mean()) - dmean for k in BR}
    order = sorted(BR, key=lambda k: sig[k])
    groups = [[order[0]]]
    for a, b in zip(order[:-1], order[1:]):
        if sig[b] / sig[a] > 1.25:
            groups.append([])
        groups[-1].append(b)
    Sn = np.zeros((H, W)); Sx = 0.0
    for g in groups:
        sg = sum(sig[k] for k in g) / len(g)
        ng = sum(bright[k].astype(np.float64) for k in g) / len(g) - D
        Sn += ng; Sx += sg
    return D, (Sn / Sx).astype(np.float32)

    out[name] = (zd, zf)
