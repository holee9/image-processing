"""QA-A-252h step 1: the global dose of a bright frame, used for the exposure scale of the noise-cost statistic, is the MEDIAN of its net signal (frame minus the dark level of the pixel) over every fifth
pixel, not the mean over all pixels. A mean is pulled by defective pixels whose value does not follow the dose (stuck high or low, hot); the median is not. The grouping of the frames by dose and every other
use of the mean signal (signal[k]) stay as they were: only the scale changes. A frame whose median net signal is not positive is refused with the same error as a frame whose mean is not positive."""
base = 'D:/workspace-github/xpe-pre/modules/preprocess/'


def edit(rel, pairs):
    s = open(base + rel, encoding='utf-8', newline='').read()
    crlf = '\r\n' in s
    s = s.replace('\r\n', '\n')
    for old, new in pairs:
        assert s.count(old) == 1, (rel, s.count(old), old[:100])
        s = s.replace(old, new)
    open(base + rel, 'w', encoding='utf-8', newline='').write(s.replace('\n', '\r\n') if crlf else s)
