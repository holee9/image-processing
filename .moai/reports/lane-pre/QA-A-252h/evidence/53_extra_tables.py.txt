def cmp(X, Y, nx, ny):
    chg = collections.Counter()
    costs = []
    for k in X:
        if k not in Y:
            continue
        if X[k][9] != Y[k][9]:
            chg[(Y[k][9], X[k][9])] += 1
        cx, cy = float(X[k][6]), float(Y[k][6])
        if cy > 0:
            costs.append(abs(cx / cy - 1))
    costs.sort()
    print('%s vs %s: label changes %d %s; |cost ratio - 1|: median %.4f, 95th %.4f, max %.4f (items %d)' % (nx, ny, sum(chg.values()), dict(chg), costs[len(costs) // 2], costs[int(len(costs) * 0.95)], costs[-1], len(costs)))

def bits(p):
    L = [l.rstrip('\n').split('\t') for l in open(p, encoding='utf-8')][1:]
    c = collections.Counter()
    for x in L:
        v = int(x[5])
        for b, n in ((1, 'R'), (2, 'L'), (4, 'N'), (8, 'S')):
            if v & b:
                c[n] += 1
    return len(L), dict(c)
