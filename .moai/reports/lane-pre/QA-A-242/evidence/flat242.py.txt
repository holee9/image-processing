def pct(y, sel=None):
    v = y[ok] if sel is None else y[sel]
    return 100.0 * float(v.std()) / float(v.mean())

def blockmeans(y, b):
    h = (N // b) * b
    v = np.where(ok, y, np.nan)[:h, :h]
    s = np.nansum(v.reshape(h // b, b, h // b, b), axis=(1, 3))
    c = np.sum(ok[:h, :h].reshape(h // b, b, h // b, b), axis=(1, 3))
    return s / np.maximum(c, 1), c

    y = Y[j]
    mu = float(y[ok].mean())
    tot = float(y[ok].var())
    row = []
    prev = tot
    for b in (4, 16, 64, 256):
        bm, c = blockmeans(y, b)
        w = c > (0.5 * b * b)
        v = float(bm[w].var())   # variance of block means = variance at scales >= b
        row.append((b, v))
    shares = [(tot - row[0][1]) / tot] + [(row[k][1] - row[k + 1][1]) / tot for k in range(3)] + [row[3][1] / tot]
    P('%s total %.3f %% | share of variance: <4 px %.0f %%, 4-16 %.0f %%, 16-64 %.0f %%, 64-256 %.0f %%, >=256 px %.0f %%   (sqrt(total var) of scales >= 16 px: %.3f %%)' % (
        names[j], 100 * np.sqrt(tot) / mu, 100 * shares[0], 100 * shares[1], 100 * shares[2], 100 * shares[3], 100 * shares[4], 100 * np.sqrt(row[1][1]) / mu))

P('== pairs of flats with the SAME large-scale shape (4, 5, 6): where does the residual of flat j under the gain of flat k come from, by scale?')
def band_shares(y):
    mu = float(y[ok].mean()); tot = float(y[ok].var())
    vs = []
    for b in (4, 16, 64, 256):
        bm, c = blockmeans(y, b); w = c > (0.5 * b * b); vs.append(float(bm[w].var()))
    sh = [(tot - vs[0]) / tot] + [(vs[i] - vs[i + 1]) / tot for i in range(3)] + [vs[3] / tot]
    return 100 * np.sqrt(tot) / mu, sh
