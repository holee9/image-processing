# QA-A-343: the "correction error median z" row (v2 worse than v1 in 117 of 178 group x frame). Read-only on committed json.
# z = |planted corrected - clean corrected| / sigma of the clean output in an 11x11 window (modules/preprocess/tools/xpe_bpm_eval.cpp:2438-2465).
# For each worse cell: v1 z, v2 z, and the same pixels left uncorrected (no defect correction; for a response-loss line after gain correction
# this is the pixel's own value, i.e. the noise floor of the metric).
import json,collections,sys
v1=json.load(open(sys.argv[1],encoding='utf-8')); v2=json.load(open(sys.argv[2],encoding='utf-8'))
rows=collections.defaultdict(list); n=0; worse=0
for f1,f2 in zip(v1['frames'],v2['frames']):
    a={g['group']:g for g in f1['correction_error']}
    for g in f2['correction_error']:
        if g['group'] not in a: continue
        n+=1; z1=a[g['group']]['corrected']['median_z']; z2=g['corrected']['median_z']; u=g['uncorrected']['median_z']
        k='/'.join(g['group'].split('/')[:2]); rows[k].append((z1,z2,u,z2>z1)); worse+=z2>z1
print('cells',n,'v2 worse',worse)
print('class/kind\tcells\tv2_worse\tv1_z median\tv2_z median\tuncorrected_z median')
med=lambda v: sorted(v)[len(v)//2]
for k,v in sorted(rows.items(), key=lambda kv:-sum(r[3] for r in kv[1])):
    print('%s\t%d\t%d\t%.3f\t%.3f\t%.3f'%(k,len(v),sum(r[3] for r in v),med([r[0] for r in v]),med([r[1] for r in v]),med([r[2] for r in v])))
