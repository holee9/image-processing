import sys, re, statistics as st, collections
# QA-A-325: paired difference A - B per (set, cfg, round) from ab.sh output (both variants measured back to back in the same round)
d = collections.defaultdict(dict)
for f in sys.argv[1:]:
    for l in open(f):
        m = re.search(r'round=(\d+) set=(\S+) cfg=(\S+) var=(\S+) .*median_excl_first=([0-9.]+)', l)
        if m: d[(m.group(2), m.group(3), m.group(1))][m.group(4)] = float(m.group(5))
agg = collections.defaultdict(list)
for (s, c, r), v in d.items():
    if 'A' in v and 'B' in v: agg[(s, c)].append(v['A'] - v['B'])
for k in sorted(agg):
    v = agg[k]
    print(f"{k[0]:12s} {k[1]:18s} n={len(v)} A-B median={st.median(v):+.3f} min={min(v):+.3f} max={max(v):+.3f}  B<A in {sum(x > 0 for x in v)}/{len(v)} rounds")
