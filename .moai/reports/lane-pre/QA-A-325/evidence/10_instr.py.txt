# QA-A-325: insert temporary section timers (every inserted line ends with // QA325TMP). Removal: delete those lines.
import sys
root = sys.argv[1]
T = ' // QA325TMP'
def ins(path, edits):
    L = open(path, encoding='utf-8', newline='').read().split('\n')
    for anchor, where, text in edits:
        hits = [i for i, l in enumerate(L) if anchor in l]
        assert len(hits) == 1, (path, anchor, hits)
        i = hits[0]
        new = [t + T for t in text]
        L[i + 1 if where == 'after' else i:i + 1 if where == 'after' else i] = new
    open(path, 'w', encoding='utf-8', newline='').write('\n'.join(L))
