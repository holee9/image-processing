N = 512
SIG = 30.0
yy = np.arange(N, dtype=np.float64)[:, None] * np.ones((1, N))


def smoothstep(t):
    t = np.clip(t, 0, 1)
    return t * t * (3 - 2 * t)


def truth(case, y0, width):
    c = y0 + (width - 1) / 2.0
    if case == 'flat':
        return np.full((N, N), 10000.0)
    if case == 'step':
        return 9000.0 + 2000.0 * smoothstep((yy - c + 2.0) / 4.0)
    if case == 'ramp':
        return 10000.0 + 10.0 * (yy - c)
    if case == 'stripe':
        T = np.full((N, N), 10000.0)
        T[y0 + width + 1, :] += 2000.0   # the bright line two pixels beyond the line group
        return T
    raise ValueError(case)


def run_one(case, width, orient, seed):
    rng = np.random.default_rng(seed)
    y0 = 256
    T = truth(case, y0, width)
    O = T + rng.normal(0.0, SIG, (N, N))
    mask = np.zeros((N, N), dtype=np.uint8)
    mask[y0:y0 + width, 50:450] = 1
    O[mask == 1] = 0.0
    seg = [(y0 + j, 50, 450) for j in range(width)]
    line = dict(orientation=0, index=y0, width=width, start=50, end=450, causes=1, flags=0, reason=0, correction=1, strength=1.0, persistence=-1.0, segments=seg)
    if orient == 'col':   # the same picture transposed
        T, O, mask = T.T.copy(), O.T.copy(), mask.T.copy()
        line = dict(orientation=1, index=y0, width=width, start=50, end=450, causes=1, flags=0, reason=0, correction=1, strength=1.0, persistence=-1.0, segments=[(y0 + j, 50, 450) for j in range(width)])
    res = {}
    for algo in ('A1', 'A2'):

        load_defect(p)
        out, rep = correct_ex(O.astype(np.float32), N, N)
        m = mask == 1

        changed = int(((out != O.astype(np.float32)) & ~m).sum())
