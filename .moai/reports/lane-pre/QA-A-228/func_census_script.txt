"""QA-A-228 part 3: SRS-CALIB-FUNC-* vs RTM check marks vs tests. Two axes: (name) the id / UT-IT id cited in a test; (behaviour) the exported API names the SRS row names are exercised by a test."""
import re, glob, os
R = 'D:/workspace-github/xpe-pre/'
OUT = []
def p(*a): OUT.append(' '.join(str(x) for x in a))

srs = open(R + 'docs/calibration/SRS-CALIB-001_Software_Requirements_Specification.md', encoding='utf-8').read().split('\n')
rtm = open(R + 'docs/calibration/RTM-CALIB-001_Requirements_Traceability_Matrix.md', encoding='utf-8').read().split('\n')

# SRS definitions: table rows starting with | **SRS-CALIB-FUNC-NNN**
sdef = {}
for i, l in enumerate(srs):
    m = re.match(r'^\|\s*\*\*SRS-CALIB-FUNC-(\d+)\*\*\s*\|(.*)$', l)
    if m:
        sdef[m.group(1)] = (i + 1, m.group(2))
p('SRS definitions parsed:', len(sdef), sorted(sdef)[:3], '..', sorted(sdef)[-3:], '| control FUNC-001 parsed =', '001' in sdef, '| FUNC-999 =', '999' in sdef)

# RTM definition table rows (section 2) : | **SRS-CALIB-FUNC-NNN** | summary | SWU | design | UT | IT | ST | HAZ | control
rrows = {}
for i, l in enumerate(rtm):
    m = re.match(r'^\|\s*\*\*SRS-CALIB-FUNC-(\d+)\*\*\s*\|(.*)$', l)
    if m:
        cells = [c.strip() for c in l.split('|')[1:-1]]
        rrows[m.group(1)] = (i + 1, cells)
p('RTM definition rows parsed:', len(rrows), '| control FUNC-001 =', '001' in rrows)
p('in SRS but no RTM definition row:', sorted(set(sdef) - set(rrows)))
p('in RTM definition table but not in SRS:', sorted(set(rrows) - set(sdef)))

# RTM check marks: any line with a tick mentioning FUNC ids (section 4, 5, 5.1)
ticked = set()
for i, l in enumerate(rtm):
    if re.search(r'\u2713|Updated|Added|Designed', l):
        for m in re.finditer(r'SRS-CALIB-FUNC-(\d+)(?:\.\.(\d+))?', l):
            a = int(m.group(1)); b = int(m.group(2) or a)
            for n in range(a, b + 1): ticked.add('%03d' % n)
# section 4 forward traceability claims every FUNC row traced (24 of 24)
p('FUNC ids named on a tick/Updated/Added/Designed line (sections 4-5.1):', sorted(ticked))
p('RTM s4 claim: Functional Req 24 of 24 traced; definition-table rows =', len(rrows))

# test sources
files = [f for f in glob.glob(R + 'modules/**/*', recursive=True) if f.endswith(('.cpp', '.h', '.hpp'))]
tests = [f for f in files if '/tests/' in f.replace(chr(92), '/')]
texts = {f: open(f, encoding='utf-8', errors='replace').read() for f in tests}
allsrc = {f: open(f, encoding='utf-8', errors='replace').read() for f in files}
p('test files:', len(tests))

def cites_id(fid):
    return sorted(set(os.path.basename(f) for f, t in texts.items() if re.search(r'FUNC-%s\b' % fid, t)))

ut_all = {}
for f, t in texts.items():
    for u in set(re.findall(r'\b(?:UT|IT|ST)-[A-Z0-9]+(?:-[A-Z0-9]+)*-\d+', t)):
        ut_all.setdefault(u, []).append(os.path.basename(f))
p('UT/IT/ST ids appearing in any test source:', len(ut_all))

# API names named in the SRS row text
def apis(text):
    return sorted(set(re.findall(r'\bxpe_[a-z0-9_]+', text)))

res = []
for fid in sorted(sdef):
    ln, body = sdef[fid]
    names = apis(body)
    rtm_cells = rrows.get(fid, (None, []))[1]
    uts = re.findall(r'(?:UT|IT|ST)-[A-Z0-9]+(?:-[A-Z0-9]+)*-\d+(?:~\d+)?', ' '.join(rtm_cells[4:7])) if rtm_cells else []
    c_id = cites_id(fid)
    c_ut = [u for u in uts if re.sub(r'~\d+$', '', u) in ut_all]
    # behaviour axis: test files that call the API names
    api_hits = {}
    for n in names:
        h = sorted(set(os.path.basename(f) for f, t in texts.items() if re.search(r'\b%s\b' % re.escape(n), t)))
        api_hits[n] = len(h)
    res.append((fid, ln, names, uts, c_id, c_ut, api_hits, fid in rrows, fid in ticked))

p('--- per-FUNC: id | SRS line | in RTM table | on tick line | test files citing the id | RTM UT/IT/ST ids found in any test | APIs named in SRS row -> #test files calling it')
for fid, ln, names, uts, c_id, c_ut, ah, inr, tk in res:
    p('FUNC-%s | srs:%d | rtm-row:%s | tick:%s | id-cites:%d %s | rtm-test-ids:%d/%d | apis: %s' % (
        fid, ln, 'Y' if inr else 'N', 'Y' if tk else 'N', len(c_id), ','.join(c_id)[:70], len(c_ut), len(uts),
        ' '.join('%s=%d' % (k, v) for k, v in ah.items())[:200]))

none_id = [r[0] for r in res if not r[4]]
p('SRS FUNC ids cited by NO test file (name axis):', len(none_id), none_id)
no_api_test = [r[0] for r in res if r[2] and all(v == 0 for v in r[6].values())]
p('SRS FUNC rows whose named APIs are called by NO test (behaviour axis):', no_api_test)
no_api_named = [r[0] for r in res if not r[2]]
p('SRS FUNC rows that name no xpe_ API (behaviour axis cannot be mechanised):', no_api_named)
open(R + '.moai/reports/lane-pre/QA-A-228/func_census_out.txt', 'w', encoding='utf-8', newline='\n').write('\n'.join(OUT) + '\n')
print('\n'.join(OUT[:9]).encode('ascii', 'replace').decode())
