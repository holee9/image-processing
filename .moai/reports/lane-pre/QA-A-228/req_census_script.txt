"""QA-A-228 part 2: REQ-P1A-* definitions vs. work-table rows vs. test citations. ASCII output only apart from file text."""
import re, glob, os, sys, json
R = 'D:/workspace-github/xpe-pre/'
SD = R + '.moai/specs/SPEC-XPE-P1A/'
OUT = []

def p(*a):
    OUT.append(' '.join(str(x) for x in a))

spec = open(SD + 'spec.md', encoding='utf-8').read().split('\n')
defs = []
for i, l in enumerate(spec):
    m = re.match(r'^#{2,6}\s+(REQ-P1A-\d+[a-z]?)\s*[:\-]?\s*(.*)$', l)
    if m:
        defs.append((m.group(1), m.group(2).strip(), i + 1))
ids = [d[0] for d in defs]
p('PARSER: heading definitions =', len(ids), '| control REQ-P1A-010 parsed =', 'REQ-P1A-010' in ids,
  '| control (id that does not exist) REQ-P1A-999 parsed =', 'REQ-P1A-999' in ids)

# the work table: plan.md milestone task rows (| Mn-k | ...) ; spec section 8 tables ; progress.md deliverables
plan = open(SD + 'plan.md', encoding='utf-8').read().split('\n')
rows = [(l.split('|')[1].strip(), l) for l in plan if re.match(r'^\|\s*M\d-\d', l)]
p('plan.md task rows:', len(rows), '| control: M2-1 present =', any(r[0] == 'M2-1' for r in rows))
row_req = {}
for tid, l in rows:
    for r in re.findall(r'REQ-P1A-\d+[a-z]?', l):
        row_req.setdefault(r, []).append(tid)
p('REQ ids named inside a plan.md task row:', sorted(row_req))

def ids_in(text):
    return set(re.findall(r'REQ-P1A-\d+[a-z]?', text))

# spec section 8 text
s8 = '\n'.join(spec[1111:1175])
prog = open(SD + 'progress.md', encoding='utf-8').read()
plan_all = '\n'.join(plan)
a_plan_rows = set(row_req)
b_s8 = ids_in(s8)
c_prog = ids_in(prog)
d_plan_text = ids_in(plan_all)
p('in plan.md task rows:', len(a_plan_rows & set(ids)), '| in spec s8:', len(b_s8 & set(ids)), '| in progress.md:', len(c_prog & set(ids)), '| anywhere in plan.md:', len(d_plan_text & set(ids)))
# range statements like "REQ-P1A-001 ~ 009" in s8 cover ids not literally named
rng = re.findall(r'REQ-P1A-(\d+)\s*~\s*(\d+)', s8)
covered_by_range = set()
for a, b in rng:
    for n in range(int(a), int(b) + 1):
        covered_by_range.add('REQ-P1A-%03d' % n)
p('s8 range statements:', rng)
linked = a_plan_rows | b_s8 | c_prog | covered_by_range
unlinked_strict = [i for i in ids if i not in (a_plan_rows)]
unlinked_loose = [i for i in ids if i not in linked]
p('NOT in any plan.md task row (strict):', len(unlinked_strict), 'of', len(ids))
p('NOT in plan task rows, spec s8 (named or by range), or progress.md (loose):', len(unlinked_loose), unlinked_loose)

# tests citing the REQ id (name census)
files = [f for f in glob.glob(R + 'modules/**/*', recursive=True) if f.endswith(('.cpp', '.h', '.hpp', '.cs'))]
tests = [f for f in files if '/tests/' in f.replace(chr(92), '/')]
cites = {}
for f in tests:
    t = open(f, encoding='utf-8', errors='replace').read()
    for r in set(re.findall(r'REQ-P1A-\d+[a-z]?', t)):
        cites.setdefault(r, []).append(os.path.basename(f))
# C# tests under clients?
cs = [f for f in glob.glob(R + 'clients/**/*.cs', recursive=True)]
for f in cs:
    t = open(f, encoding='utf-8', errors='replace').read()
    for r in set(re.findall(r'REQ-P1A-\d+[a-z]?', t)):
        cites.setdefault(r, []).append('CS:' + os.path.basename(f))
p('test sources scanned (modules tests):', len(tests), '| C# files scanned:', len(cs))
p('control: REQ-P1A-010 cited by tests =', len(cites.get('REQ-P1A-010', [])), '; nonexistent REQ-P1A-999 =', len(cites.get('REQ-P1A-999', [])))
no_cite = [i for i in ids if i not in cites]
p('REQ with ZERO test-file citations (name census):', len(no_cite), no_cite)
p('--- per-REQ table: id | title | in-plan-task-row | s8/progress/range | #test files citing | files')
for i, t, n in defs:
    where = []
    if i in a_plan_rows: where.append('plan-row ' + ','.join(row_req[i]))
    if i in b_s8: where.append('s8')
    if i in c_prog: where.append('progress')
    if i in covered_by_range and i not in b_s8: where.append('s8-range')
    c = cites.get(i, [])
    p('%s | %s | %s | %d | %s' % (i, t[:44], ';'.join(where) or '-', len(c), ','.join(sorted(set(c)))[:150]))
open(R + '.moai/reports/lane-pre/QA-A-228/req_census_out.txt', 'w', encoding='utf-8', newline='\n').write('\n'.join(OUT) + '\n')
print('\n'.join(OUT[:14]).encode('ascii', 'replace').decode())
