def build():
    r = subprocess.run(['cmd.exe', '/c', S + 'buildall.bat'], capture_output=True, text=True)
    return 'BUILD_EXIT=0' in r.stdout, r.stdout


def run_tests():
    r = subprocess.run([R + 'build/ci-preprocess/bin/xpe_preprocess_oom_tests.exe', '--gtest_filter=GhostMixCompare.*'], cwd=R + 'modules/preprocess', capture_output=True, text=True)
    failed = sorted(set(re.findall(r'^\[  FAILED  \] ([A-Za-z]\S+)', r.stdout, re.M)))
    digest = re.findall(r'tier3-digest (0x[0-9a-f]+)', r.stdout)
    return failed, r.returncode, digest, r.stdout


for name in sys.argv[1:]:
    f, old, new = ARMS[name]
    keep = S + 'keep225m4_' + os.path.basename(f)
    shutil.copy(keep, f)
    src = open(f, 'rb').read().decode('utf-8')
    crlf = '\r\n' in src
    s = src.replace('\r\n', '\n')
    if s.count(old) < 1:
        print(name, 'NOT FOUND'); continue
    idx = s.index(old)
    s = s[:idx] + new + s[idx + len(old):]
    open(f, 'wb').write((s.replace('\n', '\r\n') if crlf else s).encode('utf-8'))
    ok, out = build()
    if not ok:
        print(name, 'BUILD FAILED'); print(out[-900:])
    else:
        failed, rc, digest, so = run_tests()
        crashed = rc not in (0, 1) or (rc != 0 and not failed)
        tag = (' [PROCESS DIED, exit %d]' % rc) if crashed else ''
        lab = re.findall(r'LABEL (\w)', so)
        print('%-36s -> %d red%s: %s | digest %s | label %s' % (name, len(failed), tag, ', '.join(failed) if failed else ('NONE (arm did not fire)' if not crashed else 'no FAILED line'), digest, lab))
    shutil.copy(keep, f)
