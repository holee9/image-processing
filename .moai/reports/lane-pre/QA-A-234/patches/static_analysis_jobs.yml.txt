  clang-tidy:
    # QA-A-234 (#253): clang-tidy over modules/common and modules/preprocess with each module's .clang-tidy.
    # A baseline gate: a (check, file) pair with more findings than tools/ci/clang_tidy_baseline_<module>.txt
    # fails the job. If the runner image has no clang-tidy the script says so and fails -- that first run is how
    # we learn whether windows-2025 ships it.
    needs: detect-lane
    if: needs.detect-lane.outputs.is_main == 'true' || needs.detect-lane.outputs.lane != 'unknown'
    runs-on: windows-2025
    timeout-minutes: 30

    steps:
      - name: Checkout
        uses: actions/checkout@v6

      - name: Ensure Ninja is available
        shell: pwsh
        run: |
          if (-not (Get-Command ninja -ErrorAction SilentlyContinue)) {
            choco install ninja -y --no-progress
          }
          ninja --version

      - name: Configure MSVC developer environment
        shell: pwsh
        run: pwsh -NoProfile -ExecutionPolicy Bypass -File .\tools\ci\Use-MsvcDevShell.ps1 -PersistForGitHubActions

      - name: Configure CMake (preprocess builds common too)
        shell: pwsh
        run: cmake --preset ci-preprocess

      - name: clang-tidy modules/common against its baseline
        shell: pwsh
        run: python tools/ci/check_clang_tidy.py common build/ci-preprocess tools/ci/clang_tidy_baseline_common.txt

      - name: clang-tidy modules/preprocess against its baseline
        shell: pwsh
        run: python tools/ci/check_clang_tidy.py preprocess build/ci-preprocess tools/ci/clang_tidy_baseline_preprocess.txt

  cppcheck:
    # QA-A-234 (#253): MEASUREMENT ONLY on its first runs. It never fails the pipeline (--error-exitcode=0 and
    # no gate step); it prints the finding count per module and uploads the XML, so the baseline can be taken from
    # a real run -- cppcheck is not installed on the lane machines, so no count existed when this job was written.
    # Turn it into a gate by adding tools/ci/check_cppcheck_baseline (same shape as the clang-tidy one).
    needs: detect-lane
    if: needs.detect-lane.outputs.is_main == 'true' || needs.detect-lane.outputs.lane != 'unknown'
    runs-on: windows-2025
    timeout-minutes: 20

    steps:
      - name: Checkout
        uses: actions/checkout@v6

      - name: Install cppcheck
        shell: pwsh
        run: |
          if (-not (Get-Command cppcheck -ErrorAction SilentlyContinue)) {
            choco install cppcheck -y --no-progress
            $env:Path += ';C:\Program Files\Cppcheck'
            Add-Content -Path $env:GITHUB_PATH -Value 'C:\Program Files\Cppcheck'
          }
          & 'C:\Program Files\Cppcheck\cppcheck.exe' --version

      - name: Run cppcheck (measurement only)
        shell: pwsh
        run: |
          New-Item -ItemType Directory -Force build/cppcheck | Out-Null
          foreach ($m in 'common', 'preprocess') {
            & 'C:\Program Files\Cppcheck\cppcheck.exe' --enable=warning,performance,portability --std=c++17 `
              --inline-suppr --suppress=missingIncludeSystem --error-exitcode=0 --xml `
              -I modules/common/include -I modules/preprocess/include -I modules/preprocess/src `
              "modules/$m/src" 2> "build/cppcheck/$m.xml"
            $n = (Select-String -Path "build/cppcheck/$m.xml" -Pattern '<error ' -AllMatches).Matches.Count
            Write-Host ("cppcheck modules/{0}: {1} finding(s)" -f $m, $n)
          }

      - name: Upload cppcheck results
        if: always()
        uses: actions/upload-artifact@v6
        with:
          name: cppcheck-results
          retention-days: 14
          path: build/cppcheck/
