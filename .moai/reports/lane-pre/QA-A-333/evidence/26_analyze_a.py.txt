# QA-A-333 A: per set and frame, rows and columns separately: sample sd of ratio, fraction |ratio|>3, vs uncorrelated-noise expectation (sd 1, 0.27 %).
# Lines used: computable, not skipped, and inside the margins 6,2,1,1 (rows 2..3070, cols 6..3070) -- the edge lines are listed separately.
# Autocorrelation of the line-mean differences dn at lags 1..3 (same lines).
import sys, glob, numpy as np, os
T = sys.argv[1]
for setdir in sorted(glob.glob(T + "/*/")):
    n = os.path.basename(os.path.normpath(setdir))
    for fn in sorted(glob.glob(setdir + "lines*_f*.tsv")):
        d = np.genfromtxt(fn, dtype=None, encoding=None, delimiter="\t", names=True)
        for o, lo, hi in (("row", 2, 3070), ("col", 6, 3070)):
            s = d[(d["orient"] == o) & (d["skipped"] == 0) & (d["computable"] == 1)]
            inside = s[(s["index"] >= lo) & (s["index"] <= hi)]
            r = inside["ratio"]; dn = inside["dn"] - inside["dn"].mean()
            ac = [float(np.sum(dn[:-k] * dn[k:]) / np.sum(dn * dn)) for k in (1, 2, 3)]
            print("%-18s %s %-3s lines %4d | sd %.3f | mean %.3f | |r|>3 %.2f %% (%d) | max|r| %.1f | acf(dn) lag1..3 %.3f %.3f %.3f | median sigmaPred %.4f" %
                  (n, os.path.basename(fn)[:-4], o, len(r), r.std(ddof=1), r.mean(), 100 * np.mean(np.abs(r) > 3), int(np.sum(np.abs(r) > 3)), np.abs(r).max(), *ac, np.median(inside["sigmaPred"])))
print("expected for uncorrelated noise: sd 1.000, |r|>3 0.27 %")
