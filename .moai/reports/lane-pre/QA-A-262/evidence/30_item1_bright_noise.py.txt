def box_any(mask, r):
    """True where any pixel within Chebyshev distance r is set."""
    m = mask.astype(np.int32)
    c = np.pad(m, ((r + 1, r), (r + 1, r))).cumsum(axis=0).cumsum(axis=1)
    k = 2 * r + 1
    s = c[k:, k:] - c[:-k, k:] - c[k:, :-k] + c[:-k, :-k]
    return s[:h, :w] > 0


def box_count(mask, r):
    m = mask.astype(np.int32)
    c = np.pad(m, ((r + 1, r), (r + 1, r))).cumsum(axis=0).cumsum(axis=1)
    k = 2 * r + 1
    return (c[k:, k:] - c[:-k, k:] - c[k:, :-k] + c[:-k, :-k])[:h, :w]


def end_zone(k):
    """Item-k pixels within 8 pixels of an end of any rectangle of the item, along its length."""
    z = np.zeros((h, w), dtype=bool)
    for (x, y, rw, rh) in items[k]['rects']:
        x0, y0 = x - ML, y - MT
        if rw > rh:
            z[max(y0, 0):y0 + rh, max(x0, 0):x0 + 8] = True
            z[max(y0, 0):y0 + rh, max(x0 + rw - 8, 0):x0 + rw] = True
        else:
            z[max(y0, 0):y0 + 8, max(x0, 0):x0 + rw] = True
            z[max(y0 + rh - 8, 0):y0 + rh, max(x0, 0):x0 + rw] = True
    return z & (ids_c == k)


def hyper_sf(N, K, n, k):
    """P(X >= k) for X ~ hypergeometric(N population, K successes, n draws)."""
    def lc(a, b):
        return math.lgamma(a + 1) - math.lgamma(b + 1) - math.lgamma(a - b + 1)
    tot = 0.0
    for j in range(k, min(K, n) + 1):
        tot += math.exp(lc(K, j) + lc(N - K, n - j) - lc(N, n))
    return tot

def mc_rate(rho):
    tsum = np.zeros(NMC)
    hits = np.zeros(NMC, dtype=np.int32)
    for i, nu in enumerate(NU):
        x = rho[i] * rng.chisquare(nu, size=NMC) / nu
        tsum += nu * x
        hits += (x >= 2.0)
    return float(np.mean((tsum / 13.0 > THR) & (2 * hits >= 5)))
