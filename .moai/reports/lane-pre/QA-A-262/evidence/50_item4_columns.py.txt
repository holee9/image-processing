def line_index_rows(img):
    """D_n, sigmaPred and ratio for every row y = 1 .. S-2 (rect {0, y, S, 1}): the front neighbour is row y - 1."""
    a = img.astype(np.float64)
    P = a.mean(axis=1)
    dn = P[1:-1] - 0.5 * (P[:-2] + P[2:])
    res = a[:, 1:-1] - 0.5 * (a[:, :-2] + a[:, 2:])
    sp = res.std(axis=1, ddof=1) / np.sqrt(1.5)
    sig = sp[:-2] * np.sqrt(1.0 / S + 0.5 / S)
    return dn, sig, dn / sig            # index i -> row i + 1


def line_index_cols(img):
    a = img.astype(np.float64)
    P = a.mean(axis=0)
    dn = P[1:-1] - 0.5 * (P[:-2] + P[2:])
    res = a[1:-1, :] - 0.5 * (a[:-2, :] + a[2:, :])
    sp = res.std(axis=0, ddof=1) / np.sqrt(1.5)
    sig = sp[:-2] * np.sqrt(1.0 / S + 0.5 / S)
    return dn, sig, dn / sig            # index i -> column i + 1


def part_index(img, rect):
    """lineIndex of a truth rectangle (row-type or column-type), general case."""
    x, y, w, h = rect
    a = img.astype(np.float64)
    row = w > h
    k, ln = (h, w) if row else (w, h)
    first = y if row else x
    b, af = first - 1, first + k
    if b < 0 or af >= (S if True else 0) or ln < 8:
        return None
    if row:
        item = a[first:first + k, x:x + w].mean()
        pb, pa = a[b, x:x + w].mean(), a[af, x:x + w].mean()
        front = a[b, x:x + w]
    else:
        item = a[y:y + h, first:first + k].mean()
        pb, pa = a[y:y + h, b].mean(), a[y:y + h, af].mean()
        front = a[y:y + h, b]
    dn = item - 0.5 * (pb + pa)
    res = front[1:-1] - 0.5 * (front[:-2] + front[2:])
    sig = res.std(ddof=1) / np.sqrt(1.5) * np.sqrt(1.0 / (k * ln) + 0.5 / ln)
    return dn, sig, dn / sig
