"""QA-A-262 item 3: the correction-error retreat 118 / 178 (second edition larger than first edition, per group x frame) split into reference, map and correction-path effects.
Arms (median z of |planted - clean| / local sigma per group, corrected side; z comes from the product evaluation tool `correct`):
  A = 44_correct_v1 (first-edition sets), B = 45_correct_v2 (second-edition sets), B' = 62_correct_Bprime (second-edition planted set, first-edition clean set), C = 63_correct_C (second-edition map, first-edition path, first-edition clean set).
Pairs: B vs A (the 260 figure, must give 118 of 178), B vs B' (reference), B' vs C (correction path), C vs A (map). "x larger" = the first-named arm has the larger median z for the same group and frame.
Groups with an uncomputable median are counted apart. By class (line / pixel / block) and the distribution of the differences. Evaluation range: all truth pixels of non-control items without sharing (the tool's range).
Usage: PYTHONIOENCODING=utf-8 python 64_item3_compare.py.txt > 65_item3_result.txt"""
import json
import numpy as np

EV8 = 'D:/workspace-github/xpe-pre/.moai/reports/lane-pre/QA-A-260/evidence/'
EV2 = 'D:/workspace-github/xpe-pre/.moai/reports/lane-pre/QA-A-262/evidence/'


def say(*a):
    print(*a, flush=True)


def load(path):
    j = json.load(open(path, encoding='utf-8'))
    out = {}
    for f, fr in enumerate(j['frames']):
        for g in fr['correction_error']:
            m = g['corrected']['median_z']
            out[(g['group'], f + 1)] = (float(m) if isinstance(m, (int, float)) else None, g['pixels'], g['uncorrected']['median_z'])
    return out


ARM = {'A': load(EV8 + '44_correct_v1.json.txt'), 'B': load(EV8 + '45_correct_v2.json.txt'), "B'": load(EV2 + '62_correct_Bprime.json.txt'), 'C': load(EV2 + '63_correct_C.json.txt')}
keys = sorted(set(ARM['A']) & set(ARM['B']) & set(ARM["B'"]) & set(ARM['C']))
say('group x frame entries in all four arms: %d (A %d, B %d, B\' %d, C %d)' % (len(keys), len(ARM['A']), len(ARM['B']), len(ARM["B'"]), len(ARM['C'])))
assert len(keys) == 178, len(keys)


def cls(k):
    return k[0].split('/')[0]


def compare(x, y, label):
    big, small, tie, unc = {}, 0, 0, 0
    diffs = []
    bykind = {}
    for k in keys:
        a, b = ARM[x][k][0], ARM[y][k][0]
        c = cls(k)
        bykind.setdefault(c, [0, 0])
        bykind[c][1] += 1
        if a is None or b is None:
            unc += 1
            continue
        diffs.append(a - b)
        if a > b:
            bykind[c][0] += 1
        elif a < b:
            small += 1
        else:
            tie += 1
    d = np.array(diffs)
    n_big = sum(v[0] for v in bykind.values())
    say('  %-26s %s larger in %3d of %d (line %d/%d, pixel %d/%d, block %d/%d); smaller in %d, equal %d, uncomputable %d; difference of the medians: median %+.4f, quartiles %+.4f .. %+.4f, 5 %% %+.3f, 95 %% %+.3f' %
        (label, x, n_big, len(keys), bykind.get('line', [0, 0])[0], bykind.get('line', [0, 0])[1], bykind.get('pixel', [0, 0])[0], bykind.get('pixel', [0, 0])[1], bykind.get('block', [0, 0])[0], bykind.get('block', [0, 0])[1], small, tie, unc,
         np.median(d), np.percentile(d, 25), np.percentile(d, 75), np.percentile(d, 5), np.percentile(d, 95)))
    return n_big


say('=' * 120)
say('== retreat counts (number of group x frame entries where the first arm has the larger median z)')
r_total = compare('B', 'A', 'second vs first edition (260)')
r_ref = compare('B', "B'", 'reference effect')
r_path = compare("B'", 'C', 'correction-path effect')
r_map = compare('C', 'A', 'map effect')
say('  note: B vs A = %d (QA-A-260 reported 118 of 178)' % r_total)
say('=' * 120)
say('== the line groups: the share of the total retreat that each step carries, and the sum of the step differences (B - A = (B - B\') + (B\' - C) + (C - A)) per entry')
tot_ref, tot_path, tot_map, tot_all = [], [], [], []
for k in keys:
    v = [ARM[a][k][0] for a in ('A', 'B', "B'", 'C')]
    if any(x is None for x in v):
        continue
    A, B, Bp, C = v
    tot_ref.append(B - Bp)
    tot_path.append(Bp - C)
    tot_map.append(C - A)
    tot_all.append(B - A)
tr, tp, tm, ta = map(np.array, (tot_ref, tot_path, tot_map, tot_all))
say('  entries %d: median of B - A %+.4f; of the reference step %+.4f, the correction-path step %+.4f, the map step %+.4f; mean of B - A %+.4f = reference %+.4f + path %+.4f + map %+.4f' %
    (ta.size, np.median(ta), np.median(tr), np.median(tp), np.median(tm), ta.mean(), tr.mean(), tp.mean(), tm.mean()))
big = [(k, ARM['A'][k][0], ARM['B'][k][0], ARM["B'"][k][0], ARM['C'][k][0]) for k in keys if all(ARM[a][k][0] is not None for a in ('A', 'B', "B'", 'C'))]
big.sort(key=lambda t: -(t[2] - t[1]))
say('== the ten entries with the largest B - A (group, frame: A, B, B\', C median z)')
for k, a, b, bp, c in big[:10]:
    say('  %-44s f%d: A %.3f  B %.3f  B\' %.3f  C %.3f' % (k[0], k[1], a, b, bp, c))
say('== by class: median of B - A and of its three steps')
for c in ('line', 'pixel', 'block'):
    sel = [i for i, (k, *_r) in enumerate(big) if cls(k) == c]
    if not sel:
        continue
    kk = [big[i][0] for i in sel]
    arr = lambda x, y: np.array([ARM[x][k][0] - ARM[y][k][0] for k in kk])
    say('  %-6s entries %3d: B-A %+.4f (reference %+.4f, path %+.4f, map %+.4f)' % (c, len(kk), np.median(arr('B', 'A')), np.median(arr('B', "B'")), np.median(arr("B'", 'C')), np.median(arr('C', 'A'))))
