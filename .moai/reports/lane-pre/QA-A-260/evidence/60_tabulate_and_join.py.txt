def esc_rows(text):
    # a pipe inside a cell (|비|, |D_n|) is escaped so that the markdown columns do not split; cells are separated by ' | '
    out = []
    for line in text.splitlines():
        if line.startswith('|') and not line.startswith('|---'):
            cells = line[2:-2].split(' | ')
            line = '| ' + ' | '.join(c.replace('|', chr(92) + '|') for c in cells) + ' |'
        out.append(line)
    return chr(10).join(out) + chr(10)


table260_raw = buf.getvalue()
table260 = esc_rows(table260_raw)
open(EV8 + '50_b16_table_260.md.txt', 'w', encoding='utf-8', newline='\n').write(table260)
print('wrote 50_b16_table_260.md.txt (%d lines)' % table260.count('\n'))


def rows(text):
    out = {}
    order = []
    for line in text.splitlines():
        if not line.startswith('|') or line.startswith('|---') or line.startswith('| 행'):
            continue
        c = [x.strip() for x in line.strip().strip('|').split(' | ')]
        out[c[0]] = c
        order.append(c[0])
    return out, order


t256 = open(EV6 + '50_b16_table.md.txt', encoding='utf-8').read()
r6, order = rows(t256)
r8, order8 = rows(table260_raw)
lines = ['| 행 | 1판 (256) | 1판 (260) | 2판 (256) | 2판 (260) | 합격 기준 | 판정 (256 → 260) |', '|---|---|---|---|---|---|---|']
for k in order8:
    a = r6.get(k)
    b = r8[k]
    if a is None:
        lines.append('| %s | (256 에 없음) | %s | (256 에 없음) | %s | %s | → %s |' % (k, b[1], b[2], b[3], b[4]))
        continue
    lines.append('| %s | %s | %s | %s | %s | %s | %s → %s |' % (k, a[1], b[1], a[2], b[2], b[3], a[4], b[4]))
    lines[-1] = lines[-1].replace('|비|', chr(92) + '|비' + chr(92) + '|').replace('|D_n|', chr(92) + '|D_n' + chr(92) + '|')
open(EV8 + '51_b16_table_side_by_side.md.txt', 'w', encoding='utf-8', newline='\n').write('\n'.join(lines) + '\n')
print('wrote 51_b16_table_side_by_side.md.txt (%d rows)' % (len(lines) - 2))
print('rows in 256 table not in 260: %s' % [k for k in order if k not in r8])
