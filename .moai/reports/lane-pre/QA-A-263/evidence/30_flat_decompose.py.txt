def say(*a):
    print(*a, flush=True)


def prnu_np(y):
    y = y.astype(np.float64).ravel()
    return float(y.std(ddof=1) / y.mean() * 100.0)


def apply_offset_u16(frame, offset):
    x = frame.astype(np.float32) - offset + np.float32(0.5)
    return np.floor(np.clip(x, 0.0, 65535.0)).astype(np.float32)


def rounded_mean_u16(frames):
    acc = np.zeros(frames[0].shape, dtype=np.uint64)
    for f in frames:
        acc += f
    n = len(frames)
    return ((acc + n // 2) // n).astype(np.uint16)


def block_means(a, b):
    h, w = a.shape
    hh, ww = h // b, w // b
    return a[:hh * b, :ww * b].reshape(hh, b, ww, b).mean(axis=(1, 3))


def decompose(Yn):
    """Yn: list of frames normalised to mean 1 (float64 arrays, same group). Returns the dictionary of the decomposition (see the method file)."""
    n = len(Yn)
    stack_mean = np.zeros(Yn[0].shape, dtype=np.float64)
    for y in Yn:
        stack_mean += y
    stack_mean /= n
    var_sum = np.zeros(Yn[0].shape, dtype=np.float64)
    for y in Yn:
        var_sum += (y - stack_mean) ** 2
    V_t = float((var_sum / (n - 1)).mean())
    V_tot = [float(y.ravel().var(ddof=1)) for y in Yn]
    out = dict(n=n, V_t=V_t, V_tot=V_tot)
    R = stack_mean - stack_mean.mean()
    var_avg = float(R.ravel().var(ddof=1))
    out['V_avg'] = var_avg
    out['V_fix_avg'] = max(var_avg - V_t / n, 0.0)
    H, W = R.shape
    row = R.mean(axis=1)
    col = R.mean(axis=0)
    rem = R - row[:, None] - col[None, :] + R.mean()
    bm = block_means(rem, 64)
    low_var = float(bm.var())
    loc_var = float(rem.ravel().var()) - low_var
    parts = dict(row=float(row.var()) - V_t / (n * W), col=float(col.var()) - V_t / (n * H), low=low_var - V_t / (n * 64 * 64), local=loc_var - V_t / n * (1.0 - 1.0 / (64 * 64)))
    out['parts'] = parts
    out['parts_sum'] = sum(parts.values())
    return out, stack_mean


def run_real():
    n = S * S
    init()
    off_path, gain_path = SETD + 'offset.xcal', SETD + 'gain.xcal'
    load_set(off_path, gain_path)
    O = read_payload(off_path, n).reshape(S, S)
    G = read_payload(gain_path, n).reshape(S, S)
    F = {k: np.fromfile(CAL + 'Bright_%d.raw' % k, dtype='<u2').reshape(S, S) for k in range(17, 37)}
    darks = [np.fromfile(CAL + 'Dark_%02d.raw' % k, dtype='<u2').reshape(S, S) for k in range(8, 15)]
    ref = rounded_mean_u16(darks)
    say('clean set v1: offset %s gain %s (sha256 prefixes of the files); gain-frame net signal reference = rounded mean of Dark_08~14' % (hashlib.sha256(open(off_path, 'rb').read()).hexdigest()[:16], hashlib.sha256(open(gain_path, 'rb').read()).hexdigest()[:16]))
    say('=' * 120)
    say('STEP A -- the official FlatResidualPct (xpe_verify_gain, normalized gain) of every flat frame with the current map (group C), against numpy')
    Y = {}
    P_lib = {}
    for k in sorted(F):
        c16 = offset_correct(F[k])
        y = gain_correct(c16)
        mt = verify_gain(c16, y, G)
        P_lib[k] = mt.prnu_after
        Y[k] = y
        say('  Bright_%d: official %.6f %%  numpy %.6f %%  difference %+.2e   (limits: <=1.0 %% phase 1: %s, <=0.5 %% hardening and library gate: %s)' % (k, mt.prnu_after, prnu_np(y), prnu_np(y) - mt.prnu_after, 'yes' if mt.prnu_after <= LIMIT_PHASE1 else 'NO', 'yes' if mt.prnu_after <= LIMIT_HARDENING else 'NO'))
    say('=' * 120)
    say('STEP B -- temporal and fixed parts per frame (group of the same dose; groups of at least two frames; C = gain frames 23~28 are IN-SAMPLE for the current map, C-eval = 29, 30 are not)')
    glist = dict(GROUPS)
    glist['C-eval'] = CEVAL
    dec = {}
    avg_img = {}
    cls = {}
    for g, ids in glist.items():
        Yn = [Y[k].astype(np.float64) / float(Y[k].astype(np.float64).mean()) for k in ids]
        d, avg = decompose(Yn)
        dec[g] = d
        avg_img[g] = avg
        say('  group %-6s frames %s  (n=%d)  single-frame temporal noise P_t = %.4f %%   group-average: P_avg = %.4f %%, temporal part of the average %.4f %%, fixed P_fix,avg = %.4f %%' %
            (g, ids, d['n'], 100 * np.sqrt(d['V_t']), 100 * np.sqrt(d['V_avg']), 100 * np.sqrt(d['V_t'] / d['n']), 100 * np.sqrt(d['V_fix_avg'])))
        for k, vt in zip(ids, d['V_tot']):
            share = d['V_t'] / vt
            fixed = np.sqrt(max(vt - d['V_t'], 0.0)) * 100
            if share >= K_TEMPORAL:
                c = 'TEMPORAL-DOMINATED'
            elif share <= K_FIXED:
                c = 'FIXED-DOMINATED'
            else:
                c = 'MIXED'
            cls[k] = (c, share, fixed)
            say('      Bright_%d: official %.4f %%  = temporal %.4f %% (share of variance %.2f) + fixed %.4f %% (in quadrature)  -> %s' % (k, P_lib[k], 100 * np.sqrt(d['V_t']), share, fixed, c))
        pt = d['parts']
        tot = sum(pt.values())
        say('      shape of the fixed variance of the group average (shares of %.3e): rows %.2f, columns %.2f, low frequency (64x64 blocks) %.2f, local %.2f   [sum check: parts %.3e vs P_fix,avg^2 %.3e]' %
            (tot, pt['row'] / tot if tot > 0 else float('nan'), pt['col'] / tot if tot > 0 else float('nan'), pt['low'] / tot if tot > 0 else float('nan'), pt['local'] / tot if tot > 0 else float('nan'), tot, d['V_fix_avg']))
    say('=' * 120)
    say('STEP C -- does a gain map made from ONE dose cause the residual? Fixed part of the group-average image under maps made from other doses (frames of the map are never evaluated by their own map)')
    maps = {}
    net = {}
    for g, ids in GROUPS.items():
        acc = np.zeros((S, S), dtype=np.float64)
        for k in ids:
            acc += F[k].astype(np.float64) - ref.astype(np.float64)
        raw = (acc / len(ids)).astype(np.float32)
        maps[g] = (raw / np.float32(raw.astype(np.float64).mean())).astype(np.float32)
        net[g] = float(raw.mean())
    chk = float(np.max(np.abs(maps['C'] - G) / np.abs(G)))
    say('  control: my gain map of group C against the generator\'s gain.xcal: max relative difference %.3g' % chk)
    sigma_map = {g: np.sqrt(dec[g]['V_t'] / len(GROUPS[g])) * 100 for g in GROUPS}
    say('  noise a map of each group imprints on a corrected frame, sigma_map = s_X / sqrt(n_X): ' + ', '.join('%s %.4f %%' % (g, sigma_map[g]) for g in GROUPS))
    # all-dose maps (leave the evaluated group out): mean of the normalised maps of the other groups
    def allmap_without(excluded):
        ms = [maps[g] for g in GROUPS if g not in excluded]
        m = np.mean(ms, axis=0, dtype=np.float64)
        return (m / m.mean()).astype(np.float32)

    def sigma_all(excluded):
        # the noise of the mean of the normalised maps of the groups not excluded: independent maps, so the variances add and the mean divides by the number of maps
        sg = [sigma_map[g] for g in GROUPS if g not in excluded]
        return float(np.sqrt(np.sum(np.square(sg))) / len(sg))

    tab = {}
    for eval_g in ['A', 'B', 'C-eval', 'D', 'E']:
        ids = glist[eval_g]
        own = 'C' if eval_g == 'C-eval' else eval_g
        c16 = {k: apply_offset_u16(F[k], O) for k in ids}
        row = {}
        for mname in ['A', 'B', 'C', 'D', 'E', 'ALL']:
            if mname == own:
                continue   # a map is never used on the group it was made from (C-eval is NOT a gain frame, but C is the current map and is shown)
            mp = allmap_without({own}) if mname == 'ALL' else maps[mname]
            Yn = []
            per_frame = []
            for k in ids:
                y = (c16[k] / mp).astype(np.float64)
                per_frame.append(prnu_np(y))
                Yn.append(y / y.mean())
            d, _ = decompose(Yn)
            row[mname] = (per_frame, np.sqrt(max(d['V_fix_avg'], 0.0)) * 100)
        if eval_g == 'C-eval':
            mp = maps['C']
            Yn = []
            per_frame = []
            for k in ids:
                y = (c16[k] / mp).astype(np.float64)
                per_frame.append(prnu_np(y))
                Yn.append(y / y.mean())
            d, _ = decompose(Yn)
            row['C'] = (per_frame, np.sqrt(max(d['V_fix_avg'], 0.0)) * 100)
        tab[eval_g] = row
        say('  evaluated group %-6s frames %s:' % (eval_g, ids))
        for mname, (pf, fx) in sorted(row.items()):
            extra = ' (current map)' if mname == 'C' else ''
            say('      map %-4s single-frame FlatResidualPct %s -> fixed part of the group average %.4f %%   [map noise sigma_map %s]%s' % (mname, ', '.join('%.4f' % v for v in pf), fx, ('%.4f %%' % sigma_map[mname]) if mname in sigma_map else '%.4f %% (mean of the other maps)' % sigma_all({'C' if eval_g == 'C-eval' else eval_g}), extra))
    say('  classification of the map question, per evaluated group (a different map must lower the fixed part of the current map by >= %.0f %% AND by more than the map-noise difference explains):' % (100 * K_MAP_GAIN))
    verdict = {}
    for eval_g, row in tab.items():
        cur = row.get('C')
        if cur is None:
            say('      %-6s: the group made the current map itself (in-sample): not evaluated' % eval_g)
            verdict[eval_g] = 'in-sample'
            continue
        best_name, best_fx = min(((m, v[1]) for m, v in row.items() if m != 'C'), key=lambda t: t[1])
        red = 1.0 - best_fx / cur[1] if cur[1] > 0 else 0.0
        sig_best = sigma_all({'C' if eval_g == 'C-eval' else eval_g}) if best_name == 'ALL' else sigma_map[best_name]   # the first run of this script had no noise figure for the all-dose map and took the current map's
        explained = np.sqrt(max(sigma_map['C'] ** 2 - sig_best ** 2, 0.0))
        gain_pts = cur[1] - best_fx
        v = 'MAP-DOSE EFFECT' if (red >= K_MAP_GAIN and gain_pts > explained + 0.02) else 'NO MAP-DOSE EFFECT'
        verdict[eval_g] = v
        say('      %-6s: current map fixed part %.4f %%, best other map %s %.4f %% (reduction %.0f %%, in points %.4f; the map-noise difference explains %.4f)  -> %s' % (eval_g, cur[1], best_name, best_fx, 100 * red, gain_pts, explained, v))
    # ---- the library cross-check of two numpy maps
    say('=' * 120)
    say('STEP D -- the library against numpy for two other maps (maps E and D written as gain.xcal with the generator\'s header; loaded by the DLL in a child process)')
    import os
    os.makedirs(WORK, exist_ok=True)
    tpl = open(gain_path, 'rb').read()
    hdr = bytearray(tpl[:152])
    for mname in ('E', 'D'):
        payload = maps[mname].astype('<f4').tobytes()
        h = bytearray(hdr)
        h[120:152] = hashlib.sha256(payload).digest()
        path = WORK + 'gain_%s.xcal' % mname
        open(path, 'wb').write(bytes(h) + payload)
        res = subprocess.run([sys.executable, EVD + '30_flat_decompose.py.txt', '--dll-eval', path, '17', '29'], capture_output=True, text=True, env=dict(os.environ, PYTHONIOENCODING='utf-8'))
        lines = [l for l in res.stdout.splitlines() if l.startswith('{')]
        for l in lines:
            r = json.loads(l)
            k = r['frame']
            c16 = apply_offset_u16(F[k], O)
            npv = prnu_np((c16 / maps[mname]))
            say('  map %s, Bright_%d: library %.6f %%  numpy %.6f %%  difference %+.2e  -> %s' % (mname, k, r['prnu_after'], npv, npv - r['prnu_after'], 'AGREE' if abs(npv - r['prnu_after']) < 1e-4 else 'DISAGREE'))
        if res.returncode != 0 or len(lines) != 2:
            say('  child process problem: %s %s' % (res.returncode, res.stderr[-300:]))
    say('=' * 120)
    say('SUMMARY (per frame: official %, temporal share, class)')
    for k in sorted(cls):
        say('  Bright_%d: %.4f %%  temporal share %.2f  fixed %.4f %%  %s' % (k, P_lib[k], cls[k][1], cls[k][2], cls[k][0]))
    say('  map question: ' + ', '.join('%s=%s' % (g, v) for g, v in verdict.items()))


def run_dll_eval(gain_path, frames):
    n = S * S
    init()
    load_set(SETD + 'offset.xcal', gain_path)
    G = read_payload(gain_path, n).reshape(S, S)
    for k in frames:
        raw = np.fromfile(CAL + 'Bright_%s.raw' % k, dtype='<u2').reshape(S, S)
        c16 = offset_correct(raw)
        y = gain_correct(c16)
        mt = verify_gain(c16, y, G)
        print(json.dumps(dict(frame=int(k), prnu_after=mt.prnu_after)), flush=True)


def run_selftest():
    """Synthetic: temporal noise 1.0 % of the mean, a fixed row pattern of 0.30 % and a fixed local pattern of 0.40 % (white). The decomposition must give them back."""
    rng = np.random.default_rng(3)
    H = W = 512
    row = rng.normal(0, 0.003, size=(H, 1))
    loc = rng.normal(0, 0.004, size=(H, W))
    Yn = [1.0 + row + loc + rng.normal(0, 0.010, size=(H, W)) for _ in range(4)]
    Yn = [y / y.mean() for y in Yn]
    d, _ = decompose(Yn)
    fixed_tot = np.sqrt(d['V_tot'][0] - d['V_t']) * 100
    say('selftest: P_t = %.4f %% (put in 1.0000), single-frame fixed part %.4f %% (put in %.4f), shape of the fixed variance: ' % (100 * np.sqrt(d['V_t']), fixed_tot, 100 * np.sqrt(0.003 ** 2 + 0.004 ** 2)) +
        'rows %.2f (put in %.2f) local %.2f (put in %.2f)' % (d['parts']['row'] / d['parts_sum'], 0.003 ** 2 / (0.003 ** 2 + 0.004 ** 2), d['parts']['local'] / d['parts_sum'], 0.004 ** 2 / (0.003 ** 2 + 0.004 ** 2)))
    ok = abs(100 * np.sqrt(d['V_t']) - 1.0) < 0.02 and abs(fixed_tot - 0.5) < 0.03 and abs(d['parts']['row'] / d['parts_sum'] - 0.36) < 0.05
    say('selftest result: %s' % ('PASS' if ok else 'FAIL'))
    assert ok
