def items():
    res = []
    for l in open(TRUTH, encoding='utf-8'):

    return res


IT = items()

# ------------------------------------------------------------------------------------------------ cause bits of the lines
groups = []
for l in list(open(EV + '68_lines_planted.tsv.txt', encoding='utf-8'))[1:]:
    p = l.rstrip('\n').split('\t')
    groups.append({'ori': p[0], 'idx': int(p[1]), 'width': int(p[2]), 'a': int(p[3]), 'b': int(p[4]), 'causes': int(p[5])})


def match(it):
    x, y, w, h = it['rects'][0]
    ori = 'row' if w > h else 'column'
    idx, wd = (y, h) if ori == 'row' else (x, w)
    a, ln = (x, w) if ori == 'row' else (y, h)
    b = a + ln
    best = None
    for g in groups:
        if g['ori'] != ori or g['idx'] > idx + wd or g['idx'] + g['width'] - 1 < idx - 1:
            continue
        ov = min(g['b'], b) - max(g['a'], a)
        if ov >= 0.5 * ln and (best is None or ov > best[0]):
            best = (ov, g)
    return (best[1] if best else None), a, b

print('%-16s %-6s %6s %4s %4s %4s %4s   %s' % ('kind', 'level', 'found', 'R', 'L', 'N', 'S', 'combinations'))
LINES = [i for i in RESP if i['shape'] != 'pixel']
for kind in ('gain_loss', 'efficiency_loss'):
    for lv in LVL:
        sel = [i for i in LINES if i['kind'] == kind and i['level'] == lv]
        found, bits, combo = 0, [0, 0, 0, 0], []
        for it in sel:
            g, a, b = match(it)
            if g is None:
                combo.append('-')
                continue
            found += 1
            for k, m in enumerate((1, 2, 4, 8)):
                bits[k] += 1 if g['causes'] & m else 0
            combo.append(''.join(c for c, m in zip('RLNS', (1, 2, 4, 8)) if g['causes'] & m) or '0')
        print('%-16s %-6s %3d/%-2d %4d %4d %4d %4d   %s' % (kind, lv, found, len(sel), bits[0], bits[1], bits[2], bits[3], ' '.join(combo)))

        errs.append((it['id'], it['kind'], it['shape'], it['level'], a - g['a'], g['b'] - b))
ok = [e for e in errs if e[4] is not None]
