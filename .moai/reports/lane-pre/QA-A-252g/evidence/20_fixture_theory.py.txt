"""QA-A-252g: the theoretical detection rate of the pixel noise-cost check for the grid pixels of the unit test fixture (256 x 256, dose groups of three frames at 2000, 4500, 9000, 18000).
Efficiency loss r = 1/3 in the fixture (signal scaled, the quantum noise 0.25 x signal follows it, the readout noise 3 stays): the gain-corrected variance against the neighbours' is
rho_g = (0.25 s / r + 9 / r^2) / (0.25 s + 9). The pooled statistic must exceed tau (P(chi2_8 / 8 > tau) = 1 / 65536) and the ratio of at least 2 of the 4 groups must reach 2."""
import math
import numpy as np

def gamma_q(a, x):
    gln = math.lgamma(a)
    if x < a + 1.0:
        ap, s, d = a, 1.0 / a, 1.0 / a
        for _ in range(5000):
            ap += 1.0; d *= x / ap; s += d
            if abs(d) < abs(s) * 1e-15: break
        return 1.0 - s * math.exp(-x + a * math.log(x) - gln)
    b, c = x + 1.0 - a, 1.0 / 1e-300
    d = 1.0 / b; h = d
    for i in range(1, 5000):
        an = -i * (i - a); b += 2.0
        d = an * d + b; d = d if abs(d) > 1e-300 else 1e-300
        c = b + an / c; c = c if abs(c) > 1e-300 else 1e-300
        d = 1.0 / d; de = d * c; h *= de
        if abs(de - 1.0) < 1e-15: break
    return math.exp(-x + a * math.log(x) - gln) * h

def tau(nu, p):
    a = 0.5 * nu; lo, hi = 1.0, 1e6
    for _ in range(300):
        mid = 0.5 * (lo + hi)
        if gamma_q(a, a * mid) > p: lo = mid
        else: hi = mid
    return 0.5 * (lo + hi)

doses = [2000.0, 4500.0, 9000.0, 18000.0]
nus = np.array([2.0, 2.0, 2.0, 2.0]); t = tau(8, 1.0 / 65536)
print('tau(8, 1/65536) = %.3f' % t)
rng = np.random.default_rng(7); N = 2_000_000
for r in (1 / 3, 0.5, 1.0):
    rho = np.array([(0.25 * s / r + 9 / r ** 2) / (0.25 * s + 9) for s in doses])
    chi = np.stack([rng.chisquare(2, N) / 2 for _ in range(4)], axis=1) * rho
    pooled = (chi * nus).sum(axis=1) / 8.0
    q = ((pooled > t) & (2 * (chi >= 2.0).sum(axis=1) >= 4)).mean()
    print('r = %.3f  rho per dose %s  detection probability per pixel %.4f  (144 pixels: expected %.1f, sd %.1f)' % (r, np.round(rho, 3).tolist(), q, 144 * q, math.sqrt(144 * q * (1 - q))))
