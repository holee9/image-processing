for f in (TF, GC):

ARMS = {

def build():
    r = subprocess.run(['cmd.exe', '/c', S + 'buildall.bat'], capture_output=True, text=True)
    return 'BUILD_EXIT=0' in r.stdout, r.stdout


def tests():
    r = subprocess.run([R + 'build/ci-preprocess/bin/xpe_preprocess_tests.exe', '--gtest_filter=GhostOracle.*'],
                       cwd=R + 'modules/preprocess', capture_output=True, text=True)
    failed = sorted(set(re.findall(r'^\[  FAILED  \] ([A-Za-z]\S+)', r.stdout, re.M)))
    return failed, r.returncode


for name in sys.argv[1:]:
    f, old, new = ARMS[name]
    keep = S + 'keep225_' + os.path.basename(f)
    shutil.copy(keep, f)
    src = open(f, 'rb').read().decode('utf-8')
    crlf = '\r\n' in src
    s = src.replace('\r\n', '\n')
    assert s.count(old) == 1, (name, s.count(old))
    s = s.replace(old, new)
    open(f, 'wb').write((s.replace('\n', '\r\n') if crlf else s).encode('utf-8'))
    ok, out = build()
    if not ok:
        print(name, 'BUILD FAILED'); print(out[-1200:])
    else:
        failed, rc = tests()
        crashed = rc not in (0, 1) or (rc != 0 and not failed)
        tag = (' [PROCESS DIED, exit %d]' % rc) if crashed else ''
        print('%-42s -> %d red%s: %s' % (name, len(failed), tag, ', '.join(failed) if failed else ('NONE (arm did not fire)' if not crashed else 'no FAILED line')))
    shutil.copy(keep, f)
