"""QA-A-225 design evidence (NOT product code): what GhostRemovalPct would read for xpe_ghost_correct, with the lag
truth generated by an INDEPENDENT model (Starman 2012 discrete-time LTI impulse response, TDS-CALIB-001 4.4.2 values).
The module under test is the real xpe_preprocess.dll; a numpy replica of its tier-1 recursion is used only to search a
configuration and is checked against the DLL first."""
import ctypes, os, sys, math
import numpy as np

BIN = 'D:/workspace-github/xpe-pre/build/ci-preprocess/bin'
os.add_dll_directory(BIN)
dll = ctypes.CDLL(BIN + '/xpe_preprocess.dll')
W = H = 8
SAT = 65535.0


class Buf(ctypes.Structure):
    _fields_ = [('width', ctypes.c_uint32), ('height', ctypes.c_uint32), ('bitsAllocated', ctypes.c_uint32),
                ('bitsStored', ctypes.c_uint32), ('format', ctypes.c_int32), ('data', ctypes.c_void_p),
                ('dataSize', ctypes.c_size_t)]


dll.xpe_ghost_create.argtypes = [ctypes.c_uint32, ctypes.c_uint32, ctypes.c_char_p, ctypes.POINTER(ctypes.c_void_p)]
dll.xpe_ghost_correct.argtypes = [ctypes.c_void_p, ctypes.POINTER(Buf), ctypes.c_void_p]
dll.xpe_ghost_destroy.argtypes = [ctypes.c_void_p]

# Starman et al. 2012 eq.(1): h(k) = b0*delta(k) + sum_n b_n exp(-a_n k), sum of ALL b (incl. b0) = 1.
# a_n, b_n: the values TDS-CALIB-001 4.4.2 lists for "Varex 4030CB" (a_n per frame).
A = np.array([2.5e-3, 2.1e-2, 1.6e-1, 7.6e-1])
B = np.array([7.1e-6, 1.1e-4, 1.7e-3, 1.8e-2])
B0 = 1.0 - B.sum()


def truth_forward(x):
    """y(k) = sum_j x(k-j) h(j), h(0)=b0+sum b_n, h(j>0)=sum b_n exp(-a_n j). Driven by the TRUE exposure x."""
    n = len(x)
    y = np.zeros(n)
    s = np.zeros(4)               # s_n(k) = sum_{j>=1} exp(-a_n j) x(k-j)
    for k in range(n):
        y[k] = x[k] * (B0 + B.sum()) + (B * s).sum()
        s = np.exp(-A) * (s + x[k])
    return y


def sequence(level_frac, n_pre=5, n_exp=200, n_blank=50, sigma=0.0, seed=1):
    x = np.concatenate([np.zeros(n_pre), np.full(n_exp, level_frac * SAT), np.zeros(n_blank)])
    y = truth_forward(x)
    if sigma > 0:
        y = y + np.random.default_rng(seed).normal(0, sigma, y.shape)
    return x, y, n_pre, n_exp, n_blank


def run_dll(y, cfg):
    hnd = ctypes.c_void_p()
    rc = dll.xpe_ghost_create(W, H, cfg.encode(), ctypes.byref(hnd))
    assert rc == 0, 'create rc=%d cfg=%s' % (rc, cfg)
    out = []
    meta = ctypes.create_string_buffer(96)
    for k, v in enumerate(y):
        px = np.full((H, W), v, np.float32)
        buf = Buf(W, H, 32, 32, 1, px.ctypes.data, px.nbytes)
        ctypes.c_uint64.from_buffer(meta, 80).value = k + 1          # acquisitionTime: integer seconds, 1 per frame
        rc = dll.xpe_ghost_correct(hnd, ctypes.byref(buf), meta)
        assert rc == 0, 'correct rc=%d at frame %d' % (rc, k)
        out.append(float(px.mean()))
    dll.xpe_ghost_destroy(hnd)
    return np.array(out)


def replica_tier1(y, a1, t1, a2, t2):
    """numpy replica of ghost_tier1 for uniform frames (dt = 1 frame)."""
    d1, d2 = math.exp(-1.0 / t1), math.exp(-1.0 / t2)
    h1 = h2 = 0.0
    out = np.zeros(len(y))
    for k, raw in enumerate(y):
        c = raw - a1 * h1 - a2 * h2
        h1 = d1 * h1 + raw
        h2 = d2 * h2 + raw
        out[k] = c if c > 0 else 0.0
    return out


def metrics(y, corrected, n_pre, n_exp, n_blank):
    """E2E protocol 5.6: LagResidualPct(k) = 100*|mean(Y_blank_k) - mean(DarkRef)| / mean(ExposureSignal);
    GhostRemovalPct = 100*(LagRaw - LagResidual)/LagRaw. DarkRef = the pre-exposure frames of the same output;
    ExposureSignal = the last exposure frame as the module received it (raw)."""
    exp_sig = y[n_pre + n_exp - 1]
    dark_raw = y[:n_pre].mean()
    dark_cor = corrected[:n_pre].mean()
    res = {}
    for k in (1, 10, 50):
        i = n_pre + n_exp + k - 1
        raw = 100 * abs(y[i] - dark_raw) / exp_sig
        cor = 100 * abs(corrected[i] - dark_cor) / exp_sig
        res[k] = (raw, cor, 100 * (raw - cor) / max(raw, 1e-12))
    return res


def line(label, m):
    return '%-46s ' % label + ' | '.join('k=%-2d raw %.4f%% res %.4f%% removal %6.2f%%' % (k, *m[k]) for k in (1, 50))


if __name__ == '__main__':
    print('truth: Starman 2012 eq.(1) LTI, a_n=%s b_n=%s b0=%.6f' % (A, B, B0))
    # sanity: the raw 1st-blank-frame lag this truth implies, against Starman Table 3 (uncorrected, 27% = 2.8%)
    for lv in (0.02, 0.27, 0.92):
        x, y, p, e, b = sequence(lv)
        print('  level %2d%%: raw 1st blank lag = %.3f%% of exposure signal (Starman Table 3 uncorrected: 2%%->3.7, 27%%->2.8, 92%%->2.4)'
              % (round(lv * 100), 100 * abs(y[p + e] - y[:p].mean()) / y[p + e - 1]))
    x, y, p, e, b = sequence(0.27)
    print('\nreplica vs DLL (tier 1, default config): max abs diff = %.3e ADU'
          % np.abs(replica_tier1(y, 0.9, 1.0, 0.05, 20.0) - run_dll(y, '{"tier":1}')).max())
    print('\n-- 27% saturation, defaults of the module --')
    for t in (1, 2, 3):
        print(line('DLL default tier %d' % t, metrics(y, run_dll(y, '{"tier":%d}' % t), p, e, b)))
    print(line('no correction (control: must be 0%)', metrics(y, y.copy(), p, e, b)))
    # search the best 2-term tier-1 configuration for THIS sequence (the ceiling of the module's model class)
    rng = np.random.default_rng(0)

    def cost(q):
        c = replica_tier1(y, *q)
        i0 = p + e
        return float(np.abs(c[i0:i0 + b]).mean())
    best = (0.9, 1.0, 0.05, 20.0); bc = cost(best)
    for scale in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02):
        for _ in range(4000):
            q = np.array(best) * np.exp(rng.normal(0, scale, 4))
            q[0] = min(q[0], 5.0); q[2] = min(q[2], 5.0); q[1] = max(q[1], 0.2); q[3] = max(q[3], 0.2)
            c = cost(q)
            if c < bc: best, bc = tuple(q), c
    cfg = '{"tier":1,"alpha1":%.9g,"tau1":%.9g,"alpha2":%.9g,"tau2":%.9g}' % best
    print('\nbest 2-term tier-1 config found (search on the replica, mean |blank| over 50 frames):', cfg)
    print(line('DLL tier 1 with that config (27%)', metrics(y, run_dll(y, cfg), p, e, b)))
    print('\n-- same config on the OTHER exposure levels (held out: the config was fitted on 27% only) --')
    for lv in (0.02, 0.92):
        xs, ys, pp, ee, bb = sequence(lv)
        print(line('level %d%%, fitted-on-27%% config' % round(lv * 100), metrics(ys, run_dll(ys, cfg), pp, ee, bb)))
        print(line('level %d%%, module default tier 1' % round(lv * 100), metrics(ys, run_dll(ys, '{"tier":1}'), pp, ee, bb)))
    print('\n-- noise floor: sigma = 5 ADU on every frame (clamp at 0 rectifies the dark frames) --')
    xn, yn, pn, en, bn = sequence(0.27, sigma=5.0)
    print(line('no correction, noisy', metrics(yn, yn.copy(), pn, en, bn)))
    print(line('DLL tier 1 fitted config, noisy', metrics(yn, run_dll(yn, cfg), pn, en, bn)))
