import ghost_proto as g

SAT = g.SAT


def trap_forward(x, Q=(300.0, 400.0, 600.0, 800.0)):
    """y(k) = x(k) + sum_n release_n(k). Trap class n fills with efficiency b_n while empty capacity remains and
    releases at rate a_n per frame. Low signal behaves like the LTI model; high signal saturates the traps, so the
    RELATIVE lag falls with exposure level (the trend Starman Table 3 shows: 3.7% -> 2.8% -> 2.4%)."""
    q = np.zeros(4)
    Qv = np.array(Q)
    y = np.zeros(len(x))
    for k, xk in enumerate(x):
        rel = q * (1 - np.exp(-g.A))
        q = q - rel
        fill = np.maximum(xk * g.B * (1 - q / Qv), 0.0)
        q = q + fill
        y[k] = xk + rel.sum()
    return y


def seq_from(forward, level, n_pre=5, n_exp=200, n_blank=50):
    x = np.concatenate([np.zeros(n_pre), np.full(n_exp, level * SAT), np.zeros(n_blank)])
    return x, forward(x), n_pre, n_exp, n_blank


def retention(x, corrected, n_pre, n_exp):
    """mean(corrected exposure frames, from the 50th on) / true exposure. 1.0 = the signal is kept."""
    a, b = n_pre + 49, n_pre + n_exp
    return float(corrected[a:b].mean() / x[a:b].mean())


def full(label, x, y, cor, p, e, b):
    m = g.metrics(y, cor, p, e, b)
    return '%-40s retention %7.3f | k=1 removal %6.2f%% (res %.4f%%) | k=50 removal %6.2f%% (res %.4f%%)' % (
        label, retention(x, cor, p, e), m[1][2], m[1][1], m[50][2], m[50][1])
