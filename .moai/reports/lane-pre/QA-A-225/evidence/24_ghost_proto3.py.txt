"""QA-A-225 design evidence, round 3 (NOT product code): tight retention band; a STABLE self-model truth; and a
falling step to a NON-ZERO level, which the zero clamp cannot hide."""
import sys, math
import numpy as np
sys.path.insert(0, 'C:/Users/drake/AppData/Local/Temp/claude/D--workspace-github-xpe-pre/68fc65a2-03aa-4eda-a5f4-cfdd0accbdc5/scratchpad')
import ghost_proto as g
import ghost_proto2 as g2

SAT = g.SAT
x, y, p, e, b = g2.seq_from(lambda s: g.truth_forward(s), 0.27)
i0 = p + e


def search(lo, hi, seed=0):
    rng = np.random.default_rng(seed)

    def cost(q):
        c = g.replica_tier1(y, *q)
        r = c[p + 49:p + e].mean() / x[p + 49:p + e].mean()
        pen = 0.0 if lo <= r <= hi else 1e6 * abs(r - 1)
        return float(np.abs(c[i0:i0 + b]).mean()) + pen
    best = (0.02, 3.0, 0.02, 40.0); bc = cost(best)
    for scale in (1.0, 0.6, 0.3, 0.15, 0.07, 0.03, 0.01):
        for _ in range(6000):
            q = np.array(best) * np.exp(rng.normal(0, scale, 4))
            q[1] = max(q[1], 0.3); q[3] = max(q[3], 0.3)
            c = cost(q)
            if c < bc: best, bc = tuple(q), c
    return best


def cfg_of(best):
    return '{"tier":1,"alpha1":%.9g,"tau1":%.9g,"alpha2":%.9g,"tau2":%.9g}' % best


print('== ceiling of the 2-term tier-1 class on the LTI truth (27%), by retention band ==')
for lo, hi in ((0.995, 1.005), (0.999, 1.001), (0.9999, 1.0001)):
    best = search(lo, hi)
    cor = g.run_dll(y, cfg_of(best))
    print('band [%.4f, %.4f]  %s' % (lo, hi, cfg_of(best)))
    print('   ' + g2.full('DLL tier 1', x, y, cor, p, e, b))
tight = search(0.999, 1.001)
tcfg = cfg_of(tight)

print('\n== STABLE self-model truth (the trap the card names): lag generated by the module\'s own recursion ==')
a1, t1, a2, t2 = 0.02, 3.0, 0.003, 30.0
d1, d2 = math.exp(-1.0 / t1), math.exp(-1.0 / t2)
xs = np.concatenate([np.zeros(5), np.full(200, 0.27 * SAT), np.zeros(50)])
ys = np.zeros(len(xs)); h1 = h2 = 0.0
for k, xk in enumerate(xs):
    ys[k] = xk + a1 * h1 + a2 * h2
    h1 = d1 * h1 + ys[k]; h2 = d2 * h2 + ys[k]
selfcfg = '{"tier":1,"alpha1":%g,"tau1":%g,"alpha2":%g,"tau2":%g}' % (a1, t1, a2, t2)
print('raw 1st blank lag of this generator: %.3f%% of the exposure signal' % (100 * abs(ys[205]) / ys[204]))
print(g2.full('self-model truth, module with the SAME config', xs, ys, g.run_dll(ys, selfcfg), 5, 200, 50))
print(g2.full('self-model truth, module with a WRONG config', xs, ys, g.run_dll(ys, '{"tier":1,"alpha1":0.01,"tau1":8,"alpha2":0.001,"tau2":60}'), 5, 200, 50))
print('   -> with its own parameters the module "removes" 100%; a wrong configuration shows what the gate can tell apart')

print('\n== falling step to a NON-ZERO level (27% -> 5%): signed residual, no clamp involved ==')
n_hi, n_lo = 200, 100
xf = np.concatenate([np.full(n_hi, 0.27 * SAT), np.full(n_lo, 0.05 * SAT)])
yf = g.truth_forward(xf)


def fall(label, cor):
    out = []
    for k in (1, 50):
        i = n_hi + k - 1
        out.append('k=%-2d signed residual %+.4f%% of new level' % (k, 100 * (cor[i] - xf[i]) / xf[i]))
    print('%-34s ' % label + ' | '.join(out) + ' | retention@27%%: %.4f' % (cor[n_hi - 150:n_hi].mean() / xf[n_hi - 150:n_hi].mean()))


fall('no correction', yf)
fall('module defaults tier 1', g.run_dll(yf, '{"tier":1}'))
fall('tight-band fitted tier 1', g.run_dll(yf, tcfg))
fall('0.5%-band fitted tier 1', g.run_dll(yf, cfg_of(search(0.995, 1.005))))
