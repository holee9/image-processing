"""QA-A-225 M3-a (NOT product code): why is GhostRemovalPct at k=50 negative in 16 of 36 noisy (sigma 5 ADU) module rows?
Hypotheses:
  H1 (metric): the corrected DARK frames are clamp-rectified (c = max(raw, 0) while the history is 0: mean = sigma/sqrt(2*pi))
      while the over-correcting module clamps the BLANK frames to exactly 0, so LagResidual = |0 - DarkRef_cor| = the rectification
      bias, whatever the lag is. When LagRaw(k) is smaller than that bias, removal < 0.
  H2 (corrector): the corrector amplifies the noise (so blank frames carry more noise than raw).
Measured on the real xpe_preprocess.dll with per-pixel noise (16x16 frames)."""
import ctypes, os, math, sys
import numpy as np

BIN = 'D:/workspace-github/xpe-pre/build/ci-preprocess/bin'
os.add_dll_directory(BIN)
dll = ctypes.CDLL(BIN + '/xpe_preprocess.dll')
W = H = 16
N = W * H
SAT = 65535.0


class Buf(ctypes.Structure):
    _fields_ = [('width', ctypes.c_uint32), ('height', ctypes.c_uint32), ('bitsAllocated', ctypes.c_uint32),
                ('bitsStored', ctypes.c_uint32), ('format', ctypes.c_int32), ('data', ctypes.c_void_p),
                ('dataSize', ctypes.c_size_t)]


dll.xpe_ghost_create.argtypes = [ctypes.c_uint32, ctypes.c_uint32, ctypes.c_char_p, ctypes.POINTER(ctypes.c_void_p)]
dll.xpe_ghost_correct.argtypes = [ctypes.c_void_p, ctypes.POINTER(Buf), ctypes.c_void_p]

A = np.array([2.5e-3, 2.1e-2, 1.6e-1, 7.6e-1])
B = np.array([7.1e-6, 1.1e-4, 1.7e-3, 1.8e-2])
B0 = 1.0 - B.sum()


def lti(x):
    y = np.zeros(len(x)); s = np.zeros(4)
    for k in range(len(x)):
        y[k] = x[k] * (B0 + B.sum()) + (B * s).sum()
        s = np.exp(-A) * (s + x[k])
    return y


def trap(x, Q=(300.0, 400.0, 600.0, 800.0)):
    q = np.zeros(4); Qv = np.array(Q); y = np.zeros(len(x))
    for k, xk in enumerate(x):
        rel = q * (1 - np.exp(-A)); q = q - rel
        q = q + np.maximum(xk * B * (1 - q / Qv), 0.0)
        y[k] = xk + rel.sum()
    return y


def frames_of(y, sigma, seed):
    rng = np.random.default_rng(seed)
    return [(np.full((H, W), v) + (rng.normal(0, sigma, (H, W)) if sigma > 0 else 0)).astype(np.float32) for v in y]


def run(cfg, frames):
    hnd = ctypes.c_void_p()
    assert dll.xpe_ghost_create(W, H, cfg.encode() if cfg else None, ctypes.byref(hnd)) == 0
    meta = ctypes.create_string_buffer(96)
    out = []
    for f in frames:
        px = f.copy()
        buf = Buf(W, H, 32, 32, 1, px.ctypes.data, px.nbytes)
        assert dll.xpe_ghost_correct(hnd, ctypes.byref(buf), meta) == 0
        out.append(px)
    dll.xpe_ghost_destroy(hnd)
    return out


def cfg_of(tier, a1, t1, a2, t2):
    return '{"tier":%d,"alpha1":%g,"tau1":%g,"alpha2":%g,"tau2":%g}' % (tier, a1, t1, a2, t2)


npre, nexp, nblank = 5, 200, 50
sets = {'0.02/3/0.003/30': (0.02, 3.0, 0.003, 30.0), '0.1/1/0.01/20': (0.1, 1.0, 0.01, 20.0)}
print('H1 prediction: DarkRef_cor = sigma/sqrt(2*pi) = %.4f ADU at sigma 5' % (5 / math.sqrt(2 * math.pi)))
print()
print('%-4s %-4s %-16s tier | dark_raw dark_cor | blank_k50: raw_mean cor_mean clamped-to-0 share | res(k50)%% | LagRaw(k50)%% | removal(k50)%%' % ('tru', 'lvl', 'set'))
rows = []
for tname, gen in (('T1', lti), ('T2', trap)):
    for lv in (0.02, 0.27, 0.92):
        x = np.concatenate([np.zeros(npre), np.full(nexp, lv * SAT), np.zeros(nblank)])
        y = gen(x)
        fr = frames_of(y, 5.0, 7)
        expo = float(np.mean(fr[npre + nexp - 1]))
        for sname, (a1, t1, a2, t2) in sets.items():
            for tier in (1, 2, 3):
                out = run(cfg_of(tier, a1, t1, a2, t2), fr)
                dr = np.mean([f.mean() for f in fr[:npre]]); dc = np.mean([f.mean() for f in out[:npre]])
                i = npre + nexp + 50 - 1
                rawm = float(fr[i].mean()); corm = float(out[i].mean()); zero_share = float((out[i] == 0).mean())
                lagraw = 100 * abs(rawm - dr) / expo; res = 100 * abs(corm - dc) / expo
                rem = 100 * (lagraw - res) / lagraw
                rows.append((tname, lv, sname, tier, dr, dc, rawm, corm, zero_share, res, lagraw, rem))
                print('%-4s %-4.0f %-16s %d    | %8.4f %8.4f | %9.4f %9.4f %6.3f | %8.4f | %8.4f | %9.2f' % (tname, lv * 100, sname, tier, dr, dc, rawm, corm, zero_share, res, lagraw, rem))
neg = [r for r in rows if r[-1] < 0]
print('\nnegative k=50 removal rows: %d of %d' % (len(neg), len(rows)))
print('  of them with corrected blank frame 100%% clamped to 0 : %d' % sum(1 for r in neg if r[8] == 1.0))
print('  LagRaw(k50) range over negative rows: %.4f..%.4f %%, over positive rows: %.4f..%.4f %%' % (
    min(r[10] for r in neg), max(r[10] for r in neg), min(r[10] for r in rows if r[-1] >= 0), max(r[10] for r in rows if r[-1] >= 0)))
print('  corrected DARK mean over all rows: %.4f..%.4f (H1 predicts %.4f)' % (min(r[5] for r in rows), max(r[5] for r in rows), 5 / math.sqrt(2 * math.pi)))

# H1 test with the sigma varied: residual should scale with sigma/sqrt(2*pi), not with anything the corrector does to the lag
print('\nsigma sweep, T1 27%, tier 1 0.02/3/0.003/30 (res = |cor_blank - cor_dark|):')
x = np.concatenate([np.zeros(npre), np.full(nexp, 0.27 * SAT), np.zeros(nblank)]); y = lti(x)
for sg in (0.0, 1.0, 2.0, 5.0, 10.0, 20.0):
    fr = frames_of(y, sg, 7); out = run(cfg_of(1, 0.02, 3.0, 0.003, 30.0), fr)
    dc = np.mean([f.mean() for f in out[:npre]]); i = npre + nexp + 50 - 1
    print('  sigma %5.1f: cor_dark %.4f (pred %.4f) cor_blank50 %.4f' % (sg, dc, sg / math.sqrt(2 * math.pi), out[i].mean()))

# H2: does the corrector amplify the noise? spatial std of the corrected EXPOSURE frame (no clamp involved) vs raw
print('\nH2: spatial std of the corrected exposure frame (frame 150) vs the raw one (sigma 5), T1 27%:')
fr = frames_of(lti(x), 5.0, 7)
for sname, (a1, t1, a2, t2) in sets.items():
    for tier in (1, 2, 3):
        out = run(cfg_of(tier, a1, t1, a2, t2), fr)
        i = npre + 150
        print('  %-16s tier %d: raw std %.3f -> corrected std %.3f (gain %.3f)  [clamped pixels %.3f]' % (sname, tier, fr[i].std(), out[i].std(), out[i].std() / fr[i].std(), (out[i] == 0).mean()))
