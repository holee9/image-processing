class Buf(ctypes.Structure):
    _fields_ = [('width', ctypes.c_uint32), ('height', ctypes.c_uint32), ('bitsAllocated', ctypes.c_uint32), ('bitsStored', ctypes.c_uint32), ('format', ctypes.c_int32), ('data', ctypes.c_void_p), ('dataSize', ctypes.c_size_t)]


class Meta(ctypes.Structure):
    _fields_ = [('bodyPart', ctypes.c_char * 64), ('kVp', ctypes.c_float), ('mAs', ctypes.c_float), ('SID_mm', ctypes.c_float), ('pixelPitch_mm', ctypes.c_float), ('acquisitionTime', ctypes.c_uint64), ('flags', ctypes.c_uint32)]

lib.xpe_preprocess_pipeline_out.argtypes = [ctypes.POINTER(Buf), ctypes.POINTER(Buf), ctypes.POINTER(Meta), ctypes.c_void_p, ctypes.c_void_p, ctypes.c_char_p]

def load_set(d):
    common.xpe_clear_alerts()
    for fn, nm in ((lib.xpe_calib_load_offset, 'offset'), (lib.xpe_calib_load_gain, 'gain'), (lib.xpe_calib_load_defect_map, 'defect')):
        r = fn((d + '/' + nm + '.xcal').encode())
        assert r == 0, (nm, r)


def run(frame, cfg):
    raw = np.fromfile(frame, dtype='<u2')
    assert raw.size == S * S
    out = np.zeros(S * S, dtype=np.float32)
    i = Buf(S, S, 16, 16, 0, raw.ctypes.data, raw.nbytes)
    o = Buf(S, S, 32, 32, 1, out.ctypes.data, out.nbytes)
    m = Meta()
    r = lib.xpe_preprocess_pipeline_out(ctypes.byref(i), ctypes.byref(o), ctypes.byref(m), None, None, cfg)
    common.xpe_clear_alerts()
    assert r == 0, r
    return out.reshape(S, S)

def sigma_local(C):
    """population standard deviation over the 11 x 11 window (clipped at the border) of C, as the tool computes it"""
    x = C.astype(np.float64) - float(np.median(C))
    out = {}
    def box(a):
        p = np.pad(a, ((1, 0), (1, 0)))
        cs = np.cumsum(np.cumsum(p, axis=0), axis=1)
        idx = np.arange(S)
        lo, hi = np.maximum(idx - 5, 0), np.minimum(idx + 5, S - 1) + 1
        return cs[hi][:, hi] - cs[lo][:, hi] - cs[hi][:, lo] + cs[lo][:, lo]
    s1, s2 = box(x), box(x * x)
    idx = np.arange(S)
    cnt = (np.minimum(idx + 5, S - 1) + 1 - np.maximum(idx - 5, 0)).astype(np.float64)
    m = cnt[:, None] * cnt[None, :]
    mean = s1 / m
    return np.sqrt(np.maximum(0.0, s2 / m - mean * mean))


KEYS = {}
for k, it in items.items():
    if it['cls0'] == 'control':
        continue
    KEYS[k] = it['cls0'] + '/' + it['kind'] + '/' + it['shape'] + '/' + it['level']
log = []


def P(s=''):
    print(s)
    log.append(s)


res = {}
for ed, jf in (('v1', '44_correct_v1.json.txt'), ('v2', '45_correct_v2.json.txt')):
    jd = json.load(open(EV8 + jf, encoding='utf-8'))
    for fi, f in enumerate((29, 30)):
        C = np.load(O70 + 'C_%s_%d.npy' % (ed, f))
        I = np.load(O70 + 'I_%s_%d.npy' % (ed, f))
        sl = sigma_local(C)
        z = np.abs(I.astype(np.float64) - C.astype(np.float64)) / np.where(sl > 0, sl, np.nan)
        np.save(O70 + 'Z_%s_%d.npy' % (ed, f), z.astype(np.float32))
        med = {}
        for g in jd['frames'][fi]['correction_error']:
            med[g['group']] = g['corrected']['median_z']
        bykey = {}
        for k, key in KEYS.items():
            bykey.setdefault(key, []).append(k)
        worst = 0.0
        miss = 0
        for key, ks in bykey.items():
            m = np.isin(ids, ks)
            zz = z[m]
            zz = zz[np.isfinite(zz)]
            mine = float(np.sort(zz)[zz.size // 2]) if zz.size else None   # the tool's median: v[floor(m / 2)] of the sorted values, the upper one for an even count
            res[(ed, fi, key)] = mine
            if key not in med or mine is None or med[key] in (None, 'uncomputable'):
                miss += 1
                continue
            worst = max(worst, abs(mine - med[key]))
        P('Z0 %s frame %d: groups %d, groups without a tool value %d, largest |difference| of the group median %.6f' % (ed, f, len(bykey), miss, worst))
cells = 0
bad = []
for (ed, fi, key), v in res.items():
    if ed != 'v1' or v is None:
        continue
    w = res.get(('v2', fi, key))
    if w is None:
        continue
    cells += 1
    if w > v + 1e-9:
        bad.append((fi, key, v, w))
