def zscore(flag_idx, rest, n_boot=2000):
    obs = d[flag_idx].mean() - rest.mean()
    boots = np.array([rng.choice(rest, size=len(flag_idx), replace=True).mean() for _ in range(n_boot)]) - rest.mean()
    return obs, float(obs / boots.std()) if boots.std() > 0 else float('nan')
