# QA-A-259 real-set run (once): the whole raw dark list Dark_07..14 (first frame included), default rule. Peak working set via polling PeakWorkingSet64 (MiB).
$exe = 'D:\workspace-github\xpe-pre\build\ci-preprocess\bin\xpe_calib_fixture_gen.exe'
$ev = 'D:\workspace-github\xpe-pre\.moai\reports\lane-pre\QA-A-259\evidence'
$out = 'D:\workspace-github\xpe-data\bpm-eval-259\set_default'
$a = @('--out', $out, '--expiry-ms', '4102444800000', '--dark-list', "$ev\50_darks.txt", '--bright-list', "$ev\51_gain_brights.txt", '--bpm-bright-list', "$ev\52_bpm_brights.txt", '--bpm-v2', '--bpm-margins', '6,2,1,1')
