# QA-A-318 comparison of PRE outputs vs the existing-engine files (scratch; saved as evidence 20_compare.py.txt)
import hashlib, os, sys
import numpy as np
B = r"Z:/공통자료/Solution팀 자료/소프트웨어 자료/자체 개발/FT툴 관련/202550526/20250501_12"
I, C = B + "/images", B + "/corrected"
T = "D:/workspace-github/xpe-data/qa-318-tmp"
W, H = 3072, 2560
L, TP, R, BT = 15, 2, 17, 92           # CalEng.ini cuts
CW, CH = W - L - R, H - TP - BT        # 3040 x 2466
out = []
P = lambda *a: out.append(" ".join(str(x) for x in a))
def u16(p): return np.fromfile(p, dtype="<u2").reshape(H, W)
def s16(p): return np.fromfile(p, dtype="<i2").reshape(H, W)
def sha(p):
    h = hashlib.sha256(); h.update(open(p, "rb").read()); return h.hexdigest()
def xcal(p):
    raw = open(p, "rb").read()
    import struct
    magic, ver, typ, fmt, w, h = struct.unpack_from("<4s5I", raw, 0)
    cj, pl = struct.unpack_from("<QQ", raw, 104)
    payload = raw[152 + cj: 152 + cj + pl]
    P(f"  xcal {os.path.basename(p)} magic={magic} type={typ} fmt={fmt} {w}x{h} cfg='{raw[152:152+cj].decode()}' payload={pl} total={len(raw)} sha256={sha(p)}")
    assert fmt == 1 and w == W and h == H and len(raw) == 152 + cj + pl
    return np.frombuffer(payload, dtype="<f4").reshape(H, W).astype(np.float64)
def q(a, ps=(0, 1, 50, 99, 100)): return " ".join(f"p{p}={np.percentile(a, p):.6g}" for p in ps)

md = u16(f"{I}/MasterDark.raw").astype(np.float64)
eng = u16(f"{C}/TOTAL_BPM.raw") != 0
area = np.zeros((H, W), bool); area[TP:H - BT, L:W - R] = True
good = ~eng

# ---------- offset
P("## offset")
darks = np.stack([u16(f"{I}/dark03_{i:02d}.raw").astype(np.float64) for i in range(10)])
for tag, f, ref in (("O-a all10", "offset_all10.xcal", darks.mean(0)), ("O-b skip1", "offset_skip1.xcal", darks[1:].mean(0))):
    v = xcal(f"{T}/{f}")
    d = v - md
    P(f"{tag}: floor(v)==MasterDark {100*np.mean(np.floor(v)==md):.4f} % | rint(v)==MasterDark {100*np.mean(np.rint(v)==md):.4f} % |"
      f" v-MasterDark: mean {d.mean():.4f} max|.| {np.abs(d).max():.4f} {q(d)} | control max|v - numpy float64 mean| {np.abs(v-ref).max():.3e}")
    P(f"{tag}: |v-MasterDark| >= 1: {int(np.count_nonzero(np.abs(d) >= 1))} px; inside cut & not engine-defect max|d| {np.abs(d[area & good]).max():.4f}")
del darks

# ---------- gain
P("## gain")
mb = s16(f"{I}/MasterBright.raw").astype(np.float64)
gc = mb / mb.mean()
ga = xcal(f"{T}/gain_a.xcal"); gb = xcal(f"{T}/gain_b.xcal")
P(f"control G-a vs G-b max|diff| {np.abs(ga-gb).max():.3e} identical_bytes_payload {bool(np.array_equal(ga, gb))}")
# control: numpy recomputation of the PRE definition
acc = np.zeros((H, W));
for i in range(10): acc += np.maximum(u16(f"{I}/Bright02_{i:02d}.raw").astype(np.float64) - md, 0)
gn = (acc / 10); gn = gn / gn.mean()
P(f"control G-a vs numpy mean(max(0,raw-MasterDark))/mean max|diff| {np.abs(ga-gn).max():.3e}")
P(f"MasterBright int16 mean {mb.mean():.4f}, negatives {int(np.count_nonzero(mb<0))}")
for nm, m in (("all", np.ones((H, W), bool)), ("not engine defect", good), ("inside cut & not engine defect", area & good)):
    d = ga[m] - gc[m]
    r = np.corrcoef(ga[m], gc[m])[0, 1]
    P(f"G-a - G_c [{nm}] n={int(m.sum())}: median {np.median(d):.3e} p1 {np.percentile(d,1):.3e} p99 {np.percentile(d,99):.3e} max|.| {np.abs(d).max():.4f} pearson {r:.8f}")
# where the big differences are
d = np.abs(ga - gc); top = np.argsort(d.ravel())[-5:]
P("largest |G-a - G_c| pixels (row,col,G-a,G_c,engineDefect):", [(int(t//W), int(t%W), round(float(ga.flat[t]),4), round(float(gc.flat[t]),4), bool(eng.flat[t])) for t in top])

# ---------- defect map v2
P("## defect map v2")
withF = u16(f"{C}/BPM_with_Filter.raw") != 0; woF = u16(f"{C}/BPM_without_Filter.raw") != 0
def lines(m, msk):
    rows = [int(r) for r in np.where((m & msk).sum(1) > 0.5 * msk.sum(1).clip(1))[0] if msk[r].any()]
    cols = [int(c) for c in np.where((m & msk).sum(0) > 0.5 * msk.sum(0).clip(1))[0] if msk[:, c].any()]
    return rows, cols
for tag in ("v_a", "v_b", "v_c"):
    p = f"{T}/{tag}.map.u8"
    m = np.fromfile(p, dtype="u1").reshape(H, W)
    vals, cnt = np.unique(m, return_counts=True)
    feat = np.fromfile(f"{T}/{tag}.feat.u8", dtype="u1").reshape(H, W)
    P(f"{tag}: sha256 {sha(p)} | codes {dict(zip(vals.tolist(), cnt.tolist()))} | feat sha256 {sha(f'{T}/{tag}.feat.u8')}")
    pre = m != 0
    for rn, msk in (("whole frame", np.ones((H, W), bool)), ("inside cut", area)):
        for en, e in (("TOTAL_BPM", eng), ("BPM_with_Filter", withF), ("BPM_without_Filter", woF)):
            A, Bm = pre & msk, e & msk
            P(f"  {tag} [{rn}] vs {en}: PRE {int(A.sum())} engine {int(Bm.sum())} both {int((A&Bm).sum())} PRE_only {int((A&~Bm).sum())} engine_only {int((~A&Bm).sum())}")
        rows, cols = lines(pre, msk)
        P(f"  {tag} [{rn}] full lines (>50% of the line flagged): rows n={len(rows)} {rows[:15]} cols n={len(cols)} {cols[:15]}")
    # which feature flagged the engine-only and the common pixels
    for en, sel in (("engine-only(TOTAL_BPM)", eng & ~pre), ("common", eng & pre), ("PRE-only", pre & ~eng)):
        bits = feat[sel]
        P(f"  {tag} {en}: n={int(sel.sum())} feature-bit counts " + str({b: int(np.count_nonzero(bits & (1 << b))) for b in range(6)}))
er, ec = lines(eng, np.ones((H, W), bool)); P(f"engine TOTAL_BPM full lines whole frame rows {er} cols {ec}; inside cut {lines(eng, area)}")

# ---------- correction (Oc+Gc) vs engine OcGc / OcGcDc
P("## correction")
gm = good[TP:H - BT, L:W - R]
OFF = xcal(f"{T}/offset_all10.xcal")
for i in (1, 3, 5, 7, 9, 11, 13, 15):
    f = f"{T}/corr/bright03_{i:02d}.raw.ocgc.f32"
    pre = np.fromfile(f, dtype="<f4").reshape(H, W).astype(np.float64)
    oc_u16 = np.fromfile(f"{T}/corr/bright03_{i:02d}.raw.oc.u16", dtype="<u2").reshape(H, W)
    raw = u16(f"{I}/bright03_{i:02d}.raw").astype(np.float64)
    off32 = OFF.astype(np.float32); ocm = np.floor(np.clip(raw.astype(np.float32) - off32, 0, 65535) + np.float32(0.5))  # offset_correct.cpp:92-97
    P(f"bright03_{i:02d}: PRE oc control: oc_u16 == floor(clip(raw-PREoffset,0,65535)+0.5) (float32) {100*np.mean(oc_u16==ocm):.4f} % | oc_u16 == raw-MasterDark (int, >=0) {100*np.mean(oc_u16==np.clip(raw-md,0,None)):.4f} % | oc sha {sha(f'{T}/corr/bright03_{i:02d}.raw.oc.u16')[:16]} ocgc sha {sha(f)[:16]}")
    for kind in ("OcGc", "OcGcDc"):
        ep = f"{C}/Bright03_{i:02d}_{kind}.raw"
        if not os.path.exists(ep): P(f"  {kind}: file absent"); continue
        e = np.fromfile(ep, dtype="<u2").reshape(CH, CW).astype(np.float64)
        es = np.fromfile(ep, dtype="<i2").reshape(CH, CW)
        # alignment check: best shift within +-3 by correlation on good pixels
        best = None
        for dy in range(-3, 4):
            for dx in range(-3, 4):
                y0, x0 = TP + dy, L + dx
                if y0 < 0 or x0 < 0 or y0 + CH > H or x0 + CW > W: continue
                pc = pre[y0:y0 + CH, x0:x0 + CW]
                r = np.corrcoef(pc[gm][::7], e[gm][::7])[0, 1]
                if best is None or r > best[0]: best = (r, dy, dx)
        pc = pre[TP:H - BT, L:W - R]
        x, y = pc[gm], e[gm]
        r = np.corrcoef(x, y)[0, 1]
        k, b0 = np.polyfit(x, y, 1)
        res = y - (k * x + b0)
        ratio = y[x > 0] / x[x > 0]
        P(f"  {kind}: eng uint16 mean {e.mean():.2f} min {e.min():.0f} max {e.max():.0f} int16-neg {int(np.count_nonzero(es<0))} | PRE crop mean {pc.mean():.2f}"
          f" | best shift (r,dy,dx)=({best[0]:.6f},{best[1]},{best[2]}) | at (0,0): pearson {r:.6f} fit eng={k:.6f}*PRE+{b0:.3f} residual p1 {np.percentile(res,1):.2f} p99 {np.percentile(res,99):.2f} std {res.std():.3f}"
          f" | eng/PRE {q(ratio,(1,50,99))} | n={int(gm.sum())}")
open(sys.argv[1], "w", encoding="utf-8").write("\n".join(out) + "\n")
print("done", len(out))
