import hashlib, os, sys, json
import numpy as np
B = r"Z:/공통자료/Solution팀 자료/소프트웨어 자료/자체 개발/FT툴 관련/202550526/20250501_12"
I, C, D20, D6 = B + "/images", B + "/corrected", B + "/12_CalData_20", B + "/12_CalData_6"
W, H = 3072, 2560
def u16(p): return np.fromfile(p, dtype="<u2").reshape(H, W)
def s16(p): return np.fromfile(p, dtype="<i2").reshape(H, W)
def u8(p): return np.fromfile(p, dtype="u1").reshape(H, W)
def pct(a): return f"{100.0*np.count_nonzero(a)/a.size:.4f} %"
def sha(p):
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for b in iter(lambda: f.read(1 << 22), b""): h.update(b)
    return h.hexdigest()
out = []
P = lambda *a: out.append(" ".join(str(x) for x in a))

for name, fr in (("all10", darks), ("skip_first_9", darks[1:])):
    m = np.mean(fr, axis=0)
    P(f"A MasterDark == floor(mean(dark03 {name})):", pct(np.floor(m) == md), "| == rint:", pct(np.rint(m) == md),
      "| max|MasterDark-mean|:", f"{np.max(np.abs(md - m)):.4f}")
P("A dark03 frame means:", [round(float(d.mean()), 3) for d in darks])
P("A MasterDark mean:", round(float(md.mean()), 3), "min", int(md.min()), "max", int(md.max()))

# --- B. __OC__ = raw - MasterDark ?
def oc_check(raw_p, oc_p, tag):
    r = u16(raw_p).astype(np.int32); o_u = u16(oc_p).astype(np.int32); o_s = s16(oc_p).astype(np.int32)
    d = r - md.astype(np.int32)
    P(f"B {tag}: oc_int16 == raw-MD: {pct(o_s == d)} | oc_uint16 == max(0,raw-MD): {pct(o_u == np.maximum(d,0))}"
      f" | oc_uint16 == raw-MD wrap16: {pct(o_u == (d & 0xFFFF))} | raw-MD<0 count {int(np.count_nonzero(d<0))}"
      f" | oc int16 neg count {int(np.count_nonzero(o_s<0))} | oc mean(int16) {o_s.mean():.2f}")
for i in range(10): oc_check(f"{I}/Bright02_{i:02d}.raw", f"{I}/Bright02_{i:02d}.raw__OC__.raw", f"Bright02_{i:02d}")

for i in range(20): oc_check(f"{I}/bright03_{i:02d}.raw", f"{I}/Bright03_{i:02d}.raw__OC__.raw", f"bright03_{i:02d}(lower raw) vs Bright03_{i:02d}__OC__")
for i in range(20): oc_check(f"{I}/bright05_{i:02d}.raw", f"{I}/Bright05_{i:02d}.raw__OC__.raw", f"bright05_{i:02d}(lower raw) vs Bright05_{i:02d}__OC__")
