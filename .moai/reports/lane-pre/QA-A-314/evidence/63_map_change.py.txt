ld=lambda p: np.fromfile(p,np.uint8).reshape(S,S)!=0
a=ld('D:/workspace-github/xpe-data/bpm-eval-260/inj_v2_setmap.u8'); b=ld('D:/workspace-github/xpe-data/qa-314-tmp/out/inj_v2_setmap.u8')
