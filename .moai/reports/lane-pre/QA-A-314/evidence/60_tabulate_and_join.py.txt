def rows(text):
    out, order = {}, []
    for line in text.splitlines():
        if not line.startswith('|') or line.startswith('|---') or line.startswith('| 행'): continue
        c = [x.strip() for x in line.strip().strip('|').split(' | ')]
        out[c[0]] = c; order.append(c[0])
    return out, order
