base = 'D:/workspace-github/xpe-pre/modules/preprocess/'


def edit(rel, pairs):
    p = base + rel
    s = open(p, encoding='utf-8', newline='').read()
    for old, new in pairs:
        assert s.count(old) == 1, (rel, s.count(old), old[:80])
        s = s.replace(old, new)
    open(p, 'w', encoding='utf-8', newline='').write(s)
