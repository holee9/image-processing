import hashlib, os, subprocess, time

ROOT = 'D:/workspace-github/xpe-pre/'
V2 = ROOT + 'modules/preprocess/src/xpe_bpm_v2.cpp'
EV = ROOT + 'modules/preprocess/tools/xpe_bpm_eval.cpp'
LOG = ROOT + '.moai/reports/lane-pre/QA-A-252e/evidence/21_mutation_log.txt'
BAT = 'C:/Users/drake/AppData/Local/Temp/claude/D--workspace-github-xpe-pre/c51d57cf-b9fc-4933-bc4a-9af9ebf82810/scratchpad/build237.bat'
EXE = ROOT + 'build/ci-preprocess/bin/xpe_preprocess_tests.exe'
FILT = 'BpmEval*'
BS = chr(92)


def say(t):
    print(t)
    open(LOG, 'a', encoding='utf-8').write(t + chr(10))


def build():
    r1 = subprocess.run(['cmd.exe', '/c', BAT.replace('/', BS), 'xpe_bpm_eval'], capture_output=True, text=True, encoding='cp949', errors='replace')
    r = subprocess.run(['cmd.exe', '/c', BAT.replace('/', BS), 'xpe_preprocess_tests'], capture_output=True, text=True, encoding='cp949', errors='replace')
    l1 = (r1.stdout.strip().splitlines() or ['?'])[-1]
    l2 = (r.stdout.strip().splitlines() or ['?'])[-1]
    return 'BUILD_EXIT=0' if l1 == 'BUILD_EXIT=0' and l2 == 'BUILD_EXIT=0' else l1 + ' / ' + l2


def run_tests():
    r = subprocess.run([EXE, '--gtest_filter=' + FILT], capture_output=True, text=True, encoding='utf-8', errors='replace', cwd=ROOT)
    ok = sorted({l.split(']')[1].split('(')[0].strip().split('.')[1] for l in r.stdout.splitlines() if l.startswith('[       OK ]')})
    bad = sorted({l.split(']')[1].split(',')[0].split('(')[0].strip().split('.')[1] for l in r.stdout.splitlines() if l.startswith('[  FAILED  ]') and '.' in l})
    return r.returncode, ok, bad


orig = {V2: open(V2, 'rb').read(), EV: open(EV, 'rb').read()}
h0 = hashlib.sha256(orig[V2] + orig[EV]).hexdigest()[:16]
open(LOG, 'w', encoding='utf-8').write('')
say('QA-A-252e mutation check of the level pair (tools/xpe_bpm_eval.cpp). Source hash at the start: ' + h0)
say('PREDICTIONS (written before any run):')
say('  U1 the pair becomes 0.45 and 0.46 (the defect level moves onto the boundary) -> red: ResponseItemsAreDefectsBelow45Percent... and the counts test (the 0.44 items are missing / become controls)')
say('  U2 the control level becomes 0.44 as well (no 0.46)                           -> red: ResponseItemsAreDefectsBelow45Percent... and the control group test (control/efficiency_loss/pixel/0.46 missing)')
say('  U3 the boundary comparison <= (a response of exactly 0.45 would become a defect) -> GREEN: no item sits at 0.45 any more (a test gap by design, reported)')
say('  every other test stays green in each mutant')

mutants = [
    ('U1', EV, b'{"gain_loss", {0.3, 0.44, 0.46, 0.7}}, {"efficiency_loss", {0.3, 0.44, 0.46, 0.7}}};', b'{"gain_loss", {0.3, 0.45, 0.46, 0.7}}, {"efficiency_loss", {0.3, 0.45, 0.46, 0.7}}};'),
    ('U2', EV, b'{"gain_loss", {0.3, 0.44, 0.46, 0.7}}, {"efficiency_loss", {0.3, 0.44, 0.46, 0.7}}};', b'{"gain_loss", {0.3, 0.44, 0.44, 0.7}}, {"efficiency_loss", {0.3, 0.44, 0.44, 0.7}}};'),
    ('U3', EV, b'r < kResponseDefectBelow - 1e-9', b'r <= kResponseDefectBelow'),
]
try:
    say('--- step 0: unmutated')
    say('build: ' + build())
    rc, ok, bad = run_tests()
    say('tests exit %d: %d passed, failed: %s' % (rc, len(ok), bad))
    for name, path, old, new in mutants:
        cur = orig[path]
        assert cur.count(old) == 1, (name, cur.count(old))
        say('--- ' + name)
        open(path, 'wb').write(cur.replace(old, new))
        b = build()
        say('build: ' + b)
        if b != 'BUILD_EXIT=0':
            say('mutant did not compile; tests NOT run (an old binary would give a false green)')
        else:
            rc, ok, bad = run_tests()
            say('tests exit %d: %d passed, FAILED: %s' % (rc, len(ok), bad))
        open(path, 'wb').write(cur)
        os.utime(path, None)
        time.sleep(1)
finally:
    for path, data in orig.items():
        open(path, 'wb').write(data)
        os.utime(path, None)
say('--- restored: touch + rebuild')
say('source hash now: %s (start: %s)' % (hashlib.sha256(open(V2, 'rb').read() + open(EV, 'rb').read()).hexdigest()[:16], h0))
say('build: ' + build())
rc, ok, bad = run_tests()
say('tests exit %d: %d passed, failed: %s' % (rc, len(ok), bad))
