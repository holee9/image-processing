"""QA-A-252e: split the new false alarms that are not on a control item ("others") pixel by pixel.

usage: python -I 70_decompose.py <label> <truth_dir> <planted_map.u8> <clean_map.u8> [<planted_lines.u8> [<features.u8>]] [--tsv <out.tsv>]

A pixel is a NEW false alarm when the planted map flags it, the clean map does not, and it lies outside every planted (non-control) item grown by 2 pixels
(the definition of the score tool; the totals printed here are checked against it by the caller). The ones that sit on a control item are counted apart.
Sources: 'pixel' = the pixel detector map, 'line' = flagged only through the line mask. For the pixel part the feature bits at the pixel are listed.
"""
import sys
import numpy as np

W = 3072
FEAT = [(1, 'dark_mean'), (2, 'dark_noise'), (4, 'dose_slope'), (8, 'dose_linear'), (16, 'flat'), (32, 'bright_noise')]
args = sys.argv[1:]
tsv = None
if '--tsv' in args:
    i = args.index('--tsv')
    tsv = args[i + 1]
    del args[i:i + 2]
label, truth_dir, planted, clean = args[0], args[1], args[2], args[3]
lines = args[4] if len(args) > 4 and args[4] else None
feats = args[5] if len(args) > 5 and args[5] else None


def u8(p):
    return np.fromfile(p, dtype=np.uint8).reshape(W, W)


pix = u8(planted) > 0
cln = u8(clean) > 0
lin = (u8(lines) > 0) if lines else np.zeros((W, W), bool)
fb = u8(feats) if feats else np.zeros((W, W), np.uint8)
mp = pix | lin

rects = []   # (id, class, kind, shape, x, y, w, h)
real = np.zeros((W, W), bool)
ctrl = np.zeros((W, W), bool)
for ln in open(truth_dir + '/truth.tsv', encoding='utf-8'):
    p = ln.rstrip('\n').split('\t')
    if len(p) >= 6 and p[0].isdigit():
        for r in p[5].split(';'):
            x, y, w, h = map(int, r.split(','))
            rects.append((int(p[0]), p[1], p[2], p[3], x, y, w, h))
            if p[1] == 'control':
                ctrl[y:y + h, x:x + w] = True
            else:
                real[y:y + h, x:x + w] = True
pad = np.pad(real, 2)
wide = np.zeros_like(real)
for dy in range(5):
    for dx in range(5):
        wide |= pad[dy:dy + W, dx:dx + W]
new = mp & ~cln & ~wide
# the score tool attributes a pixel to an item through the id map (truth_ids.u16), not through the rectangles: use the same attribution
ids = np.fromfile(truth_dir + '/truth_ids.u16', dtype=np.uint16).reshape(W, W)
is_ctrl_id = np.zeros(65536, bool)
for r in rects:
    if r[1] == 'control':
        is_ctrl_id[r[0]] = True
ctrl_by_id = is_ctrl_id[ids]
on_ctrl = new & ctrl_by_id
other = new & ~ctrl_by_id
print('==', label)
print('new false alarms', int(new.sum()), ' on control items', int(on_ctrl.sum()), ' others', int(other.sum()))
print('others by source: pixel detector map', int((other & pix).sum()), ' line mask only', int((other & lin & ~pix).sum()))

ys, xs = np.nonzero(other)
R = np.array([(r[4], r[5], r[4] + r[6] - 1, r[5] + r[7] - 1) for r in rects]) if rects else np.zeros((0, 4), int)
rows = []
for y, x in zip(ys, xs):
    dx = np.maximum(np.maximum(R[:, 0] - x, x - R[:, 2]), 0)
    dy = np.maximum(np.maximum(R[:, 1] - y, y - R[:, 3]), 0)
    d = np.maximum(dx, dy)
    k = int(np.argmin(d))
    names = '+'.join(n for b, n in FEAT if fb[y, x] & b) or '-'
    rows.append((int(x), int(y), 'pixel' if pix[y, x] else 'line', names, rects[k][0], rects[k][1], rects[k][2], rects[k][3], int(d[k])))

# groups of 8-connected pixels
seen = set()
pts = {(r[0], r[1]): r for r in rows}
groups = []
for key in pts:
    if key in seen:
        continue
    stack = [key]
    seen.add(key)
    g = []
    while stack:
        cx, cy = stack.pop()
        g.append(pts[(cx, cy)])
        for ddx in (-1, 0, 1):
            for ddy in (-1, 0, 1):
                nk = (cx + ddx, cy + ddy)
                if nk in pts and nk not in seen:
                    seen.add(nk)
                    stack.append(nk)
    groups.append(g)
print('groups (8-connected):', len(groups), ' sizes of the pixel-detector groups:',
      sorted(len(g) for g in groups if g[0][2] == 'pixel'))
by = {}
for r in rows:
    k = (r[2], r[3])
    by[k] = by.get(k, 0) + 1
for k in sorted(by):
    print('  source', k[0], 'features', k[1], ':', by[k])
dist = {}
for r in rows:
    b = 'within 3..8 px of a planted item' if r[8] <= 8 else ('9..40 px' if r[8] <= 40 else 'farther')
    key = (r[2], b)
    dist[key] = dist.get(key, 0) + 1
for k in sorted(dist):
    print('  source', k[0], 'distance to the nearest planted item:', k[1], ':', dist[k])
nk = {}
for r in rows:
    key = (r[2], r[5] + '/' + r[6])
    nk[key] = nk.get(key, 0) + 1
for k in sorted(nk, key=lambda t: -nk[t])[:12]:
    print('  nearest item class/kind', k[1], 'source', k[0], ':', nk[k])
if tsv:
    with open(tsv, 'w', encoding='utf-8', newline='\n') as f:
        f.write('x\ty\tsource\tfeatures\tnearest_item\tclass\tkind\tdistance\n')
        for r in sorted(rows, key=lambda t: (t[1], t[0])):
            f.write('%d\t%d\t%s\t%s\t%d\t%s\t%s\t%d\n' % (r[0], r[1], r[2], r[3], r[4], r[5], r[6], r[8]))
