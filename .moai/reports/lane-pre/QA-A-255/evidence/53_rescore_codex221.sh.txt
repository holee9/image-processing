#!/bin/bash
# QA-A-255 0-8 (Codex #221): noise-cost once more (the overlap measurement now carries response_used / response_valid / the unused reasons; the item truth must stay identical to QA-A-252g's) and the two
# maps scored once more with the masks at the overlaps ASSIGNED from the final class. Nothing is detected again.
set -u
ROOT=D:/workspace-github/xpe-pre
BIN=$ROOT/build/ci-preprocess/bin
OLD=D:/workspace-github/xpe-data/bpm-eval-251
IN=D:/workspace-github/xpe-data/bpm-eval-252f
G1=D:/workspace-github/xpe-data/bpm-eval-252g
G2=D:/workspace-github/xpe-data/bpm-eval-252h
INJ=$IN/inj_d/frames
EVD=$ROOT/.moai/reports/lane-pre/QA-A-255/evidence
NT=$ROOT/.moai/reports/lane-pre/QA-A-252g/evidence/62_noise_truth.tsv.txt
run() { local name=$1; shift; "$@" > "$EVD/$name.out.txt" 2>&1; local rc=$?; echo "$name exit $rc"; return $rc; }
NC=(); for k in 08 09 10 11 12 13 14; do NC+=(--dark "$INJ/Dark_$k.raw"); done
run N2_noise_cost "$BIN/xpe_bpm_eval.exe" noise-cost --truth "$IN/inj_d" "${NC[@]}" --verify "$INJ/Bright_29.raw" --verify "$INJ/Bright_30.raw" --out "$EVD/70_noise_truth_recomputed.tsv.txt" --overlap-out "$EVD/71_overlaps_measured.tsv.txt" || exit 1
cmp "$EVD/70_noise_truth_recomputed.tsv.txt" "$NT" && echo "item truth identical to 252g"
S1=(score --truth "$IN/inj_d" --noise-truth "$NT" --map "$G1/inj_set/defect.xcal" --clean-map "$OLD/clean_set/defect.xcal" --half-a "$G1/v1_half_a.u8" --half-b "$G1/v1_half_b.u8")
S2=(score --truth "$IN/inj_d" --noise-truth "$NT" --map "$G2/v3g_planted.u8" --clean-map "$G2/v3g_clean.u8" --half-a "$G2/v3g_half_a.u8" --half-b "$G2/v3g_half_b.u8" --area "$G2/v2g_planted_area.u8")
run R3_gen1 "$BIN/xpe_bpm_eval.exe" "${S1[@]}" --overlap composite --overlap-truth "$EVD/71_overlaps_measured.tsv.txt" --overlap-out "$EVD/72_gen1_overlaps.tsv.txt" --out "$EVD/73_gen1_rescored.json.txt" || exit 1
run R4_gen2 "$BIN/xpe_bpm_eval.exe" "${S2[@]}" --overlap composite --overlap-truth "$EVD/71_overlaps_measured.tsv.txt" --overlap-out "$EVD/74_gen2_overlaps.tsv.txt" --out "$EVD/75_gen2_rescored.json.txt" || exit 1
echo done
