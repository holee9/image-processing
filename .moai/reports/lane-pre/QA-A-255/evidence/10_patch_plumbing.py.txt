"""QA-A-255 step 1: the layers of a version-2 defect file (imaging area, lines with their individual stretches, blob marks) are kept in the runtime store, in the snapshot a frame takes, and in the
calibration cache entry, as ONE immutable shared object (`DefectLayers`). A version-1 file leaves the pointer empty, which is what makes the version-1 path the old path: the correction function
takes the new route only when the snapshot carries layers.
 - defect_ext.hpp/.cpp: struct DefectLayers, make_defect_layers
 - xpe_preprocess_internal.h: forward declaration, members of the store, the snapshot and StagedDefect
 - xpe_calibration.cpp: the snapshot copies it
 - xpe_calib_load_defect_map.cpp: the stage builds it from the version-2 parts; the commit installs it (null for a version-1 file, so loading a version-1 file after a version-2 file clears it)
 - calibration_cache.cpp: the entry keeps it, a hit installs it."""
base = 'D:/workspace-github/xpe-pre/modules/preprocess/'


def edit(rel, pairs):
    s = open(base + rel, encoding='utf-8', newline='').read()
    crlf = '\r\n' in s
    s = s.replace('\r\n', '\n')
    for old, new in pairs:
        assert s.count(old) == 1, (rel, s.count(old), old[:110])
        s = s.replace(old, new)
    open(base + rel, 'w', encoding='utf-8', newline='').write(s.replace('\n', '\r\n') if crlf else s)
