Differences from the old runners (05_mutation_runner.py.txt of 251b, 20_mutation_common.py.txt / 25_mutation_tool.py.txt of 252f), all of which touched no consumer of a mutated header:
  * the two consumers of the header (the object of the tests' test_bpm_eval.cpp and of the tool's xpe_bpm_eval.cpp) are touched before every build of a mutant;
  * for every mutant the runner records which consumer objects are newer than the moment the mutant was written, the sha256 of the tests executable and of the tool executable against the baseline's,
    and the executables' modification times; a mutant whose consumers were not all rebuilt, or whose executable does not differ from the baseline, is INVALID and is not counted;

    are repeated, and the run ends with an error if they are not as at the start;
  * step P (before any mutant, announced in PRED): one header mutant is built WITHOUT touching the consumers, to learn whether the old runners' header mutants were compiled in at all.

def say(t):
    print(t, flush=True)
    open(LOG, 'a', encoding='utf-8').write(t + chr(10))


def sha(path):
    return hashlib.sha256(open(path, 'rb').read()).hexdigest()[:16]


def build(touch):
    for f in touch:
        os.utime(f, None)
    r = subprocess.run(['cmd.exe', '/c', BAT.replace('/', BS), 'xpe_preprocess_tests'], capture_output=True, text=True, encoding='cp949', errors='replace')
    return (r.stdout.strip().splitlines() or ['?'])[-1]


def run_tests():
    r = subprocess.run([EXE_TEST, '--gtest_filter=' + FILT], capture_output=True, text=True, encoding='utf-8', errors='replace', cwd=ROOT)
    ok = sorted({l.split(']')[1].split('(')[0].strip().split('.')[1] for l in r.stdout.splitlines() if l.startswith('[       OK ]')})
    bad = sorted({l.split(']')[1].split(',')[0].split('(')[0].strip().split('.')[1] for l in r.stdout.splitlines() if l.startswith('[  FAILED  ]') and '.' in l})
    return r.returncode, ok, bad
