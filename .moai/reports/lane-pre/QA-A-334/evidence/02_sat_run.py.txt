# QA-A-334 item 3: run the shipped pipeline (xpe_preprocess_pipeline, calibPath = cyan_test clean set) on Bright_34 twice:
# (a) {"saturationLevel":"off"}  (b) {"saturationLevel":40000}  (cyan_test has no raw 65535 pixel; 40000 is below the frame mean 40499).
# Reported: S_out (value written to raw >= 40000 pixels), 40000 / median(gain stage ratio), and the (a) outputs of the same pixels / of raw 39990..39999 pixels.
import ctypes as C, numpy as np, os, sys
os.add_dll_directory(sys.argv[1]); dll = C.CDLL(sys.argv[1] + "/xpe_preprocess.dll")
class Buf(C.Structure): _fields_ = [("width", C.c_uint32), ("height", C.c_uint32), ("bitsAllocated", C.c_uint32), ("bitsStored", C.c_uint32), ("format", C.c_int), ("data", C.c_void_p), ("dataSize", C.c_size_t)]
class Meta(C.Structure): _fields_ = [("bodyPart", C.c_char * 64), ("kVp", C.c_float), ("mAs", C.c_float), ("SID_mm", C.c_float), ("pixelPitch_mm", C.c_float), ("acquisitionTime", C.c_uint64), ("flags", C.c_uint32)]
dll.xpe_preprocess_pipeline.argtypes = [C.POINTER(Buf), C.POINTER(Meta), C.c_char_p, C.c_void_p, C.c_char_p]
dll.xpe_preprocess_init.argtypes = [C.c_char_p]; print("init rc", dll.xpe_preprocess_init(None))
raw = np.fromfile(sys.argv[3], dtype=np.uint16)
def run(cfg):
    store = np.zeros(raw.size * 2, dtype=np.uint16); store[:raw.size] = raw   # room for float32 result
    b = Buf(3072, 3072, 16, 16, 0, store.ctypes.data, store.nbytes); m = Meta()
    rc = dll.xpe_preprocess_pipeline(C.byref(b), C.byref(m), sys.argv[2].encode(), None, cfg.encode())
    out = store.view(np.float32)[:raw.size].copy() if b.format != 0 else store[:raw.size].astype(np.float32)
    return rc, b.format, out
rcA, fA, A = run('{"saturationLevel":"off"}'); rcB, fB, B = run('{"saturationLevel":40000}')
print("rc off", rcA, "format", fA, "| rc 40000", rcB, "format", fB)
sat = raw >= 40000; near = (raw >= 39990) & (raw < 40000)
print("pixels raw>=40000", int(sat.sum()))
print("run(b) values on raw>=40000: unique count", len(np.unique(B[sat])), "first", B[sat][:3])
print("run(a) on the same pixels: median %.4f min %.4f max %.4f" % (np.median(A[sat]), A[sat].min(), A[sat].max()))
print("run(a) on raw 39990..39999 (n=%d): median %.4f max %.4f" % (int(near.sum()), np.median(A[near]), A[near].max()))
print("run(a) max over raw<40000 pixels: %.4f" % A[~sat].max())
print("run(b) max over raw<40000 pixels: %.4f; pixels changed by clamp: %d" % (B[~sat].max(), int(np.sum(B[~sat] != A[~sat]))))
# (c) no config at all (the default a caller gets): does saturation handling act?
store = np.zeros(raw.size * 2, dtype=np.uint16); store[:raw.size] = raw
b = Buf(3072, 3072, 16, 16, 0, store.ctypes.data, store.nbytes); m = Meta()
rcC = dll.xpe_preprocess_pipeline(C.byref(b), C.byref(m), sys.argv[2].encode(), None, None)
Cc = store.view(np.float32)[:raw.size]
print("rc default", rcC, "| default vs off: pixels different", int(np.sum(Cc != A)), "| default max %.4f" % Cc.max())
