import json, os, re
D = '.moai/reports/lane-pre/QA-A-274/'
res = json.load(open(D + 'collected.json'))
m = {r['sha']: r for r in res if r['sha']}
roots = ['D:/workspace-github/image-processing/.moai/state/reviews'] + [
    '.moai/reports/lane-pre/' + d for d in os.listdir('.moai/reports/lane-pre')
    if d.startswith(('PRE-', 'CODEX', 'INSTALLER', 'QA-A-27'))]
hits = {}
pat = re.compile(r'(?<![0-9a-f])[0-9a-f]{64}(?![0-9a-f])')
for root in roots:
    for dp, dn, fn in os.walk(root):
        for f in fn:
            if not f.lower().endswith(('.md', '.txt', '.json', '.log')):
                continue
            p = os.path.join(dp, f)
            try:
                if os.path.getsize(p) > 8_000_000:
                    continue
                t = open(p, encoding='utf-8', errors='replace').read().lower()
            except Exception:
                continue
            for h in set(pat.findall(t)):
                if h in m:
                    hits.setdefault(m[h]['path'], set()).add(p.replace(chr(92), '/').split('.moai/')[-1])
out = {k: sorted(v) for k, v in hits.items()}
json.dump(out, open(D + 'hashhits.json', 'w'), indent=1)
for r in res:
    pth = r['path']
    if not pth.startswith(('gui', 'modules', 'docs', 'cmake', 'tools')) or not r['sha']:
        continue
    if pth in out:
        print('HIT ', pth, '<-', [x.split('/')[-1] for x in out[pth]][:3])
    else:
        print('NONE', pth)
