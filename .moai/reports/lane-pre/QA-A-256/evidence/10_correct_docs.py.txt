"""QA-A-256 step 0 (Codex #224 non-blocking advice): the struct comment, the 255 report and the 255b report wording. Comment and prose only; no behaviour change."""
base = 'D:/workspace-github/xpe-pre/'


def edit(path, pairs, expect_each=1):
    s = open(base + path, encoding='utf-8', newline='').read()
    crlf = '\r\n' in s
    s = s.replace('\r\n', '\n')
    for old, new in pairs:
        assert s.count(old) == expect_each, (path, s.count(old), old[:80])
        s = s.replace(old, new)
    open(base + path, 'w', encoding='utf-8', newline='').write(s.replace('\n', '\r\n') if crlf else s)
