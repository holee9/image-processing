def det(s, group):
    for d in s['detection']:
        if d['group'] == group:
            return d['detected'], d['items']
    return None


def frac(p):
    return 'n/a' if p is None else '%d/%d = %.3f' % (p[0], p[1], p[0] / p[1] if p[1] else 0.0)


def rate(p):
    return None if p is None or p[1] == 0 else p[0] / p[1]


def groups(s, pred):
    return [d for d in s['detection'] if pred(d['group'])]


def min_rate(s, pred):
    gs = groups(s, pred)
    if not gs:
        return None, None, 0
    worst = min(gs, key=lambda d: d['detected'] / d['items'])
    return worst['detected'] / worst['items'], worst['group'], len(gs)
