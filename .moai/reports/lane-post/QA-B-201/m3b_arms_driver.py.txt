"""QA-B-201 M3b falsification arms: the output floor of xpe_edge_enhance. Break ONE defence, rebuild ci-post, run the
enhance_basic tests, expect the named tests red, restore byte-identical. Run: python build/g201_arms_m3b.py <out.txt>"""
import hashlib, re, subprocess, sys

OUT = sys.argv[1]
BUILD = r'D:\workspace-github\xpe-post\build\g200-post-full.bat'
RUN = r'D:\workspace-github\xpe-post\build\g200-run2.bat'
ED = 'modules/enhance_basic/src/edge_enhance.cpp'
NL = '\n'
RES = '            const float result = (abs_diff >= threshold) ? sharpened : orig;' + NL
FLOOR = '            row[x] = result < 0.0f ? 0.0f : result;' + NL
T_FLOOR = 'EdgeEnhanceFloor.ASharpenedPixelBesideADarkRegionIsNeverBelowZero'
T_NEG = 'EdgeEnhanceFloor.TheOutputHasNoNegativeEvenForANegativeInputThatIsNotSharpened'
T_POS = 'EdgeEnhanceFloor.APositivePixelThatIsNotSharpenedIsStillReturnedAsItCame'
T_BOUND = 'EdgeEnhanceFloor.TheUpperBoundIsUnchangedByTheFloor'

ARMS = [
    ('floor_removed', [(ED, FLOOR, '            row[x] = result;' + NL)], [T_FLOOR, T_NEG, T_POS, T_BOUND]),
    ('floor_only_on_the_sharpened_value_as_in_M3',
     [(ED, RES + FLOOR, '            row[x] = (abs_diff >= threshold) ? (sharpened < 0.0f ? 0.0f : sharpened) : orig;' + NL)], [T_NEG, T_POS]),
    ('floor_to_one', [(ED, FLOOR, '            row[x] = result < 0.0f ? 1.0f : result;' + NL)], [T_FLOOR, T_NEG, T_POS]),
    ('floor_to_the_original_value', [(ED, FLOOR, '            row[x] = result < 0.0f ? orig : result;' + NL)], [T_FLOOR, T_NEG]),
]


def sha(p):
    return hashlib.sha256(open(p, 'rb').read()).hexdigest()


def sh(cmd):
    r = subprocess.run(['cmd', '/c', cmd], capture_output=True, text=True, timeout=3000)
    return (r.stdout or '') + (r.stderr or '')


def failed(path):
    txt = open(path, encoding='utf-8', errors='replace').read()
    return set(re.findall(r'^\[  FAILED  \] (\S+?)(?: \(|$)', txt, re.M))


out = open(OUT, 'w', encoding='utf-8', newline='\n')


def log(s):
    out.write(s + '\n')
    out.flush()


for name, edits, must in ARMS:
    files = sorted({e[0] for e in edits})
    saved = {f: open(f, 'rb').read() for f in files}
    before = {f: sha(f) for f in files}
    log('=== ARM ' + name)
    try:
        for f in files:
            txt = saved[f].decode('utf-8')
            for ff, old, new in edits:
                if ff == f:
                    assert txt.count(old) == 1, (name, old[:60], txt.count(old))
                    txt = txt.replace(old, new)
            open(f, 'wb').write(txt.encode('utf-8'))
        sh(BUILD)
        if 'FAILED:' in open('build/g200-b5.txt', encoding='utf-8', errors='replace').read():
            log('RESULT: BUILD FAILED')
        else:
            sh('%s ci-post xpe_enhance_basic_tests.exe D:\\workspace-github\\xpe-post\\build\\g201-arm-m3b.txt * enhance_basic' % RUN)
            red = failed('build/g201-arm-m3b.txt')
            hit = [m for m in must if m in red]
            miss = [m for m in must if m not in hit]
            log('red tests: ' + (', '.join(sorted(red)) if red else '(none)'))
            log('RESULT: %s  (expected red: %d, red: %d, not red: %s)' % ('FIRED' if not miss else 'DID NOT FIRE', len(must), len(hit), miss))
    finally:
        for f in files:
            open(f, 'wb').write(saved[f])
        log('restored byte-identical: %s' % all(sha(f) == before[f] for f in files))
log('=== DONE')
out.close()
