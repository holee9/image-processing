"""QA-B-214b falsification arms: mutate voi_auto_window.cpp, build, run the auto-window tests, record which go red, restore, rebuild.

arm4  the representativeness check never fires (the sample is always trusted)      -> the off-grid large-image test must go red
arm5  a fallback decided on the sample is not confirmed on the whole image         -> the fallback-confirmed test must go red
arm6  the 214 code: small images sampled too, sample always trusted, fallback not confirmed -> the 8x8 Codex reproduction must go red
arm7  the float check of center/width is removed (both layers)                      -> the +-FLT_MAX tests must go red
"""
import hashlib, os, re, subprocess

ROOT = r'D:/workspace-github/xpe-post'
OUT = ROOT + '/.moai/reports/lane-post/QA-B-214b'
PATH = 'modules/display/src/voi_auto_window.cpp'
os.chdir(ROOT)


def sha(p):
    return hashlib.sha256(open(p, 'rb').read()).hexdigest()


def sh(cmd):
    return subprocess.run(cmd, shell=True, capture_output=True, text=True)


def build():
    sh('cmd.exe /c ".moai\\reports\\lane-post\\QA-B-214b\\_build.bat build\\g214b-arm-build.txt"')
    t = open('build/g214b-arm-build.txt', encoding='utf-8', errors='replace').read()
    m = re.findall(r'===BUILD=(\d+)===', t)
    return m[-1] if m else '?'


def run_tests(tag):
    summary = 'build/g214-%s-summary.txt' % tag
    if os.path.exists(summary):
        os.remove(summary)
    sh('cmd.exe /c ".moai\\reports\\lane-post\\QA-B-214\\_rundisp.bat %s"' % tag)
    t = open('build/g214-%s-test_display_voi_auto_window.txt' % tag, encoding='utf-8', errors='replace').read()
    failed = sorted({re.sub(r' \(\d+ ms\)$', '', l[len('[  FAILED  ] '):]) for l in t.splitlines()
                     if l.startswith('[  FAILED  ]') and l.rstrip().endswith('ms)')})
    passed = re.findall(r'\[  PASSED  \] (\d+) tests?', t)
    return failed, (passed[-1] if passed else '0')


TRUSTED = 'static_cast<double>(outside) <= kAutoWindowMaxUnsampledMass * static_cast<double>(count)'
NOTRUST = 'if (!trusted) stride = 1;'
CONFIRM = 'if (stride > 1 && a.fallbackWhy) {'
FULLPASS = 'constexpr size_t kAutoWindowFullPassPixelLimit = static_cast<size_t>(1) << 20;'
LOD = 'double lod   = (stride == 1) ? static_cast<double>(lo) : static_cast<double>(sampleLo);'
RANGE = 'double range = ((stride == 1) ? static_cast<double>(hi) : static_cast<double>(sampleHi)) - lod;'
CHECK1 = 'if (!(std::fabs(center) <= static_cast<double>(FLT_MAX)) || !(winSpan > 0.0) || !(winSpan <= static_cast<double>(FLT_MAX))) {'
CHECK2 = 'if (!xpe_float_is_finite(res.center) || !xpe_float_is_finite(res.width) || !(res.width > 0.0f)) {'

arms = [
    ('arm4_representativeness_never_fires', [(TRUSTED, 'static_cast<double>(outside) <= 2.0 * static_cast<double>(count)')]),
    ('arm5_sample_fallback_not_confirmed', [(CONFIRM, 'if (stride > 100 && a.fallbackWhy) {')]),
    ('arm6_the_214_code_small_images_sampled_nothing_confirmed', [
        (FULLPASS, 'constexpr size_t kAutoWindowFullPassPixelLimit = 0;'),
        (NOTRUST, 'if (stride > 100 && !trusted) stride = 1;'),
        (CONFIRM, 'if (stride > 100 && a.fallbackWhy) {'),
        (LOD, 'double lod   = static_cast<double>(lo);'),
        (RANGE, 'double range = static_cast<double>(hi) - lod;')]),
    ('arm7_float_check_removed', [(CHECK1, 'if (!(winSpan > 0.0)) {'), (CHECK2, 'if (res.width < 0.0f) {')]),
]
report = []
orig = open(PATH, 'rb').read()
h0 = sha(PATH)
text0 = orig.decode('utf-8')
for name, edits in arms:
    t = text0
    for old, new in edits:
        assert t.count(old) == 1, (name, old[:60], t.count(old))
        t = t.replace(old, new)
    open(PATH, 'wb').write(t.encode('utf-8'))
    try:
        rc = build()
        failed, passed = run_tests('arm214b') if rc == '0' else (['BUILD FAILED rc=' + rc], '0')
    finally:
        open(PATH, 'wb').write(orig)
    report.append('== %s\n   mutations:\n%s   build rc=%s\n   auto-window tests red (%d): %s\n   file restored: sha256 identical=%s\n' % (
        name, ''.join('     - %s\n     + %s\n' % (o, n) for o, n in edits), rc, len(failed), '; '.join(failed), h0 == sha(PATH)))
rc = build()
failed, passed = run_tests('restored214b')
report.append('== restored tree: build rc=%s, auto-window tests red: %d %s, passed %s\n' % (rc, len(failed), failed, passed))
open(OUT + '/arms_falsification.txt', 'w', encoding='utf-8', newline='\n').write('\n'.join(report))
print('\n'.join(report))
