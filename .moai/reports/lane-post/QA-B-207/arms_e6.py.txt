def sha(p):
    with open(p, 'rb') as f:
        return hashlib.sha256(f.read()).hexdigest()


def sh(cmd):
    r = subprocess.run(['cmd', '/c', cmd], capture_output=True, text=True, timeout=3400)
    return (r.stdout or '') + (r.stderr or '')


def failed(path):
    with open(path, 'rb') as f:
        raw = f.read()
    try:
        txt = raw.decode('utf-8')
    except UnicodeDecodeError:
        txt = raw.decode('cp949', errors='replace')
    return set(re.findall(r'^\[  FAILED  \] (\S+?)(?: \(|$)', txt, re.M))
