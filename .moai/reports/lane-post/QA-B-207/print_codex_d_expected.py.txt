import numpy as np

exact_path = sys.argv[2]
sys.argv = [sys.argv[0], sys.argv[1]]
src = open(exact_path, encoding='utf-8').read().split('FLT_MIN =')[0]
ns = {}
exec(compile(src, exact_path, 'exec'), ns)
block = np.array([np.float32(2.0 ** -100), 4095, 1, 1, 1, 1, 1, 1, 1, 1, 2, 3, 4, 5, 6, 7], np.float32).reshape(4, 4)
a = np.tile(block, (2, 2))
ref, rng = ns['reference'](a, 3.0, 2, 2)
print(', '.join(repr(float(x)) for x in ref.ravel()))
