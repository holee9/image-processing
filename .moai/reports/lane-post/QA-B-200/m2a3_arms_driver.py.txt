"""QA-B-200 M2a3 falsification arms (DICOM reader mAs validation, mismatch alert, xpe_dicom_version).
Break ONE defence, rebuild ci-dicom, run test_dicom_failure_paths.exe, expect the named tests red, restore byte-identical.
Run from the repository root: python build/g200_arms_m2a3.py <out.txt> [arm...]"""
import hashlib, re, subprocess, sys

OUT = sys.argv[1]
ONLY = sys.argv[2:]
BUILD = r'D:\workspace-github\xpe-post\build\g200-dicom-full.bat'
RUN = r'D:\workspace-github\xpe-post\build\g200-run2.bat'
RDR = 'modules/dicom/src/DicomReader.cpp'
DCM = 'modules/dicom/src/dicom.cpp'
T_GRAMMAR = 'AnIntegerStringMustMatchTheGrammarCompletelyAndNotJustByItsPrefix'
T_ALERT = 'ThreeExposureAttributesThatAgreeWithinTheirRoundingPostNoAlertAndThatDisagreePostOne'
T_VERSION = 'TheModuleExportsAVersionStringLikeTheOtherModules'

ARMS = [
    ('mismatch_never_alerts',
     [(RDR, '    if (!disagree) return;', '    if (!disagree || std::strlen("x") == 1) return;')], [T_ALERT]),
    ('tolerance_ignored_so_rounding_alerts',
     [(RDR, 'k.tolerance + chosen->tolerance + 1e-9', '1e-9')], [T_ALERT]),
    ('tolerance_too_wide_so_conflicts_pass',
     [(RDR, 'k.tolerance + chosen->tolerance + 1e-9', '1.0e9')], [T_ALERT]),
    ('is_accepts_a_valid_prefix',
     [(RDR, "        if (raw[b] < '0' || raw[b] > '9') return false;", "        if (raw[b] < '0' || raw[b] > '9') break;")], [T_GRAMMAR]),
    ('is_range_not_checked',
     [(RDR, '    if (v < -2147483648LL || v > 2147483647LL) return false;', '    if (v < -2147483648LL) return false;')], [T_GRAMMAR]),
    ('is_length_not_checked',
     [(RDR, '    if (raw.size() > 12) return false;', '    if (raw.size() > 1200) return false;')], [T_GRAMMAR]),
    ('is_padding_not_trimmed',
     [(RDR, "    while (b < e && raw[b] == ' ') ++b;", '')], [T_GRAMMAR]),
    ('multi_valued_attribute_accepted',
     [(RDR, 'return ds->findAndGetElement(key, e).good() && e && e->getVM() == 1;', 'return ds->findAndGetElement(key, e).good() && e;')], [T_GRAMMAR]),
    ('version_returns_empty',
     [(DCM, '''XPE_API const char* xpe_dicom_version(void) {
    return "1.0.0";''', '''XPE_API const char* xpe_dicom_version(void) {
    return "";''')], [T_VERSION]),
    ('version_not_semver',
     [(DCM, '''XPE_API const char* xpe_dicom_version(void) {
    return "1.0.0";''', '''XPE_API const char* xpe_dicom_version(void) {
    return "v1";''')], [T_VERSION]),
]


def sha(p):
    return hashlib.sha256(open(p, 'rb').read()).hexdigest()


def sh(cmd):
    r = subprocess.run(['cmd', '/c', cmd], capture_output=True, text=True, timeout=3000)
    return (r.stdout or '') + (r.stderr or '')


def failed(path):
    txt = open(path, encoding='utf-8', errors='replace').read()
    return set(re.findall(r'^\[  FAILED  \] (\S+?)(?: \(|$)', txt, re.M))


out = open(OUT, 'w', encoding='utf-8', newline='\n')


def log(s):
    out.write(s + '\n')
    out.flush()


for name, edits, must in ARMS:
    if ONLY and name not in ONLY:
        continue
    files = sorted({e[0] for e in edits})
    saved = {f: open(f, 'rb').read() for f in files}
    before = {f: sha(f) for f in files}
    log('=== ARM ' + name)
    try:
        for f in files:
            txt = saved[f].decode('utf-8')
            for ff, old, new in edits:
                if ff == f:
                    assert txt.count(old) == 1, (name, old[:60], txt.count(old))
                    txt = txt.replace(old, new)
            open(f, 'wb').write(txt.encode('utf-8'))
        sh(BUILD)
        if 'FAILED:' in open('build/g200-b4.txt', encoding='utf-8', errors='replace').read():
            log('RESULT: BUILD FAILED')
        else:
            sh('%s ci-dicom test_dicom_failure_paths.exe D:\\workspace-github\\xpe-post\\build\\g200-arm-m2a3.txt * dicom' % RUN)
            red = failed('build/g200-arm-m2a3.txt')
            hit = [m for m in must if any(m in f for f in red)]
            miss = [m for m in must if m not in hit]
            log('red tests: ' + (', '.join(sorted(red)) if red else '(none)'))
            log('RESULT: %s  (expected red: %d, red: %d, not red: %s)' % ('FIRED' if not miss else 'DID NOT FIRE', len(must), len(hit), miss))
    finally:
        for f in files:
            open(f, 'wb').write(saved[f])
        log('restored byte-identical: %s' % all(sha(f) == before[f] for f in files))
log('=== DONE')
out.close()
