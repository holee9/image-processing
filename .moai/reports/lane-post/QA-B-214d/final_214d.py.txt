import os, subprocess
os.chdir(r'D:/workspace-github/xpe-post')
R = r'.moai\reports\lane-post'


def sh(c):
    return subprocess.run(c, shell=True, capture_output=True, text=True)


sh(r'cmd.exe /c "%s\QA-B-214d\_build.bat build\g214d-final-build.txt"' % R)
print(open('build/g214d-final-build.txt', encoding='utf-8', errors='replace').read()[-200:])
for f in os.listdir('build'):
    if f.startswith('g214-final214d-'):
        os.remove('build/' + f)
sh(r'cmd.exe /c "%s\QA-B-214\_rundisp.bat final214d"' % R)
tot = 0
res = []
for f in sorted(os.listdir('build')):
    if f.startswith('g214-final214d-test_'):
        t = open('build/' + f, encoding='utf-8', errors='replace').read()
        ok = [l for l in t.splitlines() if '[  PASSED  ]' in l]
        bad = [l for l in t.splitlines() if 'FAILED' in l]
        n = int(ok[0].split()[3]) if ok else 0
        tot += n
        res.append('%s: %s | failed lines: %d' % (f[len('g214-final214d-'):-4], ok[0] if ok else 'NO PASSED LINE', len(bad)))
res.append('total passed: %d' % tot)
open('.moai/reports/lane-post/QA-B-214d/display_tests_results.txt', 'w', encoding='utf-8', newline='\n').write('\n'.join(res) + '\n')
print('\n'.join(res))
sh(r'cmd.exe /c "%s\QA-B-214\_rune2e.bat .moai\reports\lane-post\QA-B-214d\e2e_default_tests.txt"' % R)
t = open('.moai/reports/lane-post/QA-B-214d/e2e_default_tests.txt', encoding='utf-8', errors='replace').read()
print([l for l in t.splitlines() if 'PASSED' in l or 'FAILED' in l or 'EXIT' in l])
