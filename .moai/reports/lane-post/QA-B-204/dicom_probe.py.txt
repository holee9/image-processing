class Img(C.Structure):
    _fields_ = [('width', C.c_uint32), ('height', C.c_uint32), ('bitsAllocated', C.c_uint32), ('bitsStored', C.c_uint32),
                ('format', C.c_int32), ('data', C.c_void_p), ('dataSize', C.c_size_t)]


class Meta(C.Structure):
    _fields_ = [('bodyPart', C.c_char * 64), ('kVp', C.c_float), ('mAs', C.c_float), ('SID_mm', C.c_float),
                ('pixelPitch_mm', C.c_float), ('acquisitionTime', C.c_uint64), ('flags', C.c_uint32)]


RC = {0: 'OK', -1: 'INVALID_INPUT', -2: 'UNSUPPORTED_FORMAT', -3: 'PROCESSING_FAILED', -10: 'NETWORK_FAILED',
      -13: 'DICOM_INVALID', -14: 'DICOM_CONFORMANCE', -12: 'BUFFER_TOO_SMALL'}


def rcn(rc):
    return '%d (%s)' % (rc, RC.get(rc, '?'))


TMP = tempfile.mkdtemp(prefix='g204dcm_')

    print()

def write_file(name, acq=1700000000, dataSize=None, w=8, h=8):
    px = (np.arange(w * h, dtype=np.uint16) * 7).reshape(h, w).copy()
    img = Img(w, h, 16, 16, 0, px.ctypes.data, px.nbytes if dataSize is None else dataSize)
    meta = Meta(b'CHEST', 80.0, 2.5, 1800.0, 0.15, acq, 0)
    path = os.path.join(TMP, name)
    rc = D.xpe_dicom_write(path.encode(), C.byref(img), C.byref(meta))
    return rc, path


def open_rc(path):
    h = C.c_void_p()
    rc = D.xpe_dicom_open(path.encode(), C.byref(h))
    if rc == 0:
        D.xpe_dicom_close(h)
    return rc, bool(h.value)


def validate(path, cap=65536):
    buf = C.create_string_buffer(cap)
    rc = D.xpe_dicom_validate(path.encode(), buf, C.c_uint32(cap))
    return rc, buf.value.decode('utf-8', 'replace')

section('C6 REQ-DICOM-025: an invalid UID format -- in warnings AND errors, valid=false')
rc, p6 = write_file('c6.dcm')
raw = bytearray(open(p6, 'rb').read())
patched = []
for tagname, tag in (('StudyInstanceUID', (0x0020, 0x000D)), ('SeriesInstanceUID', (0x0020, 0x000E))):
    o = raw.find(struct.pack('<HH', *tag) + b'UI')
    ln = struct.unpack('<H', raw[o + 6:o + 8])[0]
    val = bytes(raw[o + 8:o + 8 + ln])
    if tagname == 'StudyInstanceUID':
        # replace the last digit of the value by a letter: not a legal UID character
        end = o + 8 + ln - 1
        while raw[end] in (0, 32):
            end -= 1
        raw[end] = ord('x')
        patched.append((tagname, val.rstrip(b'\x00').decode(), bytes(raw[o + 8:o + 8 + ln]).rstrip(b'\x00').decode()))
print('  patched: %s' % patched)
open(os.path.join(TMP, 'c6_baduid.dcm'), 'wb').write(bytes(raw))
rc3, rep3 = validate(os.path.join(TMP, 'c6_baduid.dcm'))
print('  rc %s' % rcn(rc3))
try:
    j3 = json.loads(rep3)
    print('  valid=%s; errors=%s; warnings=%s' % (j3['valid'], j3['errors'], j3['warnings']))

    print('  report is not JSON: %r (%s)' % (rep3[:300], e))

class Stall(threading.Thread):
    def __init__(self):
        super().__init__(daemon=True)
        self.srv = socket.socket()
        self.srv.bind(('127.0.0.1', 0))
        self.srv.listen(8)
        self.port = self.srv.getsockname()[1]
        self.conns = []
        self.stop = False

    def run(self):
        self.srv.settimeout(0.2)
        while not self.stop:
            try:
                c, _ = self.srv.accept()
                self.conns.append(c)
            except OSError:
                pass


import subprocess
import sys

if len(sys.argv) > 2 and sys.argv[1] == 'c10child':
    st = Stall()
    st.start()
    t = int(sys.argv[2])
    t0 = time.perf_counter()
    rc = D.xpe_dicom_cstore(b'127.0.0.1', st.port, b'XPE', p10.encode(), C.c_uint32(t))
    print('RESULT %.2f %s' % (time.perf_counter() - t0, rcn(rc)), flush=True)
    os._exit(0)

if len(sys.argv) > 2 and sys.argv[1] == 'c10cancel':
    st = Stall()
    st.start()
    t = int(sys.argv[2])
    box = {}

    def call():
        t0 = time.perf_counter()
        box['rc'] = D.xpe_dicom_cstore(b'127.0.0.1', st.port, b'XPE', p10.encode(), C.c_uint32(t))
        box['el'] = time.perf_counter() - t0

    th = threading.Thread(target=call, daemon=True)
    th.start()
    time.sleep(3.0)
    D.xpe_dicom_cancel()
    th.join(20)
    print('RESULT %s' % ('returned %.2f s after the call started, rc %s' % (box['el'], rcn(box['rc'])) if 'rc' in box else 'still hung 20 s after xpe_dicom_cancel'), flush=True)
    os._exit(0)

print('  (stalling peer: accept the TCP connection, read nothing, send nothing; each call runs in its own process, hard cap 40 s)')
print('  %-12s %-18s %s' % ('timeoutMs', 'returned after', 'rc'))
for t in (0, 300, 700, 999, 1000, 1500, 2500):
    try:
        r = subprocess.run([sys.executable, '-u', __file__, 'c10child', str(t)], capture_output=True, text=True, timeout=40)
        line = [l for l in r.stdout.splitlines() if l.startswith('RESULT')]
        el, rcs = (line[0].split(' ', 2)[1] + ' s', line[0].split(' ', 2)[2]) if line else ('?', 'no result: ' + r.stderr[-120:])
    except subprocess.TimeoutExpired:
        el, rcs = 'HUNG (> 40 s)', 'the call had not returned'
    print('  %-12d %-18s %s' % (t, el, rcs))
for t, cap in ((300, 130),):
    try:
        r = subprocess.run([sys.executable, '-u', __file__, 'c10child', str(t)], capture_output=True, text=True, timeout=cap)
        print('  %-12d returned (cap %d s)  %s' % (t, cap, [l for l in r.stdout.splitlines() if l.startswith('RESULT')]))
    except subprocess.TimeoutExpired:
        print('  %-12d HUNG (> %d s)        the call had not returned' % (t, cap))
r = subprocess.run([sys.executable, '-u', __file__, 'c10cancel', '300'], capture_output=True, text=True, timeout=60)
print('  timeoutMs 300, then xpe_dicom_cancel() from another thread after 3 s: ' + ' '.join(l for l in r.stdout.splitlines() if l.startswith('RESULT')))

TS = {b'1.2.840.10008.1.2': 'Implicit VR Little Endian', b'1.2.840.10008.1.2.1': 'Explicit VR Little Endian',

class Grab(threading.Thread):
    def __init__(self):
        super().__init__(daemon=True)
        self.srv = socket.socket()
        self.srv.bind(('127.0.0.1', 0))
        self.srv.listen(1)
        self.port = self.srv.getsockname()[1]
        self.data = b''

    def run(self):
        self.srv.settimeout(15)
        try:
            c, _ = self.srv.accept()
            c.settimeout(3)
            try:
                self.data = c.recv(65536)
            except OSError:
                pass
            c.close()
        except OSError:
            pass

def write_j2k(name):
    px = (np.arange(64, dtype=np.uint16) * 7).reshape(8, 8).copy()
    img = Img(8, 8, 16, 16, 0, px.ctypes.data, px.nbytes)
    meta = Meta(b'CHEST', 80.0, 2.5, 1800.0, 0.15, 1700000000, 0)
    path = os.path.join(TMP, name)
    return D.xpe_dicom_write_j2k(path.encode(), C.byref(img), C.byref(meta)), path


for label, name in (('a file written by xpe_dicom_write (Explicit LE)', 'c11.dcm'), ('a file written by xpe_dicom_write_j2k (J2K Lossless)', 'c11j.dcm')):
    rc, p = write_file(name) if name == 'c11.dcm' else write_j2k(name)
    ts = re.search(rb'1\.2\.840\.10008\.1\.2[\d.]*', open(p, 'rb').read()[132:])
    print('  (file transfer syntax: %s)' % (ts.group(0).decode() if ts else '?'))
    g = Grab()
    g.start()
    rc = D.xpe_dicom_cstore(b'127.0.0.1', g.port, b'XPE', p.encode(), C.c_uint32(3000))
    g.join(8)
    found = []
    for uid, nm in TS.items():
        for m in re.finditer(re.escape(uid) + rb'(?![\d.])', g.data):
            found.append((m.start(), nm))
    found.sort()
    print('  %s: cstore rc %s; A-ASSOCIATE-RQ %d bytes; proposed (in order): %s'
          % (label, rcn(rc), len(g.data), [n for _, n in found]))
