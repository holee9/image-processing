import subprocess, sys

R = r'D:\workspace-github\xpe-post\build\g198'
variants = sys.argv[1].split(',') if len(sys.argv) > 1 else ['base', 'job', 'restricted', 'wr', 'lowil', 'ac']
targets = sys.argv[2].split(',') if len(sys.argv) > 2 else ['probe', 'worker']
for v in variants:
    for t in targets:
        out = R + r'\out_%s_%s.txt' % (v, t)
        try:
            subprocess.run(['cmd', '/c', R + r'\run.bat', v, t, out], timeout=300, capture_output=True)
        except subprocess.TimeoutExpired:
            open(out, 'a').write('TIMEOUT\n')
        txt = open(out, encoding='utf-8', errors='replace').read()
        print('=====', v, t, len(txt.splitlines()), 'lines', flush=True)
