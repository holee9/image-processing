"""QA-B-206 M2e falsification arms (the Pixel Data rule under JPIP, PS3.5 A.6; two M2d arms re-run against the new table): the Pixel Data Provider URL is judged by the transfer syntax, and excludes Pixel Data.
Break ONE defence, rebuild ci-dicom (test_dicom_validator), run it, expect the named tests red, restore byte-identical.
An arm with no expected test is an EQUIVALENT-MUTANT probe: it records what goes red (nothing expected) instead of asserting.
Do not touch the source tree while this runs.
Run: python build/g207_arms_m2e.py <out.txt> [arm name]"""
import hashlib
import re
import subprocess
import sys

OUT = sys.argv[1]
ONLY = sys.argv[2] if len(sys.argv) > 2 else None
BUILD = r'D:\workspace-github\xpe-post\build\g206-build.bat test_dicom_validator'
RUN = r'D:\workspace-github\xpe-post\build\g206-run.bat'
VAL = 'modules/dicom/src/DicomValidator.cpp'
NL = '\n'
T_TABLE = 'DicomValidatorTest.PixelDataProviderUrlIsJudgedByTheTransferSyntaxAndExcludesPixelData'
T_DEFL = 'DicomValidatorTest.KnownDivergence_JpipReferencedDeflateCannotBeParsedByThisDcmtkBuild'

TS_EXPR = 'transferSyntax == "1.2.840.10008.1.2.4.94" || transferSyntax == "1.2.840.10008.1.2.4.95"'
GIVEN_IF = 'if (urlGiven && jpipReferenced) {'
EXCL_IF = 'if (req.key == DCM_PixelData && ds->tagExists(DCM_PixelDataProviderURL)) {'
PUSH = '                warnings.push_back(warnEntry);' + NL
PUSH_CONT = PUSH + '                continue;' + NL
HASVAL = 'elementHasValue(ds, DCM_PixelDataProviderURL)'

JPIP_IF = '        if (req.key == DCM_PixelData && jpipReferenced) {' + NL
ARMS = [
    ('jpip_pixel_data_rule_never_reported', [(JPIP_IF, '        if (req.key == DCM_PixelData && jpipReferenced && ds == nullptr) {' + NL)], [T_TABLE]),
    ('jpip_pixel_data_rule_applied_to_every_syntax', [(JPIP_IF, '        if (req.key == DCM_PixelData && (jpipReferenced || !transferSyntax.empty())) {' + NL)], [T_TABLE]),
    ('jpip_pixel_data_rule_reported_only_when_there_is_no_url',
     [(JPIP_IF, '        if (req.key == DCM_PixelData && jpipReferenced && !ds->tagExists(DCM_PixelDataProviderURL)) {' + NL)], [T_TABLE]),
    ('exclusion_never_reported', [(EXCL_IF, 'if (req.key == DCM_PixelData && ds->tagExists(DCM_PixelDataProviderURL) && ds == nullptr) {')], [T_TABLE]),
    ('jpip_syntax_never_recognised', [(TS_EXPR, '(' + TS_EXPR + ') && transferSyntax.empty()')], [T_TABLE]),
]


def sha(p):
    with open(p, 'rb') as f:
        return hashlib.sha256(f.read()).hexdigest()


def sh(cmd):
    r = subprocess.run(['cmd', '/c', cmd], capture_output=True, text=True, timeout=3000)
    return (r.stdout or '') + (r.stderr or '')


def failed(path):
    with open(path, 'rb') as f:
        raw = f.read()
    try:
        txt = raw.decode('utf-8')
    except UnicodeDecodeError:
        txt = raw.decode('cp949', errors='replace')
    return set(re.findall(r'^\[  FAILED  \] (\S+?)(?: \(|$)', txt, re.M))


out = open(OUT, 'w', encoding='utf-8', newline='\n')


def log(s):
    out.write(s + '\n')
    out.flush()


for name, edits, must in ARMS:
    if ONLY and name != ONLY:
        continue
    with open(VAL, 'rb') as f:
        saved = f.read()
    before = sha(VAL)
    log('=== ARM ' + name)
    try:
        txt = saved.decode('utf-8')
        for old, new in edits:
            n = txt.count(old)
            assert n == 1, (name, old[:70], n)
            txt = txt.replace(old, new)
        with open(VAL, 'wb') as f:
            f.write(txt.encode('utf-8'))
        sh(BUILD)
        with open('build/g206-build.txt', 'rb') as f:
            built = b'FAILED:' not in f.read()
        if not built:
            log('RESULT: BUILD FAILED (not measured)')
        else:
            sh('%s test_dicom_validator.exe * D:\\workspace-github\\xpe-post\\build\\g207-arm.txt' % RUN)
            red = failed('build/g207-arm.txt')
            log('red tests: ' + (', '.join(sorted(red)) if red else '(none)'))
            hit = [m for m in must if m in red]
            miss = [m for m in must if m not in hit]
            if not must:
                log('RESULT: EQUIVALENT-MUTANT PROBE  (nothing expected; red: %d)' % len(red))
            else:
                log('RESULT: %s  (expected red: %d, red: %d, not red: %s)' % ('FIRED' if not miss else 'DID NOT FIRE', len(must), len(hit), miss))
    finally:
        with open(VAL, 'wb') as f:
            f.write(saved)
        log('restored byte-identical: %s' % (sha(VAL) == before))
log('=== DONE')
out.close()
