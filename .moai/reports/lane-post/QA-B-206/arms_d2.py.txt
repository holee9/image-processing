NL = '\n'

def sha(p):
    with open(p, 'rb') as f:
        return hashlib.sha256(f.read()).hexdigest()


def sh(cmd):
    r = subprocess.run(['cmd', '/c', cmd], capture_output=True, text=True, timeout=3000)
    return (r.stdout or '') + (r.stderr or '')


def failed(path):
    with open(path, encoding='utf-8', errors='replace') as f:
        txt = f.read()
    return set(re.findall(r'^\[  FAILED  \] (\S+?)(?: \(|$)', txt, re.M))


out = open(OUT, 'w', encoding='utf-8', newline='\n')


def log(s):
    out.write(s + '\n')
    out.flush()


for name, edits, must in ARMS:
    with open(SRC, 'rb') as f:
        saved = f.read()
    before = sha(SRC)
    log('=== ARM ' + name)
    try:
        txt = saved.decode('utf-8')
        for old, new in edits:
            assert txt.count(old) == 1, (name, old[:60], txt.count(old))
            txt = txt.replace(old, new)
        with open(SRC, 'wb') as f:
            f.write(txt.encode('utf-8'))
        sh(BUILD)
        with open('build/g200-b5.txt', encoding='utf-8', errors='replace') as f:
            built = 'FAILED:' not in f.read()
        if not built:
            log('RESULT: BUILD FAILED')
        else:
            sh('%s ci-post test_display_modality_lut.exe D:\\workspace-github\\xpe-post\\build\\g206-arm.txt * display' % RUN)
            red = failed('build/g206-arm.txt')
            log('red tests: ' + (', '.join(sorted(red)) if red else '(none)'))
            if must is None:
                log('RESULT: observation only (no expectation set)')
            else:
                hit = [m for m in must if m in red]
                miss = [m for m in must if m not in hit]
                log('RESULT: %s  (expected red: %d, red: %d, not red: %s)' % ('FIRED' if not miss else 'DID NOT FIRE', len(must), len(hit), miss))
    finally:
        with open(SRC, 'wb') as f:
            f.write(saved)
        log('restored byte-identical: %s' % (sha(SRC) == before))
log('=== DONE')
out.close()
