"""QA-B-206 M1b falsification arms: what the public DICOM writers accept (size, descriptor). Break ONE defence, rebuild ci-dicom,
run test_dicom_writer, expect the named tests red, restore byte-identical.   Run: python build/g206_arms_m1b.py <out.txt>"""
import hashlib
import re
import subprocess
import sys

OUT = sys.argv[1]
BUILD = r'D:\workspace-github\xpe-post\build\g206-build.bat test_dicom_writer'
RUN = r'D:\workspace-github\xpe-post\build\g206-run.bat'
DC = 'modules/dicom/src/dicom.cpp'
WR = 'modules/dicom/src/DicomWriter.cpp'
LIM = 'modules/dicom/src/DicomImageLimits.h'
NL = '\n'
T_LIM = 'DicomImageLimits.TheLengthIsComputedIn64BitsAndTheBoundaryIsWhereTheElementEnds'
T_SIZE = 'DicomWriterTest.AnImageTheFileCannotDescribeIsRefusedBeforeAnyFileExistsByBothWriters'
T_BITS = 'DicomWriterTest.ADescriptorThatContradictsTheSixteenBitWordsIsRefusedByBothWritersAndNoFileIsMade'
T_OK = 'DicomWriterTest.EveryBitsStoredFromOneToSixteenIsWrittenAndReadsBack'

BITS_FN = '    return img != nullptr && xpe::dicom::image_bits_are_writable(img->bitsAllocated, img->bitsStored);' + NL
BITS_DOOR = '    if (!bits_are_writable(img)) return XPE_ERR_INVALID_INPUT;' + NL
SIZE_DOOR = '    if (!size_is_representable(img)) return XPE_ERR_INVALID_INPUT;' + NL
SIZE_INNER_W = '    if (!image_size_is_representable(img->width, img->height)) return XPE_ERR_INVALID_INPUT;' + NL
SIZE_INNER_J = '    if (!image_size_is_representable(img->width, img->height)) return XPE_ERR_INVALID_INPUT;   // QA-B-206 M1b' + NL
BITS_PRED = '    return bitsAllocated == 16u && bitsStored >= 1u && bitsStored <= 16u;' + NL
BYTES = '    return static_cast<uint64_t>(width) * static_cast<uint64_t>(height) * kBytesPerPixel;' + NL
RC_CHECK = '    if (width > kMaxRowsOrColumns || height > kMaxRowsOrColumns) return false;' + NL

ARMS = [
    ('bits_door_removed_from_both_public_writers', [(DC, BITS_DOOR, '')], [T_BITS]),
    ('bits_allocated_8_also_accepted', [(LIM, BITS_PRED, '    return (bitsAllocated == 16u || bitsAllocated == 8u) && bitsStored >= 1u && bitsStored <= 16u;' + NL)], [T_BITS]),
    ('bits_stored_0_accepted', [(LIM, BITS_PRED, '    return bitsAllocated == 16u && bitsStored <= 16u;' + NL)], [T_BITS]),
    ('bits_stored_up_to_32_accepted', [(LIM, BITS_PRED, '    return bitsAllocated == 16u && bitsStored >= 1u && bitsStored <= 32u;' + NL)], [T_BITS]),
    ('bits_allocated_0_means_16_again', [(LIM, BITS_PRED, '    return (bitsAllocated == 16u || bitsAllocated == 0u) && bitsStored >= 1u && bitsStored <= 16u;' + NL)], [T_BITS]),
    ('bits_only_one_to_fifteen_accepted', [(LIM, BITS_PRED, '    return bitsAllocated == 16u && bitsStored >= 1u && bitsStored <= 15u;' + NL)], [T_OK]),
    ('product_back_in_32_bits', [(LIM, BYTES, '    return static_cast<uint64_t>(static_cast<uint32_t>(width * height * kBytesPerPixel));' + NL)], [T_LIM, T_SIZE]),
    ('rows_columns_limit_removed', [(LIM, RC_CHECK, '')], [T_LIM, T_SIZE]),
    ('size_door_removed_from_the_public_writers_only_the_inner_checks_remain', [(DC, SIZE_DOOR, '')], None),
    ('size_door_and_the_inner_write_check_removed_j2k_inner_check_remains', [(DC, SIZE_DOOR, ''), (WR, SIZE_INNER_W, '')], None),
]


def sha(p):
    with open(p, 'rb') as f:
        return hashlib.sha256(f.read()).hexdigest()


def sh(cmd):
    r = subprocess.run(['cmd', '/c', cmd], capture_output=True, text=True, timeout=3000)
    return (r.stdout or '') + (r.stderr or '')


def failed(path):
    with open(path, encoding='utf-8', errors='replace') as f:
        txt = f.read()
    return set(re.findall(r'^\[  FAILED  \] (\S+?)(?: \(|$)', txt, re.M))


out = open(OUT, 'w', encoding='utf-8', newline='\n')


def log(s):
    out.write(s + '\n')
    out.flush()


for name, edits, must in ARMS:
    files = sorted({e[0] for e in edits})
    saved = {}
    for p in files:
        with open(p, 'rb') as f:
            saved[p] = f.read()
    before = {p: sha(p) for p in files}
    log('=== ARM ' + name)
    try:
        for p in files:
            txt = saved[p].decode('utf-8')
            for ff, old, new in edits:
                if ff == p:
                    n = txt.count(old)
                    assert n >= 1, (name, old[:60], n)
                    txt = txt.replace(old, new)
            with open(p, 'wb') as f:
                f.write(txt.encode('utf-8'))
        sh(BUILD)
        with open('build/g206-build.txt', encoding='utf-8', errors='replace') as f:
            built = 'FAILED:' not in f.read()
        if not built:
            log('RESULT: BUILD FAILED')
        else:
            sh('%s test_dicom_writer.exe * D:\\workspace-github\\xpe-post\\build\\g206-arm.txt' % RUN)
            red = failed('build/g206-arm.txt')
            with open('build/g206-arm.txt', encoding='utf-8', errors='replace') as f:
                tail = f.read()
            crashed = 'PASSED' not in tail and 'FAILED TEST' not in tail
            log('red tests: ' + (', '.join(sorted(red)) if red else '(none)') + ('   [the test executable did not finish: it crashed]' if crashed else ''))
            if must is None:
                log('RESULT: observation only (no expectation set)')
            else:
                hit = [m for m in must if m in red or crashed]
                miss = [m for m in must if m not in hit]
                log('RESULT: %s  (expected red: %d, red: %d, not red: %s)' % ('FIRED' if not miss else 'DID NOT FIRE', len(must), len(hit), miss))
    finally:
        for p in files:
            with open(p, 'wb') as f:
                f.write(saved[p])
        log('restored byte-identical: %s' % all(sha(p) == before[p] for p in files))
log('=== DONE')
out.close()
