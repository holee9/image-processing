"""QA-B-206 M1c falsification arms: the J2K fragment length and the inner writers' predicates. Break ONE defence, rebuild
ci-dicom (test_dicom_writer + test_dicom_writer_inner), run both, expect the named tests red, restore byte-identical.
Run: python build/g206_arms_m1c.py <out.txt>"""
import hashlib
import re
import subprocess
import sys

OUT = sys.argv[1]
BUILD = r'D:\workspace-github\xpe-post\build\g206-build.bat'
RUN = r'D:\workspace-github\xpe-post\build\g206-run.bat'
WR = 'modules/dicom/src/DicomWriter.cpp'
LIM = 'modules/dicom/src/DicomImageLimits.h'
NL = '\n'
T_FRAG = 'DicomImageLimits.AJ2kFragmentLengthIsNarrowedOnlyWhenItFitsAndNeverWraps'
T_FMT = 'DicomWriterInner.APixelFormatOtherThanUint16IsRefusedWithoutTheDoor'
T_BITS = 'DicomWriterInner.ADescriptorThatContradictsTheSixteenBitWordsIsRefusedWithoutTheDoor'
T_BIG = 'DicomWriterInner.AnImageNoFileCanDescribeIsRefusedWithoutTheDoorAndTheTinyBufferIsNotRead'

NARROW_IF = '    if (bytes > kMaxPixelDataBytes) return false;' + NL
NARROW_OUT = '    if (out != nullptr) *out = static_cast<uint32_t>(bytes);' + NL
INNER_W = ('    if (!image_format_is_writable(img->format) || !image_bits_are_writable(img->bitsAllocated, img->bitsStored)) {' + NL +
           '        return XPE_ERR_INVALID_INPUT;' + NL + '    }' + NL)
INNER_J = ('    if (!image_format_is_writable(img->format) || !image_bits_are_writable(img->bitsAllocated, img->bitsStored)) {' + NL +
           '        return XPE_ERR_INVALID_INPUT;   // QA-B-206 M1c' + NL + '    }' + NL)
EARLY = ('    if (!narrow_fragment_length(j2kData.size(), nullptr)) {' + NL +
         '        spdlog::warn("[DicomWriter] J2K bitstream of {} bytes does not fit one fragment", j2kData.size());' + NL +
         '        return XPE_ERR_PROCESSING_FAILED;' + NL + '    }' + NL)
SETJ = '    if (!narrow_fragment_length(j2kData.size(), &fragmentLength)) return XPE_ERR_PROCESSING_FAILED;' + NL

ARMS = [
    ('fragment_length_narrowed_by_a_plain_cast', [(LIM, NARROW_IF, '')], [T_FRAG]),
    ('fragment_limit_one_too_low', [(LIM, NARROW_IF, '    if (bytes >= kMaxPixelDataBytes) return false;' + NL)], [T_FRAG]),
    ('fragment_limit_accepts_the_undefined_length_marker', [(LIM, NARROW_IF, '    if (bytes > 0xFFFFFFFFull) return false;' + NL)], [T_FRAG]),
    ('fragment_output_written_even_when_refused', [(LIM, NARROW_IF + NARROW_OUT, NARROW_OUT + NARROW_IF)], [T_FRAG]),
    ('inner_write_format_and_bits_check_removed', [(WR, INNER_W, '')], [T_FMT, T_BITS]),
    ('inner_writeJ2K_format_and_bits_check_removed', [(WR, INNER_J, '')], [T_FMT, T_BITS]),
    ('inner_write_checks_the_format_only', [(WR, INNER_W, '    if (!image_format_is_writable(img->format)) return XPE_ERR_INVALID_INPUT;' + NL)], [T_BITS]),
    ('inner_write_checks_the_bits_only', [(WR, INNER_W, '    if (!image_bits_are_writable(img->bitsAllocated, img->bitsStored)) return XPE_ERR_INVALID_INPUT;' + NL)], [T_FMT]),
    ('writeJ2K_early_fragment_check_removed_setJ2KPixelData_keeps_its_own', [(WR, EARLY, '')], None),
    ('setJ2KPixelData_check_removed_the_early_check_in_writeJ2K_remains', [(WR, SETJ, '    fragmentLength = static_cast<uint32_t>(j2kData.size());' + NL)], None),
]


def sha(p):
    with open(p, 'rb') as f:
        return hashlib.sha256(f.read()).hexdigest()


def sh(cmd):
    r = subprocess.run(['cmd', '/c', cmd], capture_output=True, text=True, timeout=3000)
    return (r.stdout or '') + (r.stderr or '')


def failed(path):
    with open(path, encoding='utf-8', errors='replace') as f:
        txt = f.read()
    return set(re.findall(r'^\[  FAILED  \] (\S+?)(?: \(|$)', txt, re.M))


out = open(OUT, 'w', encoding='utf-8', newline='\n')


def log(s):
    out.write(s + '\n')
    out.flush()


for name, edits, must in ARMS:
    files = sorted({e[0] for e in edits})
    saved = {}
    for p in files:
        with open(p, 'rb') as f:
            saved[p] = f.read()
    before = {p: sha(p) for p in files}
    log('=== ARM ' + name)
    try:
        for p in files:
            txt = saved[p].decode('utf-8')
            for ff, old, new in edits:
                if ff == p:
                    n = txt.count(old)
                    assert n == 1, (name, old[:70], n)
                    txt = txt.replace(old, new)
            with open(p, 'wb') as f:
                f.write(txt.encode('utf-8'))
        built = True
        for target in ('test_dicom_writer', 'test_dicom_writer_inner'):
            sh(BUILD + ' ' + target)
            with open('build/g206-build.txt', encoding='utf-8', errors='replace') as f:
                built = built and ('FAILED:' not in f.read())
        if not built:
            log('RESULT: BUILD FAILED')
        else:
            red = set()
            for exe in ('test_dicom_writer.exe', 'test_dicom_writer_inner.exe'):
                sh('%s %s * D:\\workspace-github\\xpe-post\\build\\g206-arm.txt' % (RUN, exe))
                red |= failed('build/g206-arm.txt')
            log('red tests: ' + (', '.join(sorted(red)) if red else '(none)'))
            if must is None:
                log('RESULT: observation only (no expectation set)')
            else:
                hit = [m for m in must if m in red]
                miss = [m for m in must if m not in hit]
                log('RESULT: %s  (expected red: %d, red: %d, not red: %s)' % ('FIRED' if not miss else 'DID NOT FIRE', len(must), len(hit), miss))
    finally:
        for p in files:
            with open(p, 'wb') as f:
                f.write(saved[p])
        log('restored byte-identical: %s' % all(sha(p) == before[p] for p in files))
log('=== DONE')
out.close()
