def classes(rows):
    out = []

def q(r):
    """class-qualified names of the verified tests of a requirement, as markdown code spans"""
    seen, out = set(), []
    for n, ok, loc in m[r]:
        for l in loc[:1] if not n.endswith("*") else loc[:1]:
            cls = l.split("@")[-1].split(",")[0] if "@" in l else l
            key = (cls, n)
            if key not in seen:
                seen.add(key)
                out.append(f"`{cls}.{n}`")
    return out
