import ctypes, ctypes.wintypes as w, subprocess, os, time, sys, statistics

u = ctypes.windll.user32
WNDENUMPROC = ctypes.WINFUNCTYPE(w.BOOL, w.HWND, w.LPARAM)
u.GetWindowThreadProcessId.argtypes = [w.HWND, ctypes.POINTER(w.DWORD)]
u.SendMessageTimeoutW.argtypes = [w.HWND, w.UINT, w.WPARAM, w.LPARAM, w.UINT, w.UINT, ctypes.POINTER(ctypes.c_size_t)]
u.SendMessageTimeoutW.restype = ctypes.c_void_p
u.IsWindowVisible.argtypes = [w.HWND]
u.GetWindowTextLengthW.argtypes = [w.HWND]


def top_windows(pid):
    found = []

    def cb(hwnd, _):
        p = w.DWORD()
        u.GetWindowThreadProcessId(hwnd, ctypes.byref(p))
        if p.value == pid:
            buf = ctypes.create_unicode_buffer(80)
            u.GetClassNameW(hwnd, buf, 80)
            if buf.value.startswith('HwndWrapper[ImageProcTest'):
                found.append((hwnd, bool(u.IsWindowVisible(hwnd))))
        return True

    u.EnumWindows(WNDENUMPROC(cb), 0)
    return found


exe, nd, runs, label = sys.argv[1], sys.argv[2], int(sys.argv[3]), sys.argv[4]
env = dict(os.environ, XPE_NATIVE_DIR=nd)
print(f"## {label}")
print("run  handle_ms  visible_ms  ready_ms  unanswered_total_ms(0-4s)  longest_unanswered_ms")
for r in range(runs):
    p = subprocess.Popen([exe], env=env, stdin=subprocess.DEVNULL, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
    t0 = time.perf_counter()
    t_handle = t_visible = t_ready = None
    streak = 0
    un_total = 0.0
    longest = 0.0
    last = t0
    while time.perf_counter() - t0 < 4:
        now = time.perf_counter()
        wins = top_windows(p.pid)
        if wins:
            hwnd = wins[0][0]
            if t_handle is None:
                t_handle = (now - t0) * 1000
            if t_visible is None and any(v for _, v in wins):
                t_visible = (now - t0) * 1000
            res = ctypes.c_size_t()
            a = time.perf_counter()
            ok = u.SendMessageTimeoutW(hwnd, 0, 0, 0, 1, 30, ctypes.byref(res))
            b = time.perf_counter()
            if ok:
                streak += 1
                if t_visible is not None and t_ready is None and streak >= 10:
                    t_ready = (b - t0) * 1000
            else:
                streak = 0
                span = (b - a) * 1000
                un_total += span
                longest = max(longest, span)
        time.sleep(0.003)
    print(f"{r+1:>3}  {t_handle or -1:>9.0f}  {t_visible or -1:>10.0f}  {t_ready or -1:>8.0f}  {un_total:>25.0f}  {longest:>20.0f}")
    p.kill(); p.wait()
    subprocess.run(["taskkill", "/F", "/IM", "ImageProcTest.exe"], capture_output=True)
    time.sleep(1.2)
