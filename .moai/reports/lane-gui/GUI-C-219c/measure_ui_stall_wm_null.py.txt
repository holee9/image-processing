import ctypes, ctypes.wintypes as w, subprocess, os, time, sys, statistics

u = ctypes.windll.user32
EnumWindows = u.EnumWindows
WNDENUMPROC = ctypes.WINFUNCTYPE(w.BOOL, w.HWND, w.LPARAM)
u.GetWindowThreadProcessId.argtypes = [w.HWND, ctypes.POINTER(w.DWORD)]
u.SendMessageTimeoutW.argtypes = [w.HWND, w.UINT, w.WPARAM, w.LPARAM, w.UINT, w.UINT, ctypes.POINTER(ctypes.c_size_t)]
u.SendMessageTimeoutW.restype = ctypes.c_void_p
u.IsWindowVisible.argtypes = [w.HWND]
SMTO_BLOCK = 0x1
WM_NULL = 0


def windows_of(pid):
    found = []
    def cb(hwnd, _):
        p = w.DWORD()
        u.GetWindowThreadProcessId(hwnd, ctypes.byref(p))
        if p.value == pid and u.IsWindowVisible(hwnd):
            found.append(hwnd)
        return True
    EnumWindows(WNDENUMPROC(cb), 0)
    return found


exe, nd, runs, label = sys.argv[1], sys.argv[2], int(sys.argv[3]), sys.argv[4]
env = dict(os.environ, XPE_NATIVE_DIR=nd)
print(f"## {label}")
print("run  window_first_visible_ms  longest_UI_stall_ms (WM_NULL not answered)  stalls>1000ms  stalls>2000ms  total_stalled_ms")
maxes = []
for r in range(runs):
    p = subprocess.Popen([exe], env=env, stdin=subprocess.DEVNULL, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
    t0 = time.perf_counter()
    first = None
    last_ok = None
    stalls = []
    stall_start = None
    while time.perf_counter() - t0 < 8:
        now = (time.perf_counter() - t0) * 1000
        wins = windows_of(p.pid)
        if wins:
            if first is None:
                first = now
                last_ok = now
            res = ctypes.c_size_t()
            ok = u.SendMessageTimeoutW(wins[0], WM_NULL, 0, 0, SMTO_BLOCK, 50, ctypes.byref(res))
            now2 = (time.perf_counter() - t0) * 1000
            if ok:
                if stall_start is not None:
                    stalls.append(now2 - stall_start)
                    stall_start = None
                last_ok = now2
            elif stall_start is None:
                stall_start = last_ok
        time.sleep(0.003)
    if stall_start is not None:
        stalls.append((time.perf_counter() - t0) * 1000 - stall_start)
    mx = max(stalls, default=0)
    maxes.append(mx)
    print(f"{r+1:>3}  {first if first is not None else -1:>22.0f}  {mx:>40.0f}  {sum(1 for s in stalls if s > 1000):>13}  {sum(1 for s in stalls if s > 2000):>13}  {sum(stalls):>16.0f}")
    p.kill(); p.wait()
    subprocess.run(["taskkill", "/F", "/IM", "ImageProcTest.exe"], capture_output=True)
    time.sleep(1.5)
print(f"longest stall: median {statistics.median(maxes):.0f} ms, max {max(maxes):.0f} ms over {runs} launches")
